#include "kernel_operator.h"
#include "tiling.h"

class AddRmsNormBiasKernel {
public:
    __aicore__ inline AddRmsNormBiasKernel(AscendC::TPipe *pipe) : pipe_(pipe) {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma,
                                GM_ADDR bias, GM_ADDR y,
                                const __gm__ AddRmsNormBiasTilingData *tiling)
    {
        tiling_ = tiling;
        const uint32_t block = AscendC::GetBlockIdx();
        const uint32_t blocks = (tiling_->rows + tiling_->rowsPerTile - 1) / tiling_->rowsPerTile;
        rowsThisBlock_ = 0;
        if (block < blocks) {
            const uint32_t start = block * tiling_->rowsPerTile;
            rowsThisBlock_ = (start < tiling_->rows) ?
                ((tiling_->rows - start < tiling_->rowsPerTile) ?
                    tiling_->rows - start : tiling_->rowsPerTile) : 0;
            xGm_.SetGlobalBuffer((__gm__ float *)x + start * tiling_->d, rowsThisBlock_ * tiling_->d);
            residualGm_.SetGlobalBuffer((__gm__ float *)residual + start * tiling_->d,
                                        rowsThisBlock_ * tiling_->d);
            yGm_.SetGlobalBuffer((__gm__ float *)y + start * tiling_->d, rowsThisBlock_ * tiling_->d);
        }
        gammaGm_.SetGlobalBuffer((__gm__ float *)gamma, tiling_->d);
        biasGm_.SetGlobalBuffer((__gm__ float *)bias, tiling_->d);

        const uint32_t rowBytes = tiling_->dAligned * sizeof(float);
        pipe_->InitBuffer(xQueue_, 1, rowsThisBlock_ * rowBytes);
        pipe_->InitBuffer(residualQueue_, 1, rowsThisBlock_ * rowBytes);
        pipe_->InitBuffer(yQueue_, 1, rowsThisBlock_ * rowBytes);
        pipe_->InitBuffer(gammaBuf_, 1, rowBytes);
        pipe_->InitBuffer(biasBuf_, 1, rowBytes);
        pipe_->InitBuffer(squareBuf_, rowBytes);
        pipe_->InitBuffer(reduceBuf_, 32);
    }

    __aicore__ inline void Process()
    {
        if (rowsThisBlock_ == 0) return;
        CopyInVectors();
        ComputeRows();
        CopyOut();
    }

private:
    __aicore__ inline void CopyInVectors()
    {
        auto xLocal = xQueue_.AllocTensor<float>();
        auto rLocal = residualQueue_.AllocTensor<float>();
        auto gLocal = gammaBuf_.AllocTensor<float>();
        auto bLocal = biasBuf_.AllocTensor<float>();
        const uint32_t bytes = tiling_->d * sizeof(float);
        const uint32_t pad = tiling_->dAligned - tiling_->d;
        AscendC::DataCopyExtParams xCopy{static_cast<uint16_t>(rowsThisBlock_), static_cast<uint32_t>(bytes), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> xPad{true, 0, static_cast<uint8_t>(pad), 0.0f};
        AscendC::DataCopyPad(xLocal, xGm_, xCopy, xPad);
        AscendC::DataCopyPad(rLocal, residualGm_, xCopy, xPad);
        AscendC::DataCopyExtParams vCopy{1, bytes, 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> vPad{true, 0, static_cast<uint8_t>(pad), 0.0f};
        AscendC::DataCopyPad(gLocal, gammaGm_, vCopy, vPad);
        AscendC::DataCopyPad(bLocal, biasGm_, vCopy, vPad);
        xQueue_.EnQue(xLocal);
        residualQueue_.EnQue(rLocal);
        gammaBuf_.EnQue(gLocal);
        biasBuf_.EnQue(bLocal);
    }

    __aicore__ inline void ComputeRows()
    {
        auto x = xQueue_.DeQue<float>();
        auto residual = residualQueue_.DeQue<float>();
        auto gamma = gammaBuf_.DeQue<float>();
        auto bias = biasBuf_.DeQue<float>();
        auto y = yQueue_.AllocTensor<float>();
        auto square = squareBuf_.Get<float>();
        auto reduceTmp = reduceBuf_.Get<float>();
        const uint32_t stride = tiling_->dAligned;
        const int32_t d = tiling_->d;
        for (uint32_t row = 0; row < rowsThisBlock_; ++row) {
            auto xRow = x[row * stride];
            auto rRow = residual[row * stride];
            auto yRow = y[row * stride];
            auto sqRow = square[row * stride];
            AscendC::Add<float>(yRow, xRow, rRow, d);
            AscendC::Mul<float>(sqRow, yRow, yRow, d);
            auto sum = reduceBuf_.Get<float>();
            AscendC::ReduceSum<float, true>(sum, sqRow, reduceTmp, d);
            const float dValue = d;
            AscendC::Adds<float>(sum, sum, tiling_->epsilon * dValue, 1);
            AscendC::Muls<float>(sum, sum, 1.0f / dValue, 1);
            AscendC::Sqrt<float>(sum, sum, 1);
            const float inv = 1.0f / sum.GetValue(0);
            AscendC::Muls<float>(yRow, yRow, inv, d);
            AscendC::Mul<float>(yRow, yRow, gamma, d);
            AscendC::Add<float>(yRow, yRow, bias, d);
        }
        yQueue_.EnQue(y);
        xQueue_.FreeTensor(x);
        residualQueue_.FreeTensor(residual);
        gammaBuf_.FreeTensor(gamma);
        biasBuf_.FreeTensor(bias);
    }

    __aicore__ inline void CopyOut()
    {
        auto y = yQueue_.DeQue<float>();
        AscendC::DataCopyExtParams copy{static_cast<uint16_t>(rowsThisBlock_), static_cast<uint32_t>(tiling_->d * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPad(yGm_, y, copy);
        yQueue_.FreeTensor(y);
    }

    AscendC::TPipe *pipe_;
    const __gm__ AddRmsNormBiasTilingData *tiling_;
    AscendC::GlobalTensor<float> xGm_, residualGm_, gammaGm_, biasGm_, yGm_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> xQueue_, residualQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gammaBuf_, biasBuf_;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> yQueue_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> squareBuf_, reduceBuf_;
    uint32_t rowsThisBlock_ = 0;
};

__aicore__ inline void add_rms_norm_bias_kernel_impl(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR y, GM_ADDR tiling)
{
    AscendC::TPipe pipe;
    AddRmsNormBiasKernel op(&pipe);
    op.Init(x, residual, gamma, bias, y,
            (__gm__ AddRmsNormBiasTilingData *)tiling);
    op.Process();
}

extern "C" __global__ __vector__ void add_rms_norm_bias_kernel(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR y, GM_ADDR tiling)
{
    add_rms_norm_bias_kernel_impl(x, residual, gamma, bias, y, tiling);
}

extern "C" __global__ __vector__ void run_kernel(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR y, GM_ADDR tiling)
{
    add_rms_norm_bias_kernel_impl(x, residual, gamma, bias, y, tiling);
}
