// ============================================================================
// data_utils.h - File I/O utilities for Ascend C Kernel direct invocation
// ============================================================================

#ifndef DATA_UTILS_H
#define DATA_UTILS_H

#include <fcntl.h>
#include <sys/stat.h>
#include <unistd.h>
#include <stddef.h>

inline bool ReadFile(const char *filePath, size_t bufferSize, void *buffer, size_t bufferLen)
{
    if (buffer == nullptr) {
        return false;
    }
    if (bufferSize > bufferLen) {
        return false;
    }

    int fd = open(filePath, O_RDONLY);
    if (fd < 0) {
        return false;
    }

    struct stat fileStat;
    if (fstat(fd, &fileStat) != 0 || static_cast<size_t>(fileStat.st_size) != bufferSize) {
        close(fd);
        return false;
    }

    size_t totalRead = 0;
    while (totalRead < bufferSize) {
        ssize_t r = read(fd, static_cast<char *>(buffer) + totalRead, bufferSize - totalRead);
        if (r <= 0) {
            close(fd);
            return false;
        }
        totalRead += static_cast<size_t>(r);
    }

    return close(fd) == 0;
}

inline bool WriteFile(const char *filePath, const void *buffer, size_t size)
{
    if (buffer == nullptr && size > 0) {
        return false;
    }

    int fd = open(filePath.c_str(), O_RDWR | O_CREAT | O_TRUNC, S_IRUSR | S_IWUSR);
    if (fd < 0) {
        return false;
    }

    size_t totalWritten = 0;
    while (totalWritten < size) {
        ssize_t w = write(fd, static_cast<const char *>(buffer) + totalWritten, size - totalWritten);
        if (w < 0) {
            if (errno == EINTR) continue;
            close(fd);
            return false;
        }
        if (w == 0) {
            close(fd);
            return false;
        }
        totalWritten += static_cast<size_t>(w);
    }
    if (close(fd) != 0) {
        return false;
    }

    return true;
}

#endif
