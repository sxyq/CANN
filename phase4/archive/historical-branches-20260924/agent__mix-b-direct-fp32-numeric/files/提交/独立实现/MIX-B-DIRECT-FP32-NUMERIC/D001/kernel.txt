#include "kernel_operator.h"

namespace {

constexpr uint32_t kChunk = 1024;
constexpr uint32_t kCalcFloats = 5 * kChunk + 32 + 1024;

template <typename T>
class AddRmsNormBiasKernel {
public:
    using InQueue = AscendC::TQue<AscendC::TPosition::VECIN, 1>;
    using OutQueue = AscendC::TQue<AscendC::TPosition::VECOUT, 1>;

    __aicore__ explicit AddRmsNormBiasKernel(AscendC::TPipe* pipe) : pipe_(pipe)
    {
        pipe_->InitBuffer(xQueue_, 1, kChunk * sizeof(T));
        pipe_->InitBuffer(residualQueue_, 1, kChunk * sizeof(T));
        pipe_->InitBuffer(gammaQueue_, 1, kChunk * sizeof(T));
        pipe_->InitBuffer(biasQueue_, 1, kChunk * sizeof(T));
        pipe_->InitBuffer(outputQueue_, 1, kChunk * sizeof(T));
        pipe_->InitBuffer(calcBuffer_, kCalcFloats * sizeof(float));
    }

    __aicore__ void Process(AscendC::GlobalTensor<T>& xGm,
                            AscendC::GlobalTensor<T>& residualGm,
                            AscendC::GlobalTensor<T>& gammaGm,
                            AscendC::GlobalTensor<T>& biasGm,
                            AscendC::GlobalTensor<T>& outputGm,
                            uint64_t rowCount,
                            uint32_t dim,
                            float invDim,
                            float epsilon)
    {
        const uint64_t blockId = static_cast<uint64_t>(AscendC::GetBlockIdx());
        const uint64_t blockCount = static_cast<uint64_t>(AscendC::GetBlockNum());
        const uint64_t rowsPerBlock = rowCount / blockCount;
        const uint64_t extraRows = rowCount % blockCount;
        const uint64_t beginRow = blockId * rowsPerBlock +
            (blockId < extraRows ? blockId : extraRows);
        const uint64_t rowsThisBlock = rowsPerBlock + (blockId < extraRows ? 1 : 0);

        for (uint64_t row = 0; row < rowsThisBlock; ++row) {
            const uint64_t rowIndex = beginRow + row;
            const uint64_t rowBase = rowIndex * static_cast<uint64_t>(dim);
            const float rms = ComputeRms(xGm, residualGm, rowBase, dim, invDim, epsilon);
            WriteRow(xGm, residualGm, gammaGm, biasGm, outputGm, rowBase, dim, rms);
        }
    }

private:
    __aicore__ inline void CopyIn(const AscendC::GlobalTensor<T>& src,
                                  uint64_t offset,
                                  uint32_t count,
                                  InQueue& queue)
    {
        AscendC::LocalTensor<T> local = queue.AllocTensor<T>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = count * sizeof(T);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;

        AscendC::DataCopyPadExtParams<T> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = static_cast<T>(0);
        AscendC::DataCopyPad(local, src[offset], copyParams, padParams);
        queue.EnQue(local);
    }

    __aicore__ inline void CopyOut(const AscendC::LocalTensor<T>& local,
                                   AscendC::GlobalTensor<T>& dst,
                                   uint64_t offset,
                                   uint32_t count)
    {
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = count * sizeof(T);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(dst[offset], local, copyParams);
    }

    __aicore__ inline float ComputeRms(AscendC::GlobalTensor<T>& xGm,
                                       AscendC::GlobalTensor<T>& residualGm,
                                       uint64_t rowBase,
                                       uint32_t dim,
                                       float invDim,
                                       float epsilon)
    {
        AscendC::LocalTensor<float> calc = calcBuffer_.Get<float>();
        AscendC::LocalTensor<float> xFloat = calc;
        AscendC::LocalTensor<float> residualFloat = calc[kChunk];
        AscendC::LocalTensor<float> sumSquares = calc[2 * kChunk];
        AscendC::LocalTensor<float> squareBuffer = calc[3 * kChunk];
        AscendC::LocalTensor<float> reduceTmp = calc[4 * kChunk];
        AscendC::LocalTensor<float> scalar = calc[5 * kChunk];

        float sum = 0.0f;
        for (uint32_t offset = 0; offset < dim; offset += kChunk) {
            const uint32_t remaining = dim - offset;
            const uint32_t count = remaining < kChunk ? remaining : kChunk;
            CopyIn(xGm, rowBase + offset, count, xQueue_);
            CopyIn(residualGm, rowBase + offset, count, residualQueue_);

            AscendC::LocalTensor<T> x = xQueue_.DeQue<T>();
            AscendC::LocalTensor<T> residual = residualQueue_.DeQue<T>();
            AscendC::Cast<float, T>(xFloat, x, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Cast<float, T>(residualFloat, residual, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Add<float>(sumSquares, xFloat, residualFloat, count);
            AscendC::Mul<float>(squareBuffer, sumSquares, sumSquares, count);
            AscendC::ReduceSum<float, true>(scalar, squareBuffer, reduceTmp, static_cast<int32_t>(count));
            sum += scalar.GetValue(0);

            xQueue_.FreeTensor(x);
            residualQueue_.FreeTensor(residual);
        }

        AscendC::Duplicate<float>(scalar, sum, 1);
        AscendC::Muls<float>(scalar, scalar, invDim, 1);
        AscendC::Adds<float>(scalar, scalar, epsilon, 1);
        AscendC::Sqrt<float>(scalar, scalar, 1);
        return scalar.GetValue(0);
    }

    __aicore__ inline void WriteRow(AscendC::GlobalTensor<T>& xGm,
                                    AscendC::GlobalTensor<T>& residualGm,
                                    AscendC::GlobalTensor<T>& gammaGm,
                                    AscendC::GlobalTensor<T>& biasGm,
                                    AscendC::GlobalTensor<T>& outputGm,
                                    uint64_t rowBase,
                                    uint32_t dim,
                                    float rms)
    {
        AscendC::LocalTensor<float> calc = calcBuffer_.Get<float>();
        AscendC::LocalTensor<float> xFloat = calc;
        AscendC::LocalTensor<float> residualFloat = calc[kChunk];
        AscendC::LocalTensor<float> value = calc[2 * kChunk];
        AscendC::LocalTensor<float> gammaFloat = calc[3 * kChunk];
        AscendC::LocalTensor<float> biasFloat = calc[4 * kChunk];

        for (uint32_t offset = 0; offset < dim; offset += kChunk) {
            const uint32_t remaining = dim - offset;
            const uint32_t count = remaining < kChunk ? remaining : kChunk;
            CopyIn(xGm, rowBase + offset, count, xQueue_);
            CopyIn(residualGm, rowBase + offset, count, residualQueue_);
            CopyIn(gammaGm, offset, count, gammaQueue_);
            CopyIn(biasGm, offset, count, biasQueue_);

            AscendC::LocalTensor<T> x = xQueue_.DeQue<T>();
            AscendC::LocalTensor<T> residual = residualQueue_.DeQue<T>();
            AscendC::LocalTensor<T> gamma = gammaQueue_.DeQue<T>();
            AscendC::LocalTensor<T> bias = biasQueue_.DeQue<T>();
            AscendC::LocalTensor<T> output = outputQueue_.AllocTensor<T>();

            AscendC::Cast<float, T>(xFloat, x, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Cast<float, T>(residualFloat, residual, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Cast<float, T>(gammaFloat, gamma, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Cast<float, T>(biasFloat, bias, AscendC::RoundMode::CAST_NONE, count);
            AscendC::Add<float>(value, xFloat, residualFloat, count);
            AscendC::Duplicate<float>(xFloat, rms, count);
            AscendC::Div<float>(value, value, xFloat, static_cast<int32_t>(count));
            AscendC::Mul<float>(value, value, gammaFloat, count);
            AscendC::Add<float>(value, value, biasFloat, count);
            AscendC::Cast<T, float>(output, value, AscendC::RoundMode::CAST_RINT, count);
            outputQueue_.EnQue(output);
            AscendC::LocalTensor<T> outputReady = outputQueue_.DeQue<T>();
            CopyOut(outputReady, outputGm, rowBase + offset, count);

            xQueue_.FreeTensor(x);
            residualQueue_.FreeTensor(residual);
            gammaQueue_.FreeTensor(gamma);
            biasQueue_.FreeTensor(bias);
            outputQueue_.FreeTensor(outputReady);
        }
    }

    AscendC::TPipe* pipe_;
    InQueue xQueue_;
    InQueue residualQueue_;
    InQueue gammaQueue_;
    InQueue biasQueue_;
    OutQueue outputQueue_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> calcBuffer_;
};

template <typename T>
__global__ __vector__ void add_rms_norm_bias_custom_kernel(GM_ADDR x,
                                                            GM_ADDR residual,
                                                            GM_ADDR gamma,
                                                            GM_ADDR bias,
                                                            GM_ADDR output,
                                                            uint64_t rowCount,
                                                            uint32_t dim,
                                                            float invDim,
                                                            float epsilon)
{
    AscendC::TPipe pipe;
    AddRmsNormBiasKernel<T> kernel(&pipe);
    AscendC::GlobalTensor<T> xGm;
    AscendC::GlobalTensor<T> residualGm;
    AscendC::GlobalTensor<T> gammaGm;
    AscendC::GlobalTensor<T> biasGm;
    AscendC::GlobalTensor<T> outputGm;
    xGm.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(x));
    residualGm.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(residual));
    gammaGm.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(gamma));
    biasGm.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(bias));
    outputGm.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(output));
    kernel.Process(xGm, residualGm, gammaGm, biasGm, outputGm, rowCount, dim, invDim, epsilon);
}

}  // namespace

extern "C" void run_kernel(GM_ADDR x,
                            const TensorGroupInfo& info_x,
                            GM_ADDR residual,
                            const TensorGroupInfo& info_residual,
                            GM_ADDR gamma,
                            const TensorGroupInfo& info_gamma,
                            GM_ADDR bias,
                            const TensorGroupInfo& info_bias,
                            GM_ADDR output,
                            const TensorGroupInfo& info_output,
                            int64_t availableCoreNum,
                            aclrtStream stream,
                            float epsilon)
{
    (void)info_residual;
    (void)info_gamma;
    (void)info_bias;
    (void)info_output;

    const TensorInfo& xInfo = info_x.tensors[0];
    const uint32_t dim = static_cast<uint32_t>(xInfo.shape[xInfo.numDims - 1]);
    uint64_t rowCount = 1;
    for (int64_t i = 0; i + 1 < xInfo.numDims; ++i) {
        rowCount *= static_cast<uint64_t>(xInfo.shape[i]);
    }
    uint32_t blockCount = static_cast<uint32_t>(availableCoreNum);
    if (rowCount < static_cast<uint64_t>(blockCount)) {
        blockCount = static_cast<uint32_t>(rowCount);
    }
    if (blockCount == 0) {
        blockCount = 1;
    }
    const float invDim = 1.0f / static_cast<float>(dim);

    switch (xInfo.dtype) {
        case 0:
            add_rms_norm_bias_custom_kernel<float><<<blockCount, nullptr, stream>>>(
                x, residual, gamma, bias, output, rowCount, dim, invDim, epsilon);
            break;
        case 1:
            add_rms_norm_bias_custom_kernel<half><<<blockCount, nullptr, stream>>>(
                x, residual, gamma, bias, output, rowCount, dim, invDim, epsilon);
            break;
        case 2:
            add_rms_norm_bias_custom_kernel<bfloat16_t><<<blockCount, nullptr, stream>>>(
                x, residual, gamma, bias, output, rowCount, dim, invDim, epsilon);
            break;
        default:
            break;
    }
}
