#include <cmath>
#include <cstdint>
#include "kernel_operator.h"

/*
 * AddRmsNormBias - FULL-R016-V001-SCHEDULING-ARCH
 *
 * Branch:
 *   R016 / scheduling-centric architecture.
 *
 * This is a full branch rewrite.  It does NOT descend from FULL-R014 and it
 * does not use the previous branch's parameter-residency dataflow.
 *
 * Golden semantics:
 *   y   = x + residual
 *   rms = sqrt(mean(y * y) + epsilon)
 *   out = y / rms * gamma + bias
 *
 * Dominant idea:
 *   Shape-aware scheduling is the architecture.  The numerical row engine is
 *   intentionally plain and all branch-specific decisions are concentrated
 *   in how logical rows become work-items and how those work-items are mapped
 *   onto vector cores.
 *
 * Scheduling design:
 *
 *   Band A: tiny rows, D <= 256
 *     - 16 rows per logical work-item.
 *     - cyclic work-item distribution across cores.
 *     - objective: avoid spending one hardware work slot on each tiny row.
 *
 *   Band B: small rows, 256 < D <= 1024
 *     - 8 rows per work-item.
 *     - balanced contiguous work-item intervals.
 *
 *   Band C: medium rows, 1024 < D <= 4096
 *     - 4 rows per work-item.
 *     - cyclic work-item distribution.
 *
 *   Band D: large rows, 4096 < D <= 16384
 *     - 2 rows per work-item.
 *     - balanced contiguous work-item intervals.
 *
 *   Band E: very wide rows, D > 16384
 *     - 1 row per work-item.
 *     - balanced contiguous work.
 *     - active-core cap to avoid launching the complete core set against a
 *       small number of bandwidth-heavy rows.
 *
 * Important distinction:
 *   "rows per work-item" is a scheduling bundle only.  Rows are still computed
 *   one at a time and do not share UB tensors or vector instructions.  This is
 *   therefore not the R015 multi-row vector-batching route.
 *
 * Correctness rule:
 *   If one logical row is not naturally 32-byte aligned, launch one core.
 *   This avoids cross-core DataCopyPad writes touching the same physical GM
 *   block.  We deliberately do NOT import the R012 row-group optimization.
 *
 * Explicitly absent:
 *   - parameter residency;
 *   - retained-y;
 *   - simultaneous multi-row vector batching;
 *   - wide cached-row specialization;
 *   - double buffer / ping-pong pipeline;
 *   - hierarchical partial collector;
 *   - row-group alignment scheduling;
 *   - aligned DataCopy fast path;
 *   - manual-tail split;
 *   - reciprocal/Muls normalization;
 *   - R031 multimode compute architecture.
 *
 * File format:
 *   Delivered as .txt. Rename to kernel.asc for submission if required.
 */

namespace R016Schedule {

enum ScheduleMode : uint32_t {
    kCyclicTasks = 0,
    kContiguousTasks = 1
};

static inline uint64_t CeilDivU64(
    uint64_t value,
    uint64_t divisor)
{
    return value / divisor +
           ((value % divisor) == 0ULL ? 0ULL : 1ULL);
}

static inline uint64_t MinU64(
    uint64_t a,
    uint64_t b)
{
    return a < b ? a : b;
}

} // namespace R016Schedule

template <typename T>
class ScheduledRowEngine {
public:
    static constexpr int32_t kTileElems = 3072;
    static constexpr int32_t kScalarElems = 16;
    static constexpr int32_t kScalarOffset = 8;

    __aicore__ inline void Init(
        GM_ADDR x,
        GM_ADDR residual,
        GM_ADDR gamma,
        GM_ADDR bias,
        GM_ADDR output)
    {
        xGm_.SetGlobalBuffer((__gm__ T*)x);
        residualGm_.SetGlobalBuffer((__gm__ T*)residual);
        gammaGm_.SetGlobalBuffer((__gm__ T*)gamma);
        biasGm_.SetGlobalBuffer((__gm__ T*)bias);
        outputGm_.SetGlobalBuffer((__gm__ T*)output);

        /*
         * Fixed row-engine working set.
         *
         * This branch intentionally avoids shape-dependent UB layouts.  The
         * scheduling policy changes with the shape, while this row engine
         * remains fixed.
         */
        pipe_.InitBuffer(
            xNative_,
            kTileElems * sizeof(T));
        pipe_.InitBuffer(
            rNative_,
            kTileElems * sizeof(T));
        pipe_.InitBuffer(
            outNative_,
            kTileElems * sizeof(T));

        pipe_.InitBuffer(
            fp32A_,
            kTileElems * sizeof(float));
        pipe_.InitBuffer(
            fp32B_,
            kTileElems * sizeof(float));
        pipe_.InitBuffer(
            value_,
            kTileElems * sizeof(float));
        pipe_.InitBuffer(
            reduceWork_,
            kTileElems * sizeof(float));

        pipe_.InitBuffer(
            scalar_,
            kScalarElems * sizeof(float));
    }

    __aicore__ inline void ProcessRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row * rowWidth;

        const float rms =
            ComputeRms(
                rowOffset,
                rowWidth,
                invRowWidth,
                epsilon);

        WriteOutput(
            rowOffset,
            rowWidth,
            rms);
    }

private:
    __aicore__ inline int32_t TileLength(
        uint64_t remaining) const
    {
        return static_cast<int32_t>(
            remaining >
                    static_cast<uint64_t>(kTileElems)
                ? kTileElems
                : remaining);
    }

    __aicore__ inline void LoadNative(
        AscendC::LocalTensor<T> dst,
        AscendC::GlobalTensor<T>& src,
        uint64_t offset,
        int32_t valid)
    {
        AscendC::DataCopyExtParams copyParams{
            1,
            static_cast<uint32_t>(
                valid *
                static_cast<int32_t>(sizeof(T))),
            0,
            0,
            0
        };

        AscendC::DataCopyPadExtParams<T> padParams{
            false,
            0,
            0,
            0
        };

        AscendC::DataCopyPad<T>(
            dst,
            src[offset],
            copyParams,
            padParams);
    }

    __aicore__ inline void StoreNative(
        AscendC::GlobalTensor<T>& dst,
        uint64_t offset,
        AscendC::LocalTensor<T> src,
        int32_t valid)
    {
        AscendC::DataCopyExtParams copyParams{
            1,
            static_cast<uint32_t>(
                valid *
                static_cast<int32_t>(sizeof(T))),
            0,
            0,
            0
        };

        AscendC::DataCopyPad<T>(
            dst[offset],
            src,
            copyParams);
    }

    __aicore__ inline void ToFp32(
        AscendC::LocalTensor<float> dst,
        AscendC::LocalTensor<T> src,
        int32_t valid)
    {
        if constexpr (
            AscendC::IsSameType<T, float>::value) {
            AscendC::Adds(
                dst,
                src,
                0.0f,
                valid);
        } else {
            AscendC::Cast(
                dst,
                src,
                AscendC::RoundMode::CAST_NONE,
                valid);
        }
    }

    __aicore__ inline void FromFp32(
        AscendC::LocalTensor<T> dst,
        AscendC::LocalTensor<float> src,
        int32_t valid)
    {
        if constexpr (
            AscendC::IsSameType<T, float>::value) {
            AscendC::Adds(
                dst,
                src,
                0.0f,
                valid);
        } else {
            AscendC::Cast(
                dst,
                src,
                AscendC::RoundMode::CAST_RINT,
                valid);
        }
    }

    __aicore__ inline void LoadAndFormY(
        uint64_t gmOffset,
        int32_t valid)
    {
        AscendC::LocalTensor<T> xLocal =
            xNative_.Get<T>();
        AscendC::LocalTensor<T> rLocal =
            rNative_.Get<T>();

        AscendC::LocalTensor<float> fp32A =
            fp32A_.Get<float>();
        AscendC::LocalTensor<float> fp32B =
            fp32B_.Get<float>();
        AscendC::LocalTensor<float> value =
            value_.Get<float>();

        LoadNative(
            xLocal,
            xGm_,
            gmOffset,
            valid);

        LoadNative(
            rLocal,
            residualGm_,
            gmOffset,
            valid);

        AscendC::PipeBarrier<PIPE_ALL>();

        if constexpr (
            AscendC::IsSameType<T, float>::value) {
            AscendC::Add(
                value,
                xLocal,
                rLocal,
                valid);
        } else {
            ToFp32(
                fp32A,
                xLocal,
                valid);

            ToFp32(
                fp32B,
                rLocal,
                valid);

            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::Add(
                value,
                fp32A,
                fp32B,
                valid);
        }

        AscendC::PipeBarrier<PIPE_V>();
    }

    __aicore__ inline float ReadScalar(
        AscendC::LocalTensor<float> tensor)
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_S>(0);
        AscendC::WaitFlag<
            AscendC::HardEvent::V_S>(0);

        const float value =
            tensor.GetValue(0);

        AscendC::SetFlag<
            AscendC::HardEvent::S_V>(0);
        AscendC::WaitFlag<
            AscendC::HardEvent::S_V>(0);

        return value;
    }

    __aicore__ inline float ComputeRms(
        uint64_t rowOffset,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        AscendC::LocalTensor<float> value =
            value_.Get<float>();

        AscendC::LocalTensor<float> square =
            fp32A_.Get<float>();

        AscendC::LocalTensor<float> reduceTmp =
            reduceWork_.Get<float>();

        AscendC::LocalTensor<float> scalarSlots =
            scalar_.Get<float>();

        AscendC::LocalTensor<float> partial =
            scalarSlots;

        AscendC::LocalTensor<float> rowSum =
            scalarSlots[kScalarOffset];

        AscendC::Duplicate(
            rowSum,
            0.0f,
            1);

        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint64_t col = 0;
             col < rowWidth;
             col += static_cast<uint64_t>(
                 kTileElems)) {
            const int32_t valid =
                TileLength(
                    rowWidth - col);

            LoadAndFormY(
                rowOffset + col,
                valid);

            AscendC::Mul(
                square,
                value,
                value,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum(
                partial,
                square,
                reduceTmp,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                rowSum,
                rowSum,
                partial,
                1);

            AscendC::PipeBarrier<PIPE_ALL>();
        }

        AscendC::Muls(
            rowSum,
            rowSum,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rowSum,
            rowSum,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rowSum,
            rowSum,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        return ReadScalar(rowSum);
    }

    __aicore__ inline void WriteOutput(
        uint64_t rowOffset,
        uint64_t rowWidth,
        float rms)
    {
        AscendC::LocalTensor<T> xLocal =
            xNative_.Get<T>();
        AscendC::LocalTensor<T> rLocal =
            rNative_.Get<T>();
        AscendC::LocalTensor<T> outLocal =
            outNative_.Get<T>();

        AscendC::LocalTensor<float> fp32A =
            fp32A_.Get<float>();
        AscendC::LocalTensor<float> fp32B =
            fp32B_.Get<float>();
        AscendC::LocalTensor<float> value =
            value_.Get<float>();
        AscendC::LocalTensor<float> denom =
            reduceWork_.Get<float>();

        for (uint64_t col = 0;
             col < rowWidth;
             col += static_cast<uint64_t>(
                 kTileElems)) {
            const int32_t valid =
                TileLength(
                    rowWidth - col);

            LoadAndFormY(
                rowOffset + col,
                valid);

            AscendC::Duplicate(
                denom,
                rms,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Div(
                value,
                value,
                denom,
                valid);

            AscendC::PipeBarrier<PIPE_ALL>();

            /*
             * Stream gamma/bias every row.  Parameter reuse is explicitly
             * excluded from this scheduling branch.
             */
            LoadNative(
                xLocal,
                gammaGm_,
                col,
                valid);

            LoadNative(
                rLocal,
                biasGm_,
                col,
                valid);

            AscendC::PipeBarrier<PIPE_ALL>();

            if constexpr (
                AscendC::IsSameType<T, float>::value) {
                AscendC::Mul(
                    value,
                    value,
                    xLocal,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Add(
                    value,
                    value,
                    rLocal,
                    valid);
            } else {
                ToFp32(
                    fp32A,
                    xLocal,
                    valid);

                ToFp32(
                    fp32B,
                    rLocal,
                    valid);

                AscendC::PipeBarrier<PIPE_ALL>();

                AscendC::Mul(
                    value,
                    value,
                    fp32A,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Add(
                    value,
                    value,
                    fp32B,
                    valid);
            }

            AscendC::PipeBarrier<PIPE_V>();

            FromFp32(
                outLocal,
                value,
                valid);

            AscendC::PipeBarrier<PIPE_ALL>();

            StoreNative(
                outputGm_,
                rowOffset + col,
                outLocal,
                valid);

            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe_;

    AscendC::TBuf<
        AscendC::TPosition::VECIN>
        xNative_;

    AscendC::TBuf<
        AscendC::TPosition::VECIN>
        rNative_;

    AscendC::TBuf<
        AscendC::TPosition::VECOUT>
        outNative_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        fp32A_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        fp32B_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        value_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        reduceWork_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        scalar_;

    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;
};

template <typename T>
class SchedulingArchitectureKernel {
public:
    __aicore__ inline void Init(
        GM_ADDR x,
        GM_ADDR residual,
        GM_ADDR gamma,
        GM_ADDR bias,
        GM_ADDR output)
    {
        rowEngine_.Init(
            x,
            residual,
            gamma,
            bias,
            output);
    }

    __aicore__ inline void Run(
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowsPerTask,
        uint64_t taskCount,
        uint32_t scheduleMode,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t coreIdx =
            static_cast<uint64_t>(
                AscendC::GetBlockIdx());

        const uint64_t coreCount =
            static_cast<uint64_t>(
                AscendC::GetBlockNum());

        if (scheduleMode ==
            R016Schedule::kCyclicTasks) {
            RunCyclic(
                coreIdx,
                coreCount,
                rowCount,
                rowWidth,
                rowsPerTask,
                taskCount,
                invRowWidth,
                epsilon);
        } else {
            RunContiguous(
                coreIdx,
                coreCount,
                rowCount,
                rowWidth,
                rowsPerTask,
                taskCount,
                invRowWidth,
                epsilon);
        }
    }

private:
    __aicore__ inline void ProcessTask(
        uint64_t taskId,
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowsPerTask,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t firstRow =
            taskId * rowsPerTask;

        uint64_t endRow =
            firstRow + rowsPerTask;

        if (endRow > rowCount) {
            endRow = rowCount;
        }

        /*
         * Scheduling bundle only:
         * rows execute sequentially and share no vector operation.
         */
        for (uint64_t row = firstRow;
             row < endRow;
             ++row) {
            rowEngine_.ProcessRow(
                row,
                rowWidth,
                invRowWidth,
                epsilon);
        }
    }

    __aicore__ inline void RunCyclic(
        uint64_t coreIdx,
        uint64_t coreCount,
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowsPerTask,
        uint64_t taskCount,
        float invRowWidth,
        float epsilon)
    {
        /*
         * Block-cyclic work assignment prevents one core from receiving a
         * disproportionate tail when there are many small logical tasks.
         */
        for (uint64_t taskId = coreIdx;
             taskId < taskCount;
             taskId += coreCount) {
            ProcessTask(
                taskId,
                rowCount,
                rowWidth,
                rowsPerTask,
                invRowWidth,
                epsilon);
        }
    }

    __aicore__ inline void RunContiguous(
        uint64_t coreIdx,
        uint64_t coreCount,
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowsPerTask,
        uint64_t taskCount,
        float invRowWidth,
        float epsilon)
    {
        /*
         * Balanced contiguous partitioning:
         *
         *   first 'remainder' cores get one extra work-item.
         *
         * This minimizes work-item discontinuity for large rows while still
         * keeping the load difference bounded by one work-item.
         */
        const uint64_t baseTasks =
            taskCount / coreCount;

        const uint64_t remainder =
            taskCount % coreCount;

        const uint64_t localTaskCount =
            baseTasks +
            (coreIdx < remainder
                 ? 1ULL
                 : 0ULL);

        const uint64_t taskBegin =
            coreIdx * baseTasks +
            (coreIdx < remainder
                 ? coreIdx
                 : remainder);

        const uint64_t taskEnd =
            taskBegin +
            localTaskCount;

        for (uint64_t taskId = taskBegin;
             taskId < taskEnd;
             ++taskId) {
            ProcessTask(
                taskId,
                rowCount,
                rowWidth,
                rowsPerTask,
                invRowWidth,
                epsilon);
        }
    }

private:
    ScheduledRowEngine<T> rowEngine_;
};

template <typename T>
__global__ __vector__ void add_rms_norm_bias_r016(
    GM_ADDR x,
    GM_ADDR residual,
    GM_ADDR gamma,
    GM_ADDR bias,
    GM_ADDR output,
    uint64_t rowCount,
    uint64_t rowWidth,
    uint64_t rowsPerTask,
    uint64_t taskCount,
    uint32_t scheduleMode,
    float invRowWidth,
    float epsilon)
{
    SchedulingArchitectureKernel<T> kernel;

    kernel.Init(
        x,
        residual,
        gamma,
        bias,
        output);

    kernel.Run(
        rowCount,
        rowWidth,
        rowsPerTask,
        taskCount,
        scheduleMode,
        invRowWidth,
        epsilon);
}

#if !defined(ASCENDC_CPU_DEBUG)
extern "C" void run_kernel(
    GM_ADDR x,
    const TensorGroupInfo& info_x,
    GM_ADDR residual,
    const TensorGroupInfo& info_residual,
    GM_ADDR gamma,
    const TensorGroupInfo& info_gamma,
    GM_ADDR bias,
    const TensorGroupInfo& info_bias,
    GM_ADDR output,
    const TensorGroupInfo& info_output,
    int64_t availableCoreNum,
    aclrtStream stream,
    float epsilon)
{
    if (x == nullptr ||
        residual == nullptr ||
        gamma == nullptr ||
        bias == nullptr ||
        output == nullptr ||
        info_x.tensors == nullptr ||
        info_residual.tensors == nullptr ||
        info_gamma.tensors == nullptr ||
        info_bias.tensors == nullptr ||
        info_output.tensors == nullptr ||
        info_x.numTensors < 1 ||
        info_residual.numTensors < 1 ||
        info_gamma.numTensors < 1 ||
        info_bias.numTensors < 1 ||
        info_output.numTensors < 1) {
        return;
    }

    const TensorInfo& xInfo =
        info_x.tensors[0];

    const TensorInfo& residualInfo =
        info_residual.tensors[0];

    const TensorInfo& gammaInfo =
        info_gamma.tensors[0];

    const TensorInfo& biasInfo =
        info_bias.tensors[0];

    const TensorInfo& outputInfo =
        info_output.tensors[0];

    if (xInfo.shape == nullptr ||
        residualInfo.shape == nullptr ||
        gammaInfo.shape == nullptr ||
        biasInfo.shape == nullptr ||
        outputInfo.shape == nullptr ||
        xInfo.numDims < 1 ||
        residualInfo.numDims !=
            xInfo.numDims ||
        outputInfo.numDims !=
            xInfo.numDims ||
        gammaInfo.numDims != 1 ||
        biasInfo.numDims != 1 ||
        xInfo.dtype != residualInfo.dtype ||
        xInfo.dtype != gammaInfo.dtype ||
        xInfo.dtype != biasInfo.dtype ||
        xInfo.dtype != outputInfo.dtype) {
        return;
    }

    if (xInfo.dtype != 0 &&
        xInfo.dtype != 1 &&
        xInfo.dtype != 2) {
        return;
    }

    const int64_t widthSigned =
        xInfo.shape[
            xInfo.numDims - 1];

    if (widthSigned <= 0 ||
        gammaInfo.shape[0] !=
            widthSigned ||
        biasInfo.shape[0] !=
            widthSigned ||
        residualInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||
        outputInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned) {
        return;
    }

    const uint64_t maxU64 =
        ~static_cast<uint64_t>(0);

    uint64_t rowCount = 1;

    for (int64_t dim = 0;
         dim < xInfo.numDims - 1;
         ++dim) {
        const int64_t extentSigned =
            xInfo.shape[dim];

        if (extentSigned <= 0 ||
            residualInfo.shape[dim] !=
                extentSigned ||
            outputInfo.shape[dim] !=
                extentSigned) {
            return;
        }

        const uint64_t extent =
            static_cast<uint64_t>(
                extentSigned);

        if (rowCount >
            maxU64 / extent) {
            return;
        }

        rowCount *= extent;
    }

    if (rowCount == 0) {
        return;
    }

    const uint64_t rowWidth =
        static_cast<uint64_t>(
            widthSigned);

    if (rowCount >
        maxU64 / rowWidth) {
        return;
    }

    const uint64_t elementBytes =
        xInfo.dtype == 0
            ? 4ULL
            : 2ULL;

    if (rowWidth >
        maxU64 / elementBytes) {
        return;
    }

    const uint64_t rowBytes =
        rowWidth * elementBytes;

    /*
     * Scheduling plan.
     *
     * The thresholds change work granularity and core mapping only.  The row
     * numerical engine is identical in every band.
     */
    uint64_t rowsPerTask = 1ULL;
    uint32_t scheduleMode =
        R016Schedule::kContiguousTasks;

    if (rowWidth <= 256ULL) {
        rowsPerTask = 16ULL;
        scheduleMode =
            R016Schedule::kCyclicTasks;
    } else if (rowWidth <= 1024ULL) {
        rowsPerTask = 8ULL;
        scheduleMode =
            R016Schedule::kContiguousTasks;
    } else if (rowWidth <= 4096ULL) {
        rowsPerTask = 4ULL;
        scheduleMode =
            R016Schedule::kCyclicTasks;
    } else if (rowWidth <= 16384ULL) {
        rowsPerTask = 2ULL;
        scheduleMode =
            R016Schedule::kContiguousTasks;
    } else {
        rowsPerTask = 1ULL;
        scheduleMode =
            R016Schedule::kContiguousTasks;
    }

    const uint64_t taskCount =
        R016Schedule::CeilDivU64(
            rowCount,
            rowsPerTask);

    uint64_t available =
        availableCoreNum > 0
            ? static_cast<uint64_t>(
                  availableCoreNum)
            : 1ULL;

    uint64_t requestedBlocks =
        R016Schedule::MinU64(
            available,
            taskCount);

    /*
     * Non-aligned row correctness gate.
     *
     * No row-group alignment trick is used in this branch.
     */
    const bool naturallyAligned =
        (rowBytes & 31ULL) == 0ULL;

    if (!naturallyAligned) {
        requestedBlocks = 1ULL;
    }

    /*
     * Very-wide scheduling band:
     * cap active vector cores.  This is the principal wide-row scheduling
     * experiment of R016; the compute path itself remains unchanged.
     */
    if (rowWidth > 16384ULL &&
        requestedBlocks > 8ULL) {
        requestedBlocks = 8ULL;
    }

    if (requestedBlocks == 0) {
        requestedBlocks = 1ULL;
    }

    if (requestedBlocks >
        static_cast<uint64_t>(
            0xffffffffU)) {
        requestedBlocks =
            static_cast<uint64_t>(
                0xffffffffU);
    }

    const uint32_t blockCount =
        static_cast<uint32_t>(
            requestedBlocks);

    const float invRowWidth =
        1.0f /
        static_cast<float>(
            widthSigned);

    if (xInfo.dtype == 0) {
        add_rms_norm_bias_r016<float>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowsPerTask,
                taskCount,
                scheduleMode,
                invRowWidth,
                epsilon);
    } else if (xInfo.dtype == 1) {
        add_rms_norm_bias_r016<half>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowsPerTask,
                taskCount,
                scheduleMode,
                invRowWidth,
                epsilon);
    } else {
        add_rms_norm_bias_r016<bfloat16_t>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowsPerTask,
                taskCount,
                scheduleMode,
                invRowWidth,
                epsilon);
    }
}
