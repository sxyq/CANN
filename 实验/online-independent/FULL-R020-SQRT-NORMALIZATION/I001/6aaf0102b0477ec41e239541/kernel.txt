#include "kernel_operator.h"

using namespace AscendC;

namespace {

constexpr int32_t kTileElems = 2048;
constexpr int32_t kScalarElems = 8;

template <typename T>
__aicore__ inline int32_t AlignElements(const int32_t count)
{
    constexpr int32_t elemsPerBlock = 32 / static_cast<int32_t>(sizeof(T));
    return ((count + elemsPerBlock - 1) / elemsPerBlock) * elemsPerBlock;
}

template <typename T>
__aicore__ inline void LoadToFloat(const GlobalTensor<T>& source,
                                   const uint64_t offset,
                                   const int32_t count,
                                   const int32_t alignedCount,
                                   const LocalTensor<T>& raw,
                                   const LocalTensor<float>& converted)
{
    DataCopyExtParams copyParams(1, static_cast<uint32_t>(count * sizeof(T)), 0, 0, 0);
    DataCopyPadExtParams<T> padParams(true, 0,
                                      static_cast<uint8_t>(alignedCount - count), static_cast<T>(0));
    DataCopyPad(raw, source[offset], copyParams, padParams);
    PipeBarrier<PIPE_ALL>();
    Cast(converted, raw, RoundMode::CAST_NONE, alignedCount);
    PipeBarrier<PIPE_ALL>();
}

template <typename T>
__aicore__ inline void ProcessRow(const GlobalTensor<T>& xGm,
                                   const GlobalTensor<T>& residualGm,
                                   const GlobalTensor<T>& gammaGm,
                                   const GlobalTensor<T>& biasGm,
                                   const GlobalTensor<T>& outputGm,
                                   const uint64_t rowOffset,
                                   const int32_t dSize,
                                   const float epsilon,
                                   TBuf<TPosition::VECCALC>& xRawBuf,
                                   TBuf<TPosition::VECCALC>& residualRawBuf,
                                   TBuf<TPosition::VECCALC>& gammaRawBuf,
                                   TBuf<TPosition::VECCALC>& biasRawBuf,
                                   TBuf<TPosition::VECCALC>& outputRawBuf,
                                   TBuf<TPosition::VECCALC>& xBuf,
                                   TBuf<TPosition::VECCALC>& residualBuf,
                                   TBuf<TPosition::VECCALC>& gammaBuf,
                                   TBuf<TPosition::VECCALC>& biasBuf,
                                   TBuf<TPosition::VECCALC>& yBuf,
                                   TBuf<TPosition::VECCALC>& squareBuf,
                                   TBuf<TPosition::VECCALC>& rmsBuf,
                                   TBuf<TPosition::VECCALC>& reduceOutBuf,
                                   TBuf<TPosition::VECCALC>& reduceTmpBuf,
                                   TBuf<TPosition::VECCALC>& scalarInBuf,
                                   TBuf<TPosition::VECCALC>& scalarOutBuf)
{
    GlobalTensor<T> x = xGm;
    GlobalTensor<T> residual = residualGm;
    GlobalTensor<T> gamma = gammaGm;
    GlobalTensor<T> bias = biasGm;
    GlobalTensor<T> output = outputGm;

    LocalTensor<T> xRaw = xRawBuf.Get<T>();
    LocalTensor<T> residualRaw = residualRawBuf.Get<T>();
    LocalTensor<T> gammaRaw = gammaRawBuf.Get<T>();
    LocalTensor<T> biasRaw = biasRawBuf.Get<T>();
    LocalTensor<T> outputRaw = outputRawBuf.Get<T>();
    LocalTensor<float> xLocal = xBuf.Get<float>();
    LocalTensor<float> residualLocal = residualBuf.Get<float>();
    LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
    LocalTensor<float> biasLocal = biasBuf.Get<float>();
    LocalTensor<float> yLocal = yBuf.Get<float>();
    LocalTensor<float> squareLocal = squareBuf.Get<float>();
    LocalTensor<float> rmsLocal = rmsBuf.Get<float>();
    LocalTensor<float> reduceOut = reduceOutBuf.Get<float>();
    LocalTensor<float> reduceTmp = reduceTmpBuf.Get<float>();
    LocalTensor<float> scalarIn = scalarInBuf.Get<float>();
    LocalTensor<float> scalarOut = scalarOutBuf.Get<float>();

    float squareSum = 0.0f;
    for (int32_t col = 0; col < dSize; col += kTileElems) {
        const int32_t count = (dSize - col < kTileElems) ? (dSize - col) : kTileElems;
        const int32_t alignedCount = AlignElements<T>(count);
        const uint64_t offset = rowOffset + static_cast<uint64_t>(col);
        LoadToFloat(x, offset, count, alignedCount, xRaw, xLocal);
        LoadToFloat(residual, offset, count, alignedCount, residualRaw, residualLocal);
        Add(yLocal, xLocal, residualLocal, alignedCount);
        Mul(squareLocal, yLocal, yLocal, alignedCount);
        PipeBarrier<PIPE_ALL>();
        ReduceSum(reduceOut, squareLocal, reduceTmp, alignedCount);
        PipeBarrier<PIPE_ALL>();
        squareSum += reduceOut.GetValue(0);
    }

    const float meanWithEpsilon = squareSum / static_cast<float>(dSize) + epsilon;
    Duplicate(scalarIn, meanWithEpsilon, kScalarElems);
    Sqrt(scalarOut, scalarIn, kScalarElems);
    PipeBarrier<PIPE_ALL>();
    const float rms = scalarOut.GetValue(0);

    for (int32_t col = 0; col < dSize; col += kTileElems) {
        const int32_t count = (dSize - col < kTileElems) ? (dSize - col) : kTileElems;
        const int32_t alignedCount = AlignElements<T>(count);
        const uint64_t offset = rowOffset + static_cast<uint64_t>(col);
        LoadToFloat(x, offset, count, alignedCount, xRaw, xLocal);
        LoadToFloat(residual, offset, count, alignedCount, residualRaw, residualLocal);
        LoadToFloat(gamma, static_cast<uint64_t>(col), count, alignedCount, gammaRaw, gammaLocal);
        LoadToFloat(bias, static_cast<uint64_t>(col), count, alignedCount, biasRaw, biasLocal);

        Add(yLocal, xLocal, residualLocal, alignedCount);
        Duplicate(rmsLocal, rms, alignedCount);
        Div(yLocal, yLocal, rmsLocal, alignedCount);
        Mul(yLocal, yLocal, gammaLocal, alignedCount);
        Add(yLocal, yLocal, biasLocal, alignedCount);
        Cast(outputRaw, yLocal, RoundMode::CAST_RINT, alignedCount);
        PipeBarrier<PIPE_ALL>();

        DataCopyExtParams outputParams(1, static_cast<uint32_t>(count * sizeof(T)), 0, 0, 0);
        DataCopyPad(output[offset], outputRaw, outputParams);
        PipeBarrier<PIPE_ALL>();
    }
}

template <typename T>
__global__ __aicore__ void AddRmsNormBiasKernel(GM_ADDR x,
                                                GM_ADDR residual,
                                                GM_ADDR gamma,
                                                GM_ADDR bias,
                                                GM_ADDR output,
                                                const int64_t outer,
                                                const int32_t dSize,
                                                const float epsilon,
                                                const uint32_t blockNum)
{
    GlobalTensor<T> xGm;
    GlobalTensor<T> residualGm;
    GlobalTensor<T> gammaGm;
    GlobalTensor<T> biasGm;
    GlobalTensor<T> outputGm;
    xGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(x));
    residualGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(residual));
    gammaGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(gamma));
    biasGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(bias));
    outputGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(output));

    TPipe localPipe;
    TBuf<TPosition::VECCALC> xRawBuf;
    TBuf<TPosition::VECCALC> residualRawBuf;
    TBuf<TPosition::VECCALC> gammaRawBuf;
    TBuf<TPosition::VECCALC> biasRawBuf;
    TBuf<TPosition::VECCALC> outputRawBuf;
    TBuf<TPosition::VECCALC> xBuf;
    TBuf<TPosition::VECCALC> residualBuf;
    TBuf<TPosition::VECCALC> gammaBuf;
    TBuf<TPosition::VECCALC> biasBuf;
    TBuf<TPosition::VECCALC> yBuf;
    TBuf<TPosition::VECCALC> squareBuf;
    TBuf<TPosition::VECCALC> rmsBuf;
    TBuf<TPosition::VECCALC> reduceOutBuf;
    TBuf<TPosition::VECCALC> reduceTmpBuf;
    TBuf<TPosition::VECCALC> scalarInBuf;
    TBuf<TPosition::VECCALC> scalarOutBuf;

    localPipe.InitBuffer(xRawBuf, kTileElems * sizeof(T));
    localPipe.InitBuffer(residualRawBuf, kTileElems * sizeof(T));
    localPipe.InitBuffer(gammaRawBuf, kTileElems * sizeof(T));
    localPipe.InitBuffer(biasRawBuf, kTileElems * sizeof(T));
    localPipe.InitBuffer(outputRawBuf, kTileElems * sizeof(T));
    localPipe.InitBuffer(xBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(residualBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(gammaBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(biasBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(yBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(squareBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(rmsBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(reduceOutBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(reduceTmpBuf, kTileElems * sizeof(float));
    localPipe.InitBuffer(scalarInBuf, kScalarElems * sizeof(float));
    localPipe.InitBuffer(scalarOutBuf, kScalarElems * sizeof(float));

    const int64_t blockIdx = GetBlockIdx();
    for (int64_t row = blockIdx; row < outer; row += static_cast<int64_t>(blockNum)) {
        const uint64_t rowOffset = static_cast<uint64_t>(row) * static_cast<uint64_t>(dSize);
        ProcessRow(xGm, residualGm, gammaGm, biasGm, outputGm, rowOffset, dSize, epsilon,
                   xRawBuf, residualRawBuf, gammaRawBuf, biasRawBuf, outputRawBuf,
                   xBuf, residualBuf, gammaBuf, biasBuf, yBuf, squareBuf, rmsBuf,
                   reduceOutBuf, reduceTmpBuf, scalarInBuf, scalarOutBuf);
    }
}

} // namespace

extern "C" void run_kernel(GM_ADDR x,
                            const TensorGroupInfo& info_x,
                            GM_ADDR residual,
                            const TensorGroupInfo& info_residual,
                            GM_ADDR gamma,
                            const TensorGroupInfo& info_gamma,
                            GM_ADDR bias,
                            const TensorGroupInfo& info_bias,
                            GM_ADDR output,
                            const TensorGroupInfo& info_output,
                            int64_t availableCoreNum,
                            aclrtStream stream,
                            float epsilon)
{
    (void)info_residual;
    (void)info_gamma;
    (void)info_bias;
    (void)info_output;

    const TensorInfo& xInfo = info_x.tensors[0];
    const int64_t dSize64 = xInfo.shape[xInfo.numDims - 1];
    int64_t outer = 1;
    for (int64_t dim = 0; dim + 1 < xInfo.numDims; ++dim) {
        outer *= xInfo.shape[dim];
    }

    if (outer <= 0 || dSize64 <= 0 || availableCoreNum <= 0) {
        return;
    }
    const int32_t dSize = static_cast<int32_t>(dSize64);
    const uint32_t blockNum = static_cast<uint32_t>(
        (availableCoreNum < outer) ? availableCoreNum : outer);

    if (xInfo.dtype == 0) {
        AddRmsNormBiasKernel<float><<<blockNum, nullptr, stream>>>(
            x, residual, gamma, bias, output, outer, dSize, epsilon, blockNum);
    } else if (xInfo.dtype == 1) {
        AddRmsNormBiasKernel<half><<<blockNum, nullptr, stream>>>(
            x, residual, gamma, bias, output, outer, dSize, epsilon, blockNum);
    } else if (xInfo.dtype == 2) {
        AddRmsNormBiasKernel<bfloat16_t><<<blockNum, nullptr, stream>>>(
            x, residual, gamma, bias, output, outer, dSize, epsilon, blockNum);
    }
}
