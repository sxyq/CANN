#include <stdint.h>
#include "kernel_operator.h"

struct TensorInfo;
struct TensorGroupInfo;
typedef void* aclrtStream;

constexpr float RNB_EPSILON = 1.0e-5f;

struct RnbTensorInfoView {
    const int64_t* shape;
    int64_t numDims;
    int32_t dtype;
};

struct RnbTensorGroupInfoView {
    const RnbTensorInfoView* tensors;
    int64_t numTensors;
};

struct AddRmsNormBiasTilingData {
    uint32_t blockNum;
    uint32_t outer;
    uint32_t dim;
    uint32_t alignedDim;
    uint32_t rowsPerBlock;
    uint32_t tailRows;
    float epsilon;
};

class KernelAddRmsNormBias {
public:
    __aicore__ inline explicit KernelAddRmsNormBias(AscendC::TPipe* pipe) : pipe_(pipe) {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias,
                                GM_ADDR out, uint32_t blockNum, uint32_t dim,
                                uint32_t alignedDim, uint32_t rowsPerBlock,
                                uint32_t tailRows, float invDim, float epsilon)
    {
        blockNum_ = blockNum;
        dim_ = dim;
        alignedDim_ = alignedDim;
        rowsPerBlock_ = rowsPerBlock;
        tailRows_ = tailRows;
        invDim_ = invDim;
        epsilon_ = epsilon;

        const uint32_t blockIdx = AscendC::GetBlockIdx();
        rowCount_ = (blockIdx + 1 == blockNum_) ? tailRows_ : rowsPerBlock_;
        rowStart_ = blockIdx * rowsPerBlock_;
        xGm_.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(x));
        residualGm_.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(residual));
        gammaGm_.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(gamma));
        biasGm_.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(bias));
        outGm_.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(out));

        const uint32_t rowBytes = alignedDim_ * sizeof(float);
        pipe_->InitBuffer(xBuf_, rowBytes);
        pipe_->InitBuffer(residualBuf_, rowBytes);
        pipe_->InitBuffer(gammaBuf_, rowBytes);
        pipe_->InitBuffer(biasBuf_, rowBytes);
        pipe_->InitBuffer(outBuf_, rowBytes);
        pipe_->InitBuffer(squareBuf_, rowBytes);
        pipe_->InitBuffer(reduceBuf_, rowBytes);
        pipe_->InitBuffer(scalarBuf_, 2 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = 0; row < rowCount_; ++row) {
            ProcessRow(rowStart_ + row);
        }
    }

private:
    __aicore__ inline void CopyIn(AscendC::LocalTensor<float>& dst,
                                  AscendC::GlobalTensor<float>& src, uint32_t row)
    {
        AscendC::DataCopyExtParams copyParams{
            1, static_cast<uint32_t>(dim_ * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> padParams{
            true, 0, static_cast<uint8_t>(alignedDim_ - dim_), 0.0f};
        AscendC::DataCopyPad(dst, src[row * dim_], copyParams, padParams);
    }

    __aicore__ inline void ProcessRow(uint32_t row)
    {
        auto x = xBuf_.Get<float>();
        auto residual = residualBuf_.Get<float>();
        auto gamma = gammaBuf_.Get<float>();
        auto bias = biasBuf_.Get<float>();
        auto out = outBuf_.Get<float>();
        auto square = squareBuf_.Get<float>();
        auto reduceTmp = reduceBuf_.Get<float>();
        auto scalars = scalarBuf_.Get<float>();

        CopyIn(x, xGm_, row);
        CopyIn(residual, residualGm_, row);
        CopyIn(gamma, gammaGm_, 0);
        CopyIn(bias, biasGm_, 0);
        AscendC::Add(out, x, residual, dim_);
        AscendC::Mul(square, out, out, dim_);
        AscendC::ReduceSum<float, true>(
            scalars, square, reduceTmp, static_cast<int32_t>(dim_));
        AscendC::Muls(scalars, scalars, invDim_, 1);
        AscendC::Adds(scalars, scalars, epsilon_, 1);
        AscendC::Sqrt(scalars, scalars, 1);
        AscendC::Duplicate(scalars[1], 1.0f, 1);
        AscendC::Div(scalars, scalars[1], scalars, 1);
        AscendC::Muls(out, out, scalars.GetValue(0), dim_);
        AscendC::Mul(out, out, gamma, dim_);
        AscendC::Add(out, out, bias, dim_);

        AscendC::DataCopyExtParams copyParams{
            1, static_cast<uint32_t>(dim_ * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPad(outGm_[row * dim_], out, copyParams);
    }

    AscendC::TPipe* pipe_;
    AscendC::GlobalTensor<float> xGm_, residualGm_, gammaGm_, biasGm_, outGm_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf_, residualBuf_, gammaBuf_, biasBuf_, outBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> squareBuf_, reduceBuf_, scalarBuf_;
    uint32_t blockNum_ = 1;
    uint32_t dim_ = 0;
    uint32_t alignedDim_ = 0;
    uint32_t rowsPerBlock_ = 0;
    uint32_t tailRows_ = 0;
    uint32_t rowStart_ = 0;
    uint32_t rowCount_ = 0;
    float invDim_ = 0.0f;
    float epsilon_ = RNB_EPSILON;
};

__aicore__ inline void RunAddRmsNormBias(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR out,
    uint32_t blockNum, uint32_t dim, uint32_t alignedDim,
    uint32_t rowsPerBlock, uint32_t tailRows, float invDim, float epsilon)
{
    AscendC::TPipe pipe;
    KernelAddRmsNormBias op(&pipe);
    op.Init(x, residual, gamma, bias, out, blockNum, dim, alignedDim,
            rowsPerBlock, tailRows, invDim, epsilon);
    op.Process();
}

extern "C" __global__ __vector__ void add_rms_norm_bias_kernel(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR out, GM_ADDR tiling)
{
    const __gm__ AddRmsNormBiasTilingData* tilingData =
        reinterpret_cast<const __gm__ AddRmsNormBiasTilingData*>(tiling);
    RunAddRmsNormBias(x, residual, gamma, bias, out,
                      tilingData->blockNum, tilingData->dim,
                      tilingData->alignedDim, tilingData->rowsPerBlock,
                      tilingData->tailRows, 0.000244140625f,
                      tilingData->epsilon);
}

extern "C" __global__ __vector__ void add_rms_norm_bias_custom(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR out,
    uint32_t blockNum, uint32_t dim, uint32_t alignedDim,
    uint32_t rowsPerBlock, uint32_t tailRows, float invDim, float epsilon)
{
    RunAddRmsNormBias(x, residual, gamma, bias, out, blockNum, dim, alignedDim,
                      rowsPerBlock, tailRows, invDim, epsilon);
}

extern "C" void run_kernel(
    GM_ADDR x, const TensorGroupInfo& info_x,
    GM_ADDR residual, const TensorGroupInfo& info_residual,
    GM_ADDR gamma, const TensorGroupInfo& info_gamma,
    GM_ADDR bias, const TensorGroupInfo& info_bias,
    GM_ADDR out, const TensorGroupInfo& info_out,
    int64_t availableCoreNum, aclrtStream stream, float epsilon)
{
    if (x == nullptr || residual == nullptr || gamma == nullptr || bias == nullptr ||
        out == nullptr) {
        return;
    }

    const RnbTensorGroupInfoView& xGroup =
        reinterpret_cast<const RnbTensorGroupInfoView&>(info_x);
    const RnbTensorGroupInfoView& residualGroup =
        reinterpret_cast<const RnbTensorGroupInfoView&>(info_residual);
    const RnbTensorGroupInfoView& gammaGroup =
        reinterpret_cast<const RnbTensorGroupInfoView&>(info_gamma);
    const RnbTensorGroupInfoView& biasGroup =
        reinterpret_cast<const RnbTensorGroupInfoView&>(info_bias);
    const RnbTensorGroupInfoView& outGroup =
        reinterpret_cast<const RnbTensorGroupInfoView&>(info_out);
    if (xGroup.tensors == nullptr || residualGroup.tensors == nullptr ||
        gammaGroup.tensors == nullptr || biasGroup.tensors == nullptr ||
        outGroup.tensors == nullptr || xGroup.numTensors < 1 ||
        residualGroup.numTensors < 1 || gammaGroup.numTensors < 1 ||
        biasGroup.numTensors < 1 || outGroup.numTensors < 1) {
        return;
    }

    const RnbTensorInfoView& xInfo = xGroup.tensors[0];
    const RnbTensorInfoView& residualInfo = residualGroup.tensors[0];
    const RnbTensorInfoView& gammaInfo = gammaGroup.tensors[0];
    const RnbTensorInfoView& biasInfo = biasGroup.tensors[0];
    const RnbTensorInfoView& outInfo = outGroup.tensors[0];
    if (xInfo.shape == nullptr || residualInfo.shape == nullptr ||
        gammaInfo.shape == nullptr || biasInfo.shape == nullptr ||
        outInfo.shape == nullptr || xInfo.numDims < 1 ||
        residualInfo.numDims != xInfo.numDims || outInfo.numDims != xInfo.numDims ||
        gammaInfo.numDims != 1 || biasInfo.numDims != 1 || xInfo.dtype != 0 ||
        residualInfo.dtype != xInfo.dtype || gammaInfo.dtype != xInfo.dtype ||
        biasInfo.dtype != xInfo.dtype || outInfo.dtype != xInfo.dtype) {
        return;
    }

    const int64_t dimSigned = xInfo.shape[xInfo.numDims - 1];
    if (dimSigned <= 0 || dimSigned > 0x7fffffffLL ||
        gammaInfo.shape[0] != dimSigned || biasInfo.shape[0] != dimSigned ||
        residualInfo.shape[xInfo.numDims - 1] != dimSigned ||
        outInfo.shape[xInfo.numDims - 1] != dimSigned) {
        return;
    }

    uint64_t outer = 1;
    for (int64_t axis = 0; axis + 1 < xInfo.numDims; ++axis) {
        const int64_t extent = xInfo.shape[axis];
        if (extent <= 0 || residualInfo.shape[axis] != extent ||
            outInfo.shape[axis] != extent) {
            return;
        }
        outer *= static_cast<uint64_t>(extent);
    }
    if (outer == 0 || outer > 0xffffffffULL || availableCoreNum <= 0) {
        return;
    }

    const uint64_t dim = static_cast<uint64_t>(dimSigned);
    const uint64_t rowBytes = dim * sizeof(float);
    uint64_t gcdA = rowBytes;
    uint64_t gcdB = 32;
    while (gcdB != 0) {
        const uint64_t remainder = gcdA % gcdB;
        gcdA = gcdB;
        gcdB = remainder;
    }
    const uint64_t rowGroup = 32 / gcdA;
    const uint64_t groupCount = (outer + rowGroup - 1) / rowGroup;
    uint64_t blockCount64 = static_cast<uint64_t>(availableCoreNum);
    if (blockCount64 > groupCount) {
        blockCount64 = groupCount;
    }
    if (blockCount64 == 0 || blockCount64 > 0xffffffffULL) {
        return;
    }
    const uint64_t groupsPerBlock =
        (groupCount + blockCount64 - 1) / blockCount64;
    const uint64_t rowsPerBlock = groupsPerBlock * rowGroup;
    const uint64_t tailRows = outer - (blockCount64 - 1) * rowsPerBlock;
    if (dim > 0xffffffffULL || rowsPerBlock > 0xffffffffULL ||
        tailRows > 0xffffffffULL) {
        return;
    }
    const uint32_t blockCount = static_cast<uint32_t>(blockCount64);
    const uint32_t dim32 = static_cast<uint32_t>(dim);
    const uint32_t alignedDim = (dim32 + 7U) / 8U * 8U;

    add_rms_norm_bias_custom<<<blockCount, nullptr, stream>>>(
        x, residual, gamma, bias, out, blockCount, dim32, alignedDim,
        static_cast<uint32_t>(rowsPerBlock), static_cast<uint32_t>(tailRows),
        1.0f / static_cast<float>(dim32), epsilon);
}
