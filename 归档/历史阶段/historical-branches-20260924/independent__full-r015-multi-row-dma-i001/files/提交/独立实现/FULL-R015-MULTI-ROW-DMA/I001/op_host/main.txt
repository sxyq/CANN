#include <cstdint>
#include <vector>
#include "acl/acl.h"
#ifndef TENSOR_GROUP_INFO_DEFINED
#define TENSOR_GROUP_INFO_DEFINED
struct TensorInfo {
    const int64_t* shape;
    int64_t numDims;
    int32_t dtype;
};
struct TensorGroupInfo {
    const TensorInfo* tensors;
    int64_t numTensors;
};
#endif
#include "../op_kernel/kernel.asc"

int main()
{
    int64_t shape[] = {8, 1024};
    TensorInfo tensor[] = {{shape, 2, 0}};
    TensorGroupInfo info = {tensor, 1};
    run_kernel(nullptr, info, nullptr, info, nullptr, info, nullptr, info,
               nullptr, info, 1, nullptr, 1e-5f);
    return 0;
}
