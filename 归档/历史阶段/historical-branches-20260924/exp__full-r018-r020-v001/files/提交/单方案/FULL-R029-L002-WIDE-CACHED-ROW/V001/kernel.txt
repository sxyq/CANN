#include <cmath>
#include <cstdint>
#include "kernel_operator.h"

/*
 * AddRmsNormBias - FULL-R029-L002-WIDE-CACHED-ROW
 *
 * Parent:
 *   MIX-R014-R002-R019-V001
 *
 * Combination status:
 *   R029-V002 FP16 wide cached-y: independently online validated 15/15.
 *   R029-V003 BF16 wide cached-y: independently online validated 15/15,
 *   official platform score 28.68.
 *
 * L002 mutation:
 *   Keep the unified native-T retained-y row cache, but use a 7680-element
 *   wide segment.  The smaller segment keeps the same wide-row ownership and
 *   two-pass reuse while leaving more UB headroom at the tail boundary.
 *
 * FP16 path:
 *   - 8192 < D <= 32768
 *   - rowCount <= blockCount
 *   - compute RMS from FP32 y
 *   - retain y in FP16 UB cache
 *   - skip the second x/residual GM read
 *
 * BF16 path:
 *   - 8192 < D <= 32768
 *   - rowCount <= blockCount
 *   - compute RMS from FP32 y
 *   - retain y in FP16 UB cache
 *   - skip the second x/residual GM read
 *
 * Existing parent mechanisms remain unchanged outside those two branches:
 *   - R014-V002 parameter residency for small rows
 *   - R002 retained-y small-row path
 *   - R019 scalar reciprocal + vector Muls normalization
 *
 * R029-V001 FP32 wide-cache is NOT included in this version because it has
 * not yet been independently validated online.
 *
 * Candidate status:
 *   FAST-BREADTH lane D candidate.
 */

template <typename T>
class AddRmsNormBiasKernel {
public:
    static constexpr int32_t kTileElems = 8192;
    static constexpr int32_t kParamCacheOffset = 4096;
    static constexpr uint64_t kParamCacheMaxWidth = 4096ULL;
    static constexpr int32_t kScalarElems = 16;
    static constexpr int32_t kWideCachedTileElems = 7680;

    static constexpr int32_t kWideFp16CachedTileElems = 7680;
    static constexpr uint64_t kWideFp16CachedRowMaxWidth = 32768ULL;

    static constexpr int32_t kWideBf16CachedTileElems = 7680;
    static constexpr uint64_t kWideBf16CachedRowMaxWidth = 32768ULL;

    __aicore__ inline void Init(
        GM_ADDR x,
        GM_ADDR residual,
        GM_ADDR gamma,
        GM_ADDR bias,
        GM_ADDR output,
        uint64_t rowWidth,
        uint64_t rowCount,
        uint32_t blockCount)
    {
        xGm_.SetGlobalBuffer((__gm__ T *)x);
        residualGm_.SetGlobalBuffer((__gm__ T *)residual);
        gammaGm_.SetGlobalBuffer((__gm__ T *)gamma);
        biasGm_.SetGlobalBuffer((__gm__ T *)bias);
        outputGm_.SetGlobalBuffer((__gm__ T *)output);

        wideFp16CachedMode_ = false;
        wideBf16CachedMode_ = false;

        if constexpr (
            AscendC::IsSameType<T, half>::value) {

            wideFp16CachedMode_ =
                rowWidth >
                    static_cast<uint64_t>(
                        kTileElems) &&
                rowWidth <=
                    kWideFp16CachedRowMaxWidth &&
                rowCount <=
                    static_cast<uint64_t>(
                        blockCount);
        }

        if constexpr (
            AscendC::IsSameType<
                T,
                bfloat16_t>::value) {

            wideBf16CachedMode_ =
                rowWidth >
                    static_cast<uint64_t>(
                        kTileElems) &&
                rowWidth <=
                    kWideBf16CachedRowMaxWidth &&
                rowCount <=
                    static_cast<uint64_t>(
                        blockCount);
        }

        /*
         * Shared FP16/BF16 wide cached-row UB layout.
         *
         * Approximate D=32768 payload:
         *
         * x native tile        15 KiB
         * residual native     15 KiB
         * full 16-bit y       64 KiB
         * value FP32          30 KiB
         * x FP32              30 KiB
         * residual FP32       30 KiB
         *
         * ~= 184 KiB + scalar slots.
         */
        if (wideFp16CachedMode_ ||
            wideBf16CachedMode_) {

            const int32_t wideTileElems =
                kWideCachedTileElems;

            tpipe.InitBuffer(
                xBuf_,
                wideTileElems *
                    sizeof(T));

            tpipe.InitBuffer(
                residualBuf_,
                wideTileElems *
                    sizeof(T));

            tpipe.InitBuffer(
                outputBuf_,
                rowWidth *
                    sizeof(T));

            tpipe.InitBuffer(
                valueFp32Buf_,
                wideTileElems *
                    sizeof(float));

            tpipe.InitBuffer(
                xFp32Buf_,
                wideTileElems *
                    sizeof(float));

            tpipe.InitBuffer(
                residualFp32Buf_,
                wideTileElems *
                    sizeof(float));

            tpipe.InitBuffer(
                sumBuf_,
                kScalarElems *
                    sizeof(float));

            return;
        }

        /*
         * Existing parent allocation.
         */
        tpipe.InitBuffer(
            xBuf_,
            kTileElems *
                sizeof(T));

        tpipe.InitBuffer(
            residualBuf_,
            kTileElems *
                sizeof(T));

        tpipe.InitBuffer(
            xFp32Buf_,
            kTileElems *
                sizeof(float));

        tpipe.InitBuffer(
            residualFp32Buf_,
            kTileElems *
                sizeof(float));

        tpipe.InitBuffer(
            yFp32Buf_,
            kTileElems *
                sizeof(float));

        tpipe.InitBuffer(
            sumBuf_,
            kScalarElems *
                sizeof(float));

        if constexpr (
            !AscendC::IsSameType<
                T,
                float>::value) {

            tpipe.InitBuffer(
                gammaFp32Buf_,
                kParamCacheOffset *
                    sizeof(float));

            tpipe.InitBuffer(
                biasFp32Buf_,
                kParamCacheOffset *
                    sizeof(float));
        }
    }

    __aicore__ inline void Process(
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowGroup,
        uint64_t groupCount,
        uint32_t blockCount,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t blockIdx =
            static_cast<uint64_t>(
                AscendC::GetBlockIdx());

        const uint64_t baseGroups =
            groupCount /
            static_cast<uint64_t>(
                blockCount);

        const uint64_t extraGroups =
            groupCount %
            static_cast<uint64_t>(
                blockCount);

        const uint64_t beginGroup =
            blockIdx *
                baseGroups +
            (blockIdx < extraGroups
                 ? blockIdx
                 : extraGroups);

        const uint64_t localGroups =
            baseGroups +
            (blockIdx < extraGroups
                 ? 1ULL
                 : 0ULL);

        const uint64_t endGroup =
            beginGroup +
            localGroups;

        const uint64_t beginRow =
            beginGroup *
            rowGroup;

        uint64_t endRow =
            endGroup *
            rowGroup;

        if (endRow > rowCount) {
            endRow = rowCount;
        }

        const uint64_t localRows =
            endRow > beginRow
                ? endRow - beginRow
                : 0ULL;

        /*
         * R029-V002 component.
         */
        if constexpr (
            AscendC::IsSameType<
                T,
                half>::value) {

            if (wideFp16CachedMode_) {

                for (uint64_t row = beginRow;
                     row < endRow;
                     ++row) {

                    ProcessWideCachedRow(
                        row,
                        rowWidth,
                        invRowWidth,
                        epsilon);
                }

                return;
            }
        }

        /*
         * R029-V003 component.
         */
        if constexpr (
            AscendC::IsSameType<
                T,
                bfloat16_t>::value) {

            if (wideBf16CachedMode_) {

                for (uint64_t row = beginRow;
                     row < endRow;
                     ++row) {

                    ProcessWideCachedRow(
                        row,
                        rowWidth,
                        invRowWidth,
                        epsilon);
                }

                return;
            }
        }

        const bool useParamCache =
            rowWidth <=
                kParamCacheMaxWidth &&
            localRows > 1ULL;

        if (useParamCache) {

            PrepareParameterCache(
                rowWidth);

            for (uint64_t row = beginRow;
                 row < endRow;
                 ++row) {

                ProcessCachedParamsRetainedYRow(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);
            }

        } else {

            for (uint64_t row = beginRow;
                 row < endRow;
                 ++row) {

                ProcessRow(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);
            }
        }
    }

private:

    __aicore__ inline int32_t TileLength(
        uint64_t remain) const
    {
        return static_cast<int32_t>(
            remain >
                    static_cast<uint64_t>(
                        kTileElems)
                ? kTileElems
                : remain);
    }

    __aicore__ inline int32_t
    WideFp16CachedTileLength(
        uint64_t remain) const
    {
        return static_cast<int32_t>(
            remain >
                    static_cast<uint64_t>(
                        kWideFp16CachedTileElems)
                ? kWideFp16CachedTileElems
                : remain);
    }

    __aicore__ inline int32_t
    WideBf16CachedTileLength(
        uint64_t remain) const
    {
        return static_cast<int32_t>(
            remain >
                    static_cast<uint64_t>(
                        kWideBf16CachedTileElems)
                ? kWideBf16CachedTileElems
                : remain);
    }

    __aicore__ inline void Load(
        const AscendC::LocalTensor<T>& dst,
        const AscendC::GlobalTensor<T>& src,
        uint64_t offset,
        int32_t count)
    {
        const AscendC::DataCopyExtParams
            copyParams(
                1,
                static_cast<uint32_t>(
                    count *
                    static_cast<int32_t>(
                        sizeof(T))),
                0,
                0,
                0);

        const AscendC::
            DataCopyPadExtParams<T>
                padParams;

        AscendC::DataCopyPad(
            dst,
            src[offset],
            copyParams,
            padParams);
    }

    __aicore__ inline void Store(
        const AscendC::GlobalTensor<T>& dst,
        uint64_t offset,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        const AscendC::DataCopyExtParams
            copyParams(
                1,
                static_cast<uint32_t>(
                    count *
                    static_cast<int32_t>(
                        sizeof(T))),
                0,
                0,
                0);

        AscendC::DataCopyPad(
            dst[offset],
            src,
            copyParams);
    }

    __aicore__ inline void ToFloat(
        const AscendC::LocalTensor<float>& dst,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::Adds(
                dst,
                src,
                0.0f,
                count);

        } else {

            AscendC::Cast(
                dst,
                src,
                AscendC::RoundMode::
                    CAST_NONE,
                count);
        }
    }

    __aicore__ inline void SyncMTE2ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE2_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE2_V>(0);
    }

    __aicore__ inline void SyncVToMTE2()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE2>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE2>(0);
    }

    __aicore__ inline void SyncVToMTE3()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE3>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE3>(0);
    }

    __aicore__ inline void SyncMTE3ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE3_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE3_V>(0);
    }

    __aicore__ inline void SyncVToS()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_S>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_S>(0);
    }

    __aicore__ inline void SyncSToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::S_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::S_V>(0);
    }

    /*
     * =============================================================
     * L001 unified wide-row retained-y path
     * =============================================================
     *
     * The full row is owned by cachedY for both half and bfloat16_t.  Pass 1
     * reads x/residual once, computes the FP32 RMS state, and retains y in
     * the row cache.  Pass 2 consumes that cache with only parameter reads.
     */
    __aicore__ inline void
    ProcessWideCachedRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<T> cachedY =
            outputBuf_.Get<T>();

        AscendC::LocalTensor<float> valueFp32 =
            valueFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        int32_t tileIndex = 0;

        for (uint64_t col = 0;
             col < rowWidth;
             col +=
                 kWideCachedTileElems) {

            const int32_t valid =
                static_cast<int32_t>(
                    rowWidth - col >
                            static_cast<uint64_t>(
                                kWideCachedTileElems)
                        ? kWideCachedTileElems
                        : rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                valueFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Cast(
                cachedY[col],
                valueFp32,
                AscendC::RoundMode::CAST_RINT,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                residualFp32,
                valueFp32,
                valueFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum(
                sumSlots[tileIndex],
                residualFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            ++tileIndex;
            SyncVToMTE2();
        }

        AscendC::ReduceSum(
            residualFp32,
            sumSlots,
            xFp32,
            tileIndex);

        AscendC::PipeBarrier<PIPE_V>();
        SyncVToS();

        const float squareSum =
            residualFp32.GetValue(0);

        const float meanSquare =
            squareSum *
                invRowWidth +
            epsilon;

        SyncSToV();

        AscendC::Duplicate(
            xFp32,
            meanSquare,
            1);

        AscendC::Sqrt(
            xFp32,
            xFp32,
            1);

        AscendC::PipeBarrier<PIPE_V>();
        SyncVToS();

        const float invRms =
            1.0f /
            xFp32.GetValue(0);

        SyncSToV();
        SyncVToMTE2();

        for (uint64_t col = 0;
             col < rowWidth;
             col +=
                 kWideCachedTileElems) {

            const int32_t valid =
                static_cast<int32_t>(
                    rowWidth - col >
                            static_cast<uint64_t>(
                                kWideCachedTileElems)
                        ? kWideCachedTileElems
                        : rowWidth - col);

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                valueFp32,
                cachedY[col],
                valid);

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Muls(
                valueFp32,
                valueFp32,
                invRms,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                valueFp32,
                valueFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                valueFp32,
                valueFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Cast(
                cachedY[col],
                valueFp32,
                AscendC::RoundMode::CAST_RINT,
                valid);

            AscendC::PipeBarrier<PIPE_V>();
            SyncVToMTE2();
            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset + col,
                cachedY[col],
                valid);

            SyncMTE3ToV();
        }
    }

    /*
     * =============================================================
     * Historical split wide cached-y paths
     * =============================================================
     */
    __aicore__ inline void
    ProcessWideFp16CachedRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<T> cachedY =
            outputBuf_.Get<T>();

        AscendC::LocalTensor<float> valueFp32 =
            valueFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        int32_t tileIndex = 0;

        /*
         * Pass 1:
         * RMS uses exact FP32 y;
         * second-pass copy is stored in FP16.
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col +=
                 kWideFp16CachedTileElems) {

            const int32_t valid =
                WideFp16CachedTileLength(
                    rowWidth -
                    col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                valueFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Cast(
                cachedY[col],
                valueFp32,
                AscendC::RoundMode::
                    CAST_RINT,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Mul(
                residualFp32,
                valueFp32,
                valueFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::ReduceSum(
                sumSlots[tileIndex],
                residualFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            ++tileIndex;

            SyncVToMTE2();
        }

        /*
         * At most five partials.
         */
        AscendC::ReduceSum(
            residualFp32,
            sumSlots,
            xFp32,
            tileIndex);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToS();

        const float squareSum =
            residualFp32.GetValue(0);

        const float meanSquare =
            squareSum *
                invRowWidth +
            epsilon;

        SyncSToV();

        AscendC::Duplicate(
            xFp32,
            meanSquare,
            1);

        AscendC::Sqrt(
            xFp32,
            xFp32,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToS();

        const float invRms =
            1.0f /
            xFp32.GetValue(0);

        SyncSToV();

        SyncVToMTE2();

        /*
         * Pass 2:
         *
         * cached y + gamma/bias only.
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col +=
                 kWideFp16CachedTileElems) {

            const int32_t valid =
                WideFp16CachedTileLength(
                    rowWidth -
                    col);

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                valueFp32,
                cachedY[col],
                valid);

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Muls(
                valueFp32,
                valueFp32,
                invRms,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Mul(
                valueFp32,
                valueFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                valueFp32,
                valueFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Cast(
                cachedY[col],
                valueFp32,
                AscendC::RoundMode::
                    CAST_RINT,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            SyncVToMTE2();
            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset + col,
                cachedY[col],
                valid);

            SyncMTE3ToV();
        }
    }

    /*
     * =============================================================
     * BF16 wide cached-y path
     * =============================================================
     */
    __aicore__ inline void
    ProcessWideBf16CachedRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        /*
         * Cache y as FP16 for better mantissa retention
         * at the same 16-bit UB footprint.
         */
        AscendC::LocalTensor<half> cachedY =
            outputBuf_.Get<half>();

        AscendC::LocalTensor<T> outputLocal =
            outputBuf_.Get<T>();

        AscendC::LocalTensor<float> valueFp32 =
            valueFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        int32_t tileIndex = 0;

        /*
         * Pass 1:
         * original FP32 y is used for RMS.
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col +=
                 kWideBf16CachedTileElems) {

            const int32_t valid =
                WideBf16CachedTileLength(
                    rowWidth -
                    col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                valueFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Cast(
                cachedY[col],
                valueFp32,
                AscendC::RoundMode::
                    CAST_ROUND,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Mul(
                residualFp32,
                valueFp32,
                valueFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::ReduceSum(
                sumSlots[tileIndex],
                residualFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            ++tileIndex;

            SyncVToMTE2();
        }

        AscendC::ReduceSum(
            residualFp32,
            sumSlots,
            xFp32,
            tileIndex);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToS();

        const float squareSum =
            residualFp32.GetValue(0);

        const float meanSquare =
            squareSum *
                invRowWidth +
            epsilon;

        SyncSToV();

        AscendC::Duplicate(
            xFp32,
            meanSquare,
            1);

        AscendC::Sqrt(
            xFp32,
            xFp32,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToS();

        const float invRms =
            1.0f /
            xFp32.GetValue(0);

        SyncSToV();

        SyncVToMTE2();

        /*
         * Pass 2:
         * no second x/residual GM read.
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col +=
                 kWideBf16CachedTileElems) {

            const int32_t valid =
                WideBf16CachedTileLength(
                    rowWidth -
                    col);

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            AscendC::Cast(
                valueFp32,
                cachedY[col],
                AscendC::RoundMode::
                    CAST_NONE,
                valid);

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Muls(
                valueFp32,
                valueFp32,
                invRms,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Mul(
                valueFp32,
                valueFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                valueFp32,
                valueFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Cast(
                outputLocal[col],
                valueFp32,
                AscendC::RoundMode::
                    CAST_RINT,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            SyncVToMTE2();
            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset + col,
                outputLocal[col],
                valid);

            SyncMTE3ToV();
        }
    }

    /*
     * =============================================================
     * R014-V002 parameter cache
     * =============================================================
     */
    __aicore__ inline void
    PrepareParameterCache(
        uint64_t rowWidth)
    {
        const int32_t valid =
            static_cast<int32_t>(
                rowWidth);

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<T> gammaCached =
            xLocal[
                kParamCacheOffset];

        AscendC::LocalTensor<T> biasCached =
            residualLocal[
                kParamCacheOffset];

        Load(
            gammaCached,
            gammaGm_,
            0,
            valid);

        Load(
            biasCached,
            biasGm_,
            0,
            valid);

        SyncMTE2ToV();

        if constexpr (
            !AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::LocalTensor<float>
                gammaFp32 =
                    gammaFp32Buf_
                        .Get<float>();

            AscendC::LocalTensor<float>
                biasFp32 =
                    biasFp32Buf_
                        .Get<float>();

            ToFloat(
                gammaFp32,
                gammaCached,
                valid);

            ToFloat(
                biasFp32,
                biasCached,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            SyncVToMTE2();
        }
    }

    /*
     * =============================================================
     * Small-row:
     * R014 + R002 + R019
     * =============================================================
     */
    __aicore__ inline void
    ProcessCachedParamsRetainedYRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        const int32_t valid =
            static_cast<int32_t>(
                rowWidth);

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<T> gammaCached =
            xLocal[
                kParamCacheOffset];

        AscendC::LocalTensor<T> biasCached =
            residualLocal[
                kParamCacheOffset];

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        Load(
            xLocal,
            xGm_,
            rowOffset,
            valid);

        Load(
            residualLocal,
            residualGm_,
            rowOffset,
            valid);

        SyncMTE2ToV();

        ToFloat(
            xFp32,
            xLocal,
            valid);

        ToFloat(
            residualFp32,
            residualLocal,
            valid);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Add(
            yFp32,
            xFp32,
            residualFp32,
            valid);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Mul(
            xFp32,
            yFp32,
            yFp32,
            valid);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::ReduceSum(
            partial,
            xFp32,
            residualFp32,
            valid);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Add(
            rms,
            rms,
            partial,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToMTE2();

        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToS();

        const float invRms =
            1.0f /
            rms.GetValue(0);

        SyncSToV();

        AscendC::Muls(
            yFp32,
            yFp32,
            invRms,
            valid);

        AscendC::PipeBarrier<
            PIPE_V>();

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::Mul(
                yFp32,
                yFp32,
                gammaCached,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                biasCached,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

        } else {

            AscendC::LocalTensor<float>
                gammaFp32 =
                    gammaFp32Buf_
                        .Get<float>();

            AscendC::LocalTensor<float>
                biasFp32 =
                    biasFp32Buf_
                        .Get<float>();

            AscendC::Mul(
                yFp32,
                yFp32,
                gammaFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                biasFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();
        }

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset,
                yFp32,
                valid);

            SyncMTE3ToV();

        } else {

            AscendC::Cast(
                xLocal,
                yFp32,
                AscendC::RoundMode::
                    CAST_RINT,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset,
                xLocal,
                valid);

            SyncMTE3ToV();
        }

        SyncVToMTE2();
    }

    /*
     * =============================================================
     * Generic fallback
     * =============================================================
     */
    __aicore__ inline void ProcessRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        /*
         * Pass 1.
         */
        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth -
                    col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Mul(
                xFp32,
                yFp32,
                yFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::ReduceSum(
                partial,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                rms,
                rms,
                partial,
                1);

            AscendC::PipeBarrier<
                PIPE_V>();

            SyncVToMTE2();
        }

        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<
            PIPE_V>();

        SyncVToS();

        const float invRms =
            1.0f /
            rms.GetValue(0);

        SyncSToV();

        /*
         * Pass 2.
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth -
                    col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Muls(
                yFp32,
                yFp32,
                invRms,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            SyncVToMTE2();

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Mul(
                yFp32,
                yFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<
                PIPE_V>();

            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    yFp32,
                    valid);

                SyncMTE3ToV();

            } else {

                AscendC::Cast(
                    xLocal,
                    yFp32,
                    AscendC::RoundMode::
                        CAST_RINT,
                    valid);

                AscendC::PipeBarrier<
                    PIPE_V>();

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    xLocal,
                    valid);

                SyncMTE3ToV();
            }

            SyncVToMTE2();
        }
    }

private:

    bool wideFp16CachedMode_;
    bool wideBf16CachedMode_;

    AscendC::TPipe tpipe;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        yFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        outputBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        valueFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        sumBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        gammaFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        biasFp32Buf_;

    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;
};


template <typename T>
__global__ __vector__ void
add_rms_norm_bias_custom(
    GM_ADDR x,
    GM_ADDR residual,
    GM_ADDR gamma,
    GM_ADDR bias,
    GM_ADDR output,
    uint64_t rowCount,
    uint64_t rowWidth,
    uint64_t rowGroup,
    uint64_t groupCount,
    uint32_t blockCount,
    float invRowWidth,
    float epsilon)
{
    AddRmsNormBiasKernel<T> op;

    op.Init(
        x,
        residual,
        gamma,
        bias,
        output,
        rowWidth,
        rowCount,
        blockCount);

    op.Process(
        rowCount,
        rowWidth,
        rowGroup,
        groupCount,
        blockCount,
        invRowWidth,
        epsilon);
}


#if !defined(ASCENDC_CPU_DEBUG)

extern "C" void run_kernel(
    GM_ADDR x,
    const TensorGroupInfo& info_x,

    GM_ADDR residual,
    const TensorGroupInfo& info_residual,

    GM_ADDR gamma,
    const TensorGroupInfo& info_gamma,

    GM_ADDR bias,
    const TensorGroupInfo& info_bias,

    GM_ADDR output,
    const TensorGroupInfo& info_output,

    int64_t availableCoreNum,
    aclrtStream stream,
    float epsilon)
{
    if (x == nullptr ||
        residual == nullptr ||
        gamma == nullptr ||
        bias == nullptr ||
        output == nullptr ||

        info_x.tensors == nullptr ||
        info_residual.tensors == nullptr ||
        info_gamma.tensors == nullptr ||
        info_bias.tensors == nullptr ||
        info_output.tensors == nullptr ||

        info_x.numTensors < 1 ||
        info_residual.numTensors < 1 ||
        info_gamma.numTensors < 1 ||
        info_bias.numTensors < 1 ||
        info_output.numTensors < 1) {

        return;
    }

    const TensorInfo& xInfo =
        info_x.tensors[0];

    const TensorInfo& residualInfo =
        info_residual.tensors[0];

    const TensorInfo& gammaInfo =
        info_gamma.tensors[0];

    const TensorInfo& biasInfo =
        info_bias.tensors[0];

    const TensorInfo& outputInfo =
        info_output.tensors[0];

    if (xInfo.shape == nullptr ||
        residualInfo.shape == nullptr ||
        gammaInfo.shape == nullptr ||
        biasInfo.shape == nullptr ||
        outputInfo.shape == nullptr ||

        xInfo.numDims < 1 ||

        residualInfo.numDims !=
            xInfo.numDims ||

        outputInfo.numDims !=
            xInfo.numDims ||

        gammaInfo.numDims != 1 ||
        biasInfo.numDims != 1 ||

        xInfo.dtype !=
            residualInfo.dtype ||

        xInfo.dtype !=
            gammaInfo.dtype ||

        xInfo.dtype !=
            biasInfo.dtype ||

        xInfo.dtype !=
            outputInfo.dtype) {

        return;
    }

    if (xInfo.dtype != 0 &&
        xInfo.dtype != 1 &&
        xInfo.dtype != 2) {

        return;
    }

    const int64_t widthSigned =
        xInfo.shape[
            xInfo.numDims - 1];

    if (widthSigned <= 0 ||
        gammaInfo.shape[0] !=
            widthSigned ||
        biasInfo.shape[0] !=
            widthSigned) {

        return;
    }

    const uint64_t maxU64 =
        ~static_cast<uint64_t>(0);

    uint64_t rowCount = 1;

    for (int64_t dim = 0;
         dim < xInfo.numDims - 1;
         ++dim) {

        const int64_t extentSigned =
            xInfo.shape[dim];

        if (extentSigned <= 0) {
            return;
        }

        if (residualInfo.shape[dim] !=
                extentSigned ||
            outputInfo.shape[dim] !=
                extentSigned) {

            return;
        }

        const uint64_t extent =
            static_cast<uint64_t>(
                extentSigned);

        if (rowCount >
            maxU64 /
                extent) {

            return;
        }

        rowCount *= extent;
    }

    if (residualInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        outputInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        rowCount == 0) {

        return;
    }

    const uint64_t rowWidth =
        static_cast<uint64_t>(
            widthSigned);

    const uint64_t elementBytes =
        xInfo.dtype == 0
            ? 4ULL
            : 2ULL;

    if (rowWidth >
            maxU64 /
                elementBytes ||
        rowCount >
            maxU64 /
                rowWidth) {

        return;
    }

    const uint64_t rowBytes =
        rowWidth *
        elementBytes;

    uint64_t gcdA =
        rowBytes;

    uint64_t gcdB =
        32ULL;

    while (gcdB != 0ULL) {

        const uint64_t remainder =
            gcdA %
            gcdB;

        gcdA = gcdB;
        gcdB = remainder;
    }

    const uint64_t rowGroup =
        32ULL /
        gcdA;

    const uint64_t groupCount =
        rowCount /
            rowGroup +
        (rowCount %
                    rowGroup ==
                0ULL
             ? 0ULL
             : 1ULL);

    uint64_t requestedBlocks =
        availableCoreNum > 0
            ? static_cast<uint64_t>(
                  availableCoreNum)
            : 1ULL;

    if (requestedBlocks >
        groupCount) {

        requestedBlocks =
            groupCount;
    }

    const uint64_t maxBlockCount =
        static_cast<uint64_t>(
            0xffffffffU);

    if (requestedBlocks >
        maxBlockCount) {

        requestedBlocks =
            maxBlockCount;
    }

    if (requestedBlocks == 0ULL) {
        requestedBlocks = 1ULL;
    }

    const uint32_t blockCount =
        static_cast<uint32_t>(
            requestedBlocks);

    const float invRowWidth =
        1.0f /
        static_cast<float>(
            widthSigned);

    if (xInfo.dtype == 0) {

        add_rms_norm_bias_custom<float>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else if (xInfo.dtype == 1) {

        add_rms_norm_bias_custom<half>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else {

        add_rms_norm_bias_custom<bfloat16_t>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);
    }
}
