#ifndef TENSOR_GROUP_INFO_DEFINED
#define TENSOR_GROUP_INFO_DEFINED
struct TensorInfo {
    const int64_t* shape;
    int64_t numDims;
    int32_t dtype;
};
struct TensorGroupInfo {
    const TensorInfo* tensors;
    int64_t numTensors;
};
#endif

namespace {

using namespace AscendC;

constexpr int32_t kTileElements = 128;

template <typename T>
__aicore__ inline void LoadPadded(const GlobalTensor<T>& source, const LocalTensor<T>& target,
    uint64_t offset, int32_t validElements)
{
    const uint32_t validBytes = static_cast<uint32_t>(validElements * sizeof(T));
    const uint8_t rightPadding = static_cast<uint8_t>(kTileElements - validElements);
    DataCopyExtParams copyParams(1, validBytes, 0, 0, 0);
    DataCopyPadExtParams<T> padParams(true, 0, rightPadding, static_cast<T>(0));
    DataCopyPad(target, source[offset], copyParams, padParams);
}

template <typename T>
__aicore__ inline void StoreValid(GlobalTensor<T>& target, const LocalTensor<T>& source,
    uint64_t offset, int32_t validElements)
{
    const int32_t elementsPerBlock = static_cast<int32_t>(32 / sizeof(T));
    const int32_t copiedElements = (validElements / elementsPerBlock) * elementsPerBlock;
    if (copiedElements > 0) {
        DataCopy(target[offset], source, static_cast<uint32_t>(copiedElements));
    }
    for (int32_t i = copiedElements; i < validElements; ++i) {
        target.SetValue(offset + static_cast<uint64_t>(i), source.GetValue(i));
    }
}

template <typename T>
class AddRmsNormBiasKernel {
public:
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR output,
        int64_t outer, int32_t d, float epsilon)
    {
        xGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(x));
        residualGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(residual));
        gammaGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(gamma));
        biasGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(bias));
        outputGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(output));
        outer_ = outer;
        d_ = d;
        epsilon_ = epsilon;

        pipe_.InitBuffer(xBuf_, kTileElements * sizeof(T));
        pipe_.InitBuffer(residualBuf_, kTileElements * sizeof(T));
        pipe_.InitBuffer(gammaBuf_, kTileElements * sizeof(T));
        pipe_.InitBuffer(biasBuf_, kTileElements * sizeof(T));
        pipe_.InitBuffer(xfBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(rfBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(yBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(sqBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(gfBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(bfBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(sumBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(workBuf_, kTileElements * sizeof(float));
        pipe_.InitBuffer(outBuf_, kTileElements * sizeof(T));
    }

    __aicore__ inline void Process()
    {
        const int64_t blockIdx = GetBlockIdx();
        const int64_t blockNum = GetBlockNum();
        for (int64_t row = blockIdx; row < outer_; row += blockNum) {
            ProcessRow(row);
        }
    }

private:
    __aicore__ inline void LoadTile(int64_t row, int32_t col, int32_t validElements)
    {
        const uint64_t rowOffset = static_cast<uint64_t>(row) * static_cast<uint64_t>(d_);
        LoadPadded(xGm_, xBuf_.Get<T>(), rowOffset + static_cast<uint64_t>(col), validElements);
        LoadPadded(residualGm_, residualBuf_.Get<T>(), rowOffset + static_cast<uint64_t>(col), validElements);
        LoadPadded(gammaGm_, gammaBuf_.Get<T>(), static_cast<uint64_t>(col), validElements);
        LoadPadded(biasGm_, biasBuf_.Get<T>(), static_cast<uint64_t>(col), validElements);
    }

    __aicore__ inline void ToFloat(int32_t count)
    {
        Cast(xfBuf_.Get<float>(), xBuf_.Get<T>(), RoundMode::CAST_RINT, static_cast<uint32_t>(count));
        Cast(rfBuf_.Get<float>(), residualBuf_.Get<T>(), RoundMode::CAST_RINT, static_cast<uint32_t>(count));
        Cast(gfBuf_.Get<float>(), gammaBuf_.Get<T>(), RoundMode::CAST_RINT, static_cast<uint32_t>(count));
        Cast(bfBuf_.Get<float>(), biasBuf_.Get<T>(), RoundMode::CAST_RINT, static_cast<uint32_t>(count));
    }

    __aicore__ inline void ProcessRow(int64_t row)
    {
        float squareSum = 0.0f;
        for (int32_t col = 0; col < d_; col += kTileElements) {
            const int32_t validElements = (d_ - col < kTileElements) ? (d_ - col) : kTileElements;
            LoadTile(row, col, validElements);
            ToFloat(kTileElements);
            Add(yBuf_.Get<float>(), xfBuf_.Get<float>(), rfBuf_.Get<float>(), kTileElements);
            Mul(sqBuf_.Get<float>(), yBuf_.Get<float>(), yBuf_.Get<float>(), kTileElements);
            ReduceSum(sumBuf_.Get<float>(), sqBuf_.Get<float>(), workBuf_.Get<float>(), kTileElements, 1, 1);
            squareSum += sumBuf_.Get<float>().GetValue(0);
        }

        const float meanSquare = squareSum / static_cast<float>(d_) + epsilon_;
        const float inverseRms = 1.0f / SqrtScalar(meanSquare);
        const uint64_t rowOffset = static_cast<uint64_t>(row) * static_cast<uint64_t>(d_);
        for (int32_t col = 0; col < d_; col += kTileElements) {
            const int32_t validElements = (d_ - col < kTileElements) ? (d_ - col) : kTileElements;
            LoadTile(row, col, validElements);
            ToFloat(kTileElements);
            Add(yBuf_.Get<float>(), xfBuf_.Get<float>(), rfBuf_.Get<float>(), kTileElements);
            Muls(yBuf_.Get<float>(), yBuf_.Get<float>(), inverseRms, kTileElements);
            Mul(yBuf_.Get<float>(), yBuf_.Get<float>(), gfBuf_.Get<float>(), kTileElements);
            Add(yBuf_.Get<float>(), yBuf_.Get<float>(), bfBuf_.Get<float>(), kTileElements);
            Cast(outBuf_.Get<T>(), yBuf_.Get<float>(), RoundMode::CAST_RINT, kTileElements);
            StoreValid(outputGm_, outBuf_.Get<T>(), rowOffset + static_cast<uint64_t>(col), validElements);
        }
    }

    __aicore__ inline float SqrtScalar(float value)
    {
        sumBuf_.Get<float>().SetValue(0, value);
        Sqrt(sumBuf_.Get<float>(), sumBuf_.Get<float>(), 1);
        return sumBuf_.Get<float>().GetValue(0);
    }

    TPipe pipe_;
    TBuf<TPosition::VECCALC> xBuf_;
    TBuf<TPosition::VECCALC> residualBuf_;
    TBuf<TPosition::VECCALC> gammaBuf_;
    TBuf<TPosition::VECCALC> biasBuf_;
    TBuf<TPosition::VECCALC> xfBuf_;
    TBuf<TPosition::VECCALC> rfBuf_;
    TBuf<TPosition::VECCALC> yBuf_;
    TBuf<TPosition::VECCALC> sqBuf_;
    TBuf<TPosition::VECCALC> gfBuf_;
    TBuf<TPosition::VECCALC> bfBuf_;
    TBuf<TPosition::VECCALC> sumBuf_;
    TBuf<TPosition::VECCALC> workBuf_;
    TBuf<TPosition::VECCALC> outBuf_;
    GlobalTensor<T> xGm_;
    GlobalTensor<T> residualGm_;
    GlobalTensor<T> gammaGm_;
    GlobalTensor<T> biasGm_;
    GlobalTensor<T> outputGm_;
    int64_t outer_ = 0;
    int32_t d_ = 0;
    float epsilon_ = 0.0f;
};

template <typename T>
__global__ __aicore__ void AddRmsNormBiasKernelEntry(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias,
    GM_ADDR output, int64_t outer, int32_t d, float epsilon)
{
    AddRmsNormBiasKernel<T> kernel;
    kernel.Init(x, residual, gamma, bias, output, outer, d, epsilon);
    kernel.Process();
}

} // namespace

extern "C" void run_kernel(GM_ADDR x, const TensorGroupInfo& info_x, GM_ADDR residual,
    const TensorGroupInfo& info_residual, GM_ADDR gamma, const TensorGroupInfo& info_gamma, GM_ADDR bias,
    const TensorGroupInfo& info_bias, GM_ADDR output, const TensorGroupInfo& info_output,
    int64_t availableCoreNum, aclrtStream stream, float epsilon)
{
    (void)info_residual;
    (void)info_gamma;
    (void)info_bias;
    (void)info_output;
    const TensorInfo& xInfo = info_x.tensors[0];
    const int64_t d64 = xInfo.shape[xInfo.numDims - 1];
    int64_t outer = 1;
    for (int64_t i = 0; i + 1 < xInfo.numDims; ++i) {
        outer *= xInfo.shape[i];
    }
    const uint32_t blockNum = static_cast<uint32_t>((outer < availableCoreNum) ? outer : availableCoreNum);
    if (blockNum == 0) {
        return;
    }
    const int32_t d = static_cast<int32_t>(d64);
    if (xInfo.dtype == 0) {
        AddRmsNormBiasKernelEntry<float><<<blockNum, nullptr, stream>>>(x, residual, gamma, bias, output, outer, d, epsilon);
    } else if (xInfo.dtype == 1) {
        AddRmsNormBiasKernelEntry<half><<<blockNum, nullptr, stream>>>(x, residual, gamma, bias, output, outer, d, epsilon);
    } else {
        AddRmsNormBiasKernelEntry<bfloat16_t><<<blockNum, nullptr, stream>>>(x, residual, gamma, bias, output, outer, d, epsilon);
    }
}
