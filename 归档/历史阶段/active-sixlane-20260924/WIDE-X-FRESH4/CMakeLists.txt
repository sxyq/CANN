cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(wide_x_fresh4 LANGUAGES ASC CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ASCEND_HOME_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit root")
if(NOT ASCEND_HOME_PATH)
    message(FATAL_ERROR "ASCEND_HOME_PATH must point to CANN 8.5.0.alpha002")
endif()
set(ACL_INCLUDE_DIR "${ASCEND_HOME_PATH}/aarch64-linux/include")
set(ACL_LIB_DIR "${ASCEND_HOME_PATH}/aarch64-linux/lib64")

add_library(wide_x_fresh4 SHARED wide_x_fresh4.asc)
target_include_directories(wide_x_fresh4 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${ACL_INCLUDE_DIR}")
target_compile_definitions(wide_x_fresh4 PRIVATE FRESH4_LOCAL_BUILD)
target_compile_options(wide_x_fresh4 PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=dav-2201>)
target_link_directories(wide_x_fresh4 PRIVATE "${ACL_LIB_DIR}")
target_link_libraries(wide_x_fresh4 PRIVATE ascendcl ascendc_runtime tiling_api register platform unified_dlog graph_base dl m)
