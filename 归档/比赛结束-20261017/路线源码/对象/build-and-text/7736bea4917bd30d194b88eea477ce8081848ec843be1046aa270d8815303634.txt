#include "kernel_operator.h"

using namespace AscendC;

namespace MixAReduceFp32TwoPass {

constexpr uint32_t kTileElements = 2048;
constexpr uint32_t kPaddedTileElements = 2064;
constexpr uint32_t kReduceWorkElements = 1024;
constexpr uint32_t kReduceResultBytes = 32;

template <typename T>
__aicore__ inline uint32_t AlignedElements(uint32_t validElements)
{
    uint32_t bytes = validElements * static_cast<uint32_t>(sizeof(T));
    uint32_t alignedBytes = (bytes + 31U) & ~31U;
    return alignedBytes / static_cast<uint32_t>(sizeof(T));
}

template <typename T>
__aicore__ inline void CopyInPad(const LocalTensor<T>& dst, const GlobalTensor<T>& src,
    uint64_t offset, uint32_t validElements)
{
    uint32_t validBytes = validElements * static_cast<uint32_t>(sizeof(T));
    uint32_t alignedBytes = (validBytes + 31U) & ~31U;
    DataCopyExtParams copyParams(1, validBytes, 0, 0, 0);
    DataCopyPadExtParams<T> padParams(true, 0,
        static_cast<uint8_t>(alignedBytes - validBytes), static_cast<T>(0));
    DataCopyPad(dst, src[offset], copyParams, padParams);
}

template <typename T>
__aicore__ inline void CopyOut(const GlobalTensor<T>& dst, uint64_t offset,
    const LocalTensor<T>& src, uint32_t validElements)
{
    DataCopyExtParams copyParams(1,
        validElements * static_cast<uint32_t>(sizeof(T)), 0, 0, 0);
    DataCopyPad(dst[offset], src, copyParams);
}

template <typename T>
struct InputConverter {
    static __aicore__ inline void Run(const LocalTensor<float>& dst,
        const LocalTensor<T>& src, uint32_t count)
    {
        Cast(dst, src, RoundMode::CAST_NONE, count);
    }
};

template <>
struct InputConverter<float> {
    static __aicore__ inline void Run(const LocalTensor<float>& dst,
        const LocalTensor<float>& src, uint32_t count)
    {
        DataCopyParams copyParams(1,
            static_cast<uint16_t>(count * sizeof(float) / 32U), 0, 0);
        DataCopy(dst, src, copyParams);
    }
};

template <typename T>
struct OutputConverter {
    static __aicore__ inline void Run(const LocalTensor<T>& dst,
        const LocalTensor<float>& src, uint32_t count)
    {
        Cast(dst, src, RoundMode::CAST_RINT, count);
    }
};

template <>
struct OutputConverter<float> {
    static __aicore__ inline void Run(const LocalTensor<float>& dst,
        const LocalTensor<float>& src, uint32_t count)
    {
        DataCopyParams copyParams(1,
            static_cast<uint16_t>(count * sizeof(float) / 32U), 0, 0);
        DataCopy(dst, src, copyParams);
    }
};

template <typename T>
class RmsNormKernel {
public:
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma,
        GM_ADDR bias, GM_ADDR output, uint64_t outer, uint32_t dim, float inverseDim, float epsilon)
    {
        xGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(x), outer * dim);
        residualGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(residual), outer * dim);
        gammaGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(gamma), dim);
        biasGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(bias), dim);
        outputGm.SetGlobalBuffer(reinterpret_cast<__gm__ T *>(output), outer * dim);
        outer_ = outer;
        dim_ = dim;
        inverseDim_ = inverseDim;
        epsilon_ = epsilon;

        uint32_t rawBytes = kPaddedTileElements * static_cast<uint32_t>(sizeof(T));
        uint32_t fp32Bytes = kPaddedTileElements * static_cast<uint32_t>(sizeof(float));
        tpipe.InitBuffer(xQueue, 1, rawBytes);
        tpipe.InitBuffer(residualQueue, 1, rawBytes);
        tpipe.InitBuffer(gammaQueue, 1, rawBytes);
        tpipe.InitBuffer(biasQueue, 1, rawBytes);
        tpipe.InitBuffer(outputQueue, 1, rawBytes);
        tpipe.InitBuffer(xFp32Queue, 1, fp32Bytes);
        tpipe.InitBuffer(residualFp32Queue, 1, fp32Bytes);
        tpipe.InitBuffer(gammaFp32Queue, 1, fp32Bytes);
        tpipe.InitBuffer(biasFp32Queue, 1, fp32Bytes);
        tpipe.InitBuffer(yQueue, 1, fp32Bytes);
        tpipe.InitBuffer(squareQueue, 1, fp32Bytes);
        tpipe.InitBuffer(sumQueue, 1, kReduceResultBytes);
        tpipe.InitBuffer(workBuffer, kReduceWorkElements * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint64_t block = static_cast<uint64_t>(GetBlockIdx());
        uint64_t blockCount = static_cast<uint64_t>(GetBlockNum());
        uint64_t rowsPerBlock = outer_ / blockCount;
        uint64_t extraRows = outer_ % blockCount;
        uint64_t firstRow = block * rowsPerBlock + (block < extraRows ? block : extraRows);
        uint64_t rowCount = rowsPerBlock + (block < extraRows ? 1 : 0);
        for (uint64_t row = firstRow; row < firstRow + rowCount; ++row) {
            ProcessRow(row);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint64_t row)
    {
        uint64_t rowBase = row * static_cast<uint64_t>(dim_);
        LocalTensor<float> reduceWork = workBuffer.Get<float>();
        float squareSum = 0.0f;

        // Pass 1: form FP32 y tile by tile and accumulate its square sum.
        for (uint32_t offset = 0; offset < dim_; offset += kTileElements) {
            uint32_t validElements = dim_ - offset;
            if (validElements > kTileElements) {
                validElements = kTileElements;
            }
            uint32_t calcElements = AlignedElements<T>(validElements);

            LocalTensor<T> xRaw = xQueue.AllocTensor<T>();
            LocalTensor<T> residualRaw = residualQueue.AllocTensor<T>();
            CopyInPad(xRaw, xGm, rowBase + offset, validElements);
            CopyInPad(residualRaw, residualGm, rowBase + offset, validElements);
            xQueue.EnQue(xRaw);
            residualQueue.EnQue(residualRaw);
            xRaw = xQueue.DeQue<T>();
            residualRaw = residualQueue.DeQue<T>();

            LocalTensor<float> xFp32 = xFp32Queue.AllocTensor<float>();
            LocalTensor<float> residualFp32 = residualFp32Queue.AllocTensor<float>();
            InputConverter<T>::Run(xFp32, xRaw, calcElements);
            InputConverter<T>::Run(residualFp32, residualRaw, calcElements);
            xQueue.FreeTensor(xRaw);
            residualQueue.FreeTensor(residualRaw);

            LocalTensor<float> y = yQueue.AllocTensor<float>();
            LocalTensor<float> square = squareQueue.AllocTensor<float>();
            Add(y, xFp32, residualFp32, calcElements);
            Mul(square, y, y, calcElements);
            LocalTensor<float> partial = sumQueue.AllocTensor<float>();
            ReduceSum(partial, square, reduceWork, static_cast<int32_t>(calcElements));
            PipeBarrier<PIPE_V>();
            squareSum += partial.GetValue(0);
            sumQueue.FreeTensor(partial);
            squareQueue.FreeTensor(square);
            yQueue.FreeTensor(y);
            xFp32Queue.FreeTensor(xFp32);
            residualFp32Queue.FreeTensor(residualFp32);
        }

        float meanSquare = squareSum * inverseDim_;
        LocalTensor<float> rmsInput = sumQueue.AllocTensor<float>();
        LocalTensor<float> rmsOutput = yQueue.AllocTensor<float>();
        rmsInput.SetValue(0, meanSquare + epsilon_);
        PipeBarrier<PIPE_ALL>();
        Sqrt(rmsOutput, rmsInput, 1);
        PipeBarrier<PIPE_V>();
        float inverseRms = 1.0f / rmsOutput.GetValue(0);
        sumQueue.FreeTensor(rmsInput);
        yQueue.FreeTensor(rmsOutput);

        // Pass 2: reread x/residual and apply the FP32 affine epilogue.
        for (uint32_t offset = 0; offset < dim_; offset += kTileElements) {
            uint32_t validElements = dim_ - offset;
            if (validElements > kTileElements) {
                validElements = kTileElements;
            }
            uint32_t calcElements = AlignedElements<T>(validElements);

            LocalTensor<T> xRaw = xQueue.AllocTensor<T>();
            LocalTensor<T> residualRaw = residualQueue.AllocTensor<T>();
            LocalTensor<T> gammaRaw = gammaQueue.AllocTensor<T>();
            LocalTensor<T> biasRaw = biasQueue.AllocTensor<T>();
            CopyInPad(xRaw, xGm, rowBase + offset, validElements);
            CopyInPad(residualRaw, residualGm, rowBase + offset, validElements);
            CopyInPad(gammaRaw, gammaGm, offset, validElements);
            CopyInPad(biasRaw, biasGm, offset, validElements);
            xQueue.EnQue(xRaw);
            residualQueue.EnQue(residualRaw);
            gammaQueue.EnQue(gammaRaw);
            biasQueue.EnQue(biasRaw);
            xRaw = xQueue.DeQue<T>();
            residualRaw = residualQueue.DeQue<T>();
            gammaRaw = gammaQueue.DeQue<T>();
            biasRaw = biasQueue.DeQue<T>();

            LocalTensor<float> xFp32 = xFp32Queue.AllocTensor<float>();
            LocalTensor<float> residualFp32 = residualFp32Queue.AllocTensor<float>();
            LocalTensor<float> gammaFp32 = gammaFp32Queue.AllocTensor<float>();
            LocalTensor<float> biasFp32 = biasFp32Queue.AllocTensor<float>();
            InputConverter<T>::Run(xFp32, xRaw, calcElements);
            InputConverter<T>::Run(residualFp32, residualRaw, calcElements);
            InputConverter<T>::Run(gammaFp32, gammaRaw, calcElements);
            InputConverter<T>::Run(biasFp32, biasRaw, calcElements);
            xQueue.FreeTensor(xRaw);
            residualQueue.FreeTensor(residualRaw);
            gammaQueue.FreeTensor(gammaRaw);
            biasQueue.FreeTensor(biasRaw);

            LocalTensor<float> y = yQueue.AllocTensor<float>();
            Add(y, xFp32, residualFp32, calcElements);
            Muls(y, y, inverseRms, static_cast<int32_t>(calcElements));
            Mul(y, y, gammaFp32, calcElements);
            Add(y, y, biasFp32, calcElements);

            LocalTensor<T> outputRaw = outputQueue.AllocTensor<T>();
            OutputConverter<T>::Run(outputRaw, y, calcElements);
            outputQueue.EnQue(outputRaw);
            outputRaw = outputQueue.DeQue<T>();
            CopyOut(outputGm, rowBase + offset, outputRaw, validElements);
            outputQueue.FreeTensor(outputRaw);

            yQueue.FreeTensor(y);
            xFp32Queue.FreeTensor(xFp32);
            residualFp32Queue.FreeTensor(residualFp32);
            gammaFp32Queue.FreeTensor(gammaFp32);
            biasFp32Queue.FreeTensor(biasFp32);
        }
    }

    GlobalTensor<T> xGm;
    GlobalTensor<T> residualGm;
    GlobalTensor<T> gammaGm;
    GlobalTensor<T> biasGm;
    GlobalTensor<T> outputGm;
    uint64_t outer_ = 0;
    uint32_t dim_ = 0;
    float inverseDim_ = 0.0f;
    float epsilon_ = 0.0f;

    TPipe tpipe;
    TQue<TPosition::VECIN, 1> xQueue;
    TQue<TPosition::VECIN, 1> residualQueue;
    TQue<TPosition::VECIN, 1> gammaQueue;
    TQue<TPosition::VECIN, 1> biasQueue;
    TQue<TPosition::VECOUT, 1> outputQueue;
    TQue<TPosition::VECCALC, 1> xFp32Queue;
    TQue<TPosition::VECCALC, 1> residualFp32Queue;
    TQue<TPosition::VECCALC, 1> gammaFp32Queue;
    TQue<TPosition::VECCALC, 1> biasFp32Queue;
    TQue<TPosition::VECCALC, 1> yQueue;
    TQue<TPosition::VECCALC, 1> squareQueue;
    TQue<TPosition::VECCALC, 1> sumQueue;
    TBuf<TPosition::VECCALC> workBuffer;
};

template <typename T>
__global__ __vector__ void AddRmsNormBiasKernel(GM_ADDR x, GM_ADDR residual,
    GM_ADDR gamma, GM_ADDR bias, GM_ADDR output, uint64_t outer, uint32_t dim,
    float inverseDim, float epsilon)
{
    RmsNormKernel<T> kernel;
    kernel.Init(x, residual, gamma, bias, output, outer, dim, inverseDim, epsilon);
    kernel.Process();
}

} // namespace MixAReduceFp32TwoPass

extern "C" void run_kernel(GM_ADDR x, const TensorGroupInfo& info_x, GM_ADDR residual,
    const TensorGroupInfo& info_residual, GM_ADDR gamma, const TensorGroupInfo& info_gamma,
    GM_ADDR bias, const TensorGroupInfo& info_bias, GM_ADDR output,
    const TensorGroupInfo& info_output, int64_t availableCoreNum, aclrtStream stream,
    float epsilon)
{
    (void)info_residual;
    (void)info_gamma;
    (void)info_bias;
    (void)info_output;
    const TensorInfo& xInfo = info_x.tensors[0];
    uint64_t outer = 1;
    for (int64_t i = 0; i + 1 < xInfo.numDims; ++i) {
        outer *= static_cast<uint64_t>(xInfo.shape[i]);
    }
    uint32_t dim = static_cast<uint32_t>(xInfo.shape[xInfo.numDims - 1]);
    float inverseDim = 1.0f / static_cast<float>(dim);
    uint32_t blockNum = static_cast<uint32_t>(availableCoreNum);
    if (blockNum > outer) {
        blockNum = static_cast<uint32_t>(outer);
    }
    if (blockNum == 0 || dim == 0) {
        return;
    }

    if (xInfo.dtype == 0) {
        MixAReduceFp32TwoPass::AddRmsNormBiasKernel<float>
            <<<blockNum, nullptr, stream>>>(x, residual, gamma, bias, output, outer, dim, inverseDim, epsilon);
    } else if (xInfo.dtype == 1) {
        MixAReduceFp32TwoPass::AddRmsNormBiasKernel<half>
            <<<blockNum, nullptr, stream>>>(x, residual, gamma, bias, output, outer, dim, inverseDim, epsilon);
    } else if (xInfo.dtype == 2) {
        MixAReduceFp32TwoPass::AddRmsNormBiasKernel<bfloat16_t>
            <<<blockNum, nullptr, stream>>>(x, residual, gamma, bias, output, outer, dim, inverseDim, epsilon);
    }
}
