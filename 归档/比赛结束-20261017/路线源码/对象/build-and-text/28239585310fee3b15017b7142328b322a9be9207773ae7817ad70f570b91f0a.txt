#include <math.h>
#include <stdint.h>

#include "kernel_operator.h"

namespace {

constexpr uint32_t kTileElements = 256;
constexpr uint32_t kScalarElements = 8;

struct RmsNormBiasTilingData {
    uint32_t blockNum;
    uint64_t outer;
    uint64_t dim;
    float inverseDim;
    float epsilon;
};

template <typename T>
struct RmsNormBiasOps {
    __aicore__ static inline void ToFloat(const AscendC::LocalTensor<float>& dst,
                                          const AscendC::LocalTensor<T>& src,
                                          uint32_t count)
    {
        AscendC::Cast<float, T>(dst, src, AscendC::RoundMode::CAST_NONE, count);
    }

    __aicore__ static inline void Finalize(const AscendC::LocalTensor<T>& dst,
                                           const AscendC::LocalTensor<float>& value,
                                           const AscendC::LocalTensor<float>& bias,
                                           const AscendC::LocalTensor<float>& output,
                                           uint32_t count)
    {
        AscendC::Add<float>(output, value, bias, count);
        AscendC::Cast<T, float>(dst, output, AscendC::RoundMode::CAST_ROUND, count);
    }
};

template <>
struct RmsNormBiasOps<float> {
    __aicore__ static inline void ToFloat(const AscendC::LocalTensor<float>& dst,
                                          const AscendC::LocalTensor<float>& src,
                                          uint32_t count)
    {
        AscendC::DataCopy(dst, src, count);
        AscendC::PipeBarrier<PIPE_ALL>();
    }

    __aicore__ static inline void Finalize(const AscendC::LocalTensor<float>& dst,
                                           const AscendC::LocalTensor<float>& value,
                                           const AscendC::LocalTensor<float>& bias,
                                           const AscendC::LocalTensor<float>& output,
                                           uint32_t count)
    {
        AscendC::Add<float>(dst, value, bias, count);
    }
};

template <typename T>
class KernelRmsNormBias {
public:
    __aicore__ inline KernelRmsNormBias(AscendC::TPipe* pipe) : pipe_(pipe) {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias,
                                GM_ADDR output, const __gm__ RmsNormBiasTilingData* tiling)
    {
        tiling_ = tiling;
        xGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(x), tiling_->outer * tiling_->dim);
        residualGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(residual), tiling_->outer * tiling_->dim);
        gammaGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(gamma), tiling_->dim);
        biasGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(bias), tiling_->dim);
        outputGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(output), tiling_->outer * tiling_->dim);

        pipe_->InitBuffer(xQueue_, 1, kTileElements * sizeof(T));
        pipe_->InitBuffer(residualQueue_, 1, kTileElements * sizeof(T));
        pipe_->InitBuffer(gammaQueue_, 1, kTileElements * sizeof(T));
        pipe_->InitBuffer(biasQueue_, 1, kTileElements * sizeof(T));
        pipe_->InitBuffer(outputQueue_, 1, kTileElements * sizeof(T));

        pipe_->InitBuffer(xFloatBuf_, kTileElements * sizeof(float));
        pipe_->InitBuffer(residualFloatBuf_, kTileElements * sizeof(float));
        pipe_->InitBuffer(gammaFloatBuf_, kTileElements * sizeof(float));
        pipe_->InitBuffer(biasFloatBuf_, kTileElements * sizeof(float));
        pipe_->InitBuffer(outputFloatBuf_, kTileElements * sizeof(float));
        pipe_->InitBuffer(reduceTmpBuf_, 32 * 1024);
        pipe_->InitBuffer(scalarBuf_, 32);
    }

    __aicore__ inline void Process()
    {
        const uint32_t blockIdx = AscendC::GetBlockIdx();
        if (blockIdx >= tiling_->blockNum) {
            return;
        }

        // Cyclic row scheduling keeps the tail distribution independent of outer % blockNum.
        for (uint64_t row = blockIdx; row < tiling_->outer; row += tiling_->blockNum) {
            const float sumSquares = AccumulateRowSquares(row);
            float meanSquare = sumSquares * tiling_->inverseDim + tiling_->epsilon;
            AscendC::LocalTensor<float> scalar = scalarBuf_.Get<float>();
            AscendC::Duplicate<float>(scalar, meanSquare, kScalarElements);
            AscendC::Sqrt<float>(scalar, scalar, static_cast<int32_t>(kScalarElements));
            AscendC::PipeBarrier<PIPE_ALL>();
            float inverseRms = 1.0f / scalar.GetValue(0);
            NormalizeRow(row, inverseRms);
        }
    }

private:
    __aicore__ inline uint32_t GetCount(uint64_t col) const
    {
        const uint64_t remaining = tiling_->dim - col;
        return remaining < kTileElements ? static_cast<uint32_t>(remaining) : kTileElements;
    }

    __aicore__ inline void CopyOne(const AscendC::LocalTensor<T>& dst,
                                   const AscendC::GlobalTensor<T>& src,
                                   uint64_t offset, uint32_t count)
    {
        const AscendC::DataCopyParams copyParams{
            1, static_cast<uint16_t>(count * sizeof(T)), 0, 0};
        const AscendC::DataCopyPadParams padParams{
            true, 0, static_cast<uint8_t>(kTileElements - count), 0};
        AscendC::DataCopyPad(dst, src[offset], copyParams, padParams);
    }

    __aicore__ inline void CopyInPrimary(uint64_t row, uint64_t col, uint32_t count)
    {
        const uint64_t rowOffset = row * tiling_->dim + col;
        AscendC::LocalTensor<T> xLocal = xQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> residualLocal = residualQueue_.AllocTensor<T>();
        CopyOne(xLocal, xGm_, rowOffset, count);
        CopyOne(residualLocal, residualGm_, rowOffset, count);
        xQueue_.EnQue(xLocal);
        residualQueue_.EnQue(residualLocal);
    }

    __aicore__ inline void CopyInFull(uint64_t row, uint64_t col, uint32_t count)
    {
        const uint64_t rowOffset = row * tiling_->dim + col;
        AscendC::LocalTensor<T> xLocal = xQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> residualLocal = residualQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> gammaLocal = gammaQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> biasLocal = biasQueue_.AllocTensor<T>();

        CopyOne(xLocal, xGm_, rowOffset, count);
        CopyOne(residualLocal, residualGm_, rowOffset, count);
        CopyOne(gammaLocal, gammaGm_, col, count);
        CopyOne(biasLocal, biasGm_, col, count);

        xQueue_.EnQue(xLocal);
        residualQueue_.EnQue(residualLocal);
        gammaQueue_.EnQue(gammaLocal);
        biasQueue_.EnQue(biasLocal);
    }

    __aicore__ inline float ComputeTileSum()
    {
        AscendC::LocalTensor<T> xLocal = xQueue_.DeQue<T>();
        AscendC::LocalTensor<T> residualLocal = residualQueue_.DeQue<T>();
        AscendC::LocalTensor<float> xFloat = xFloatBuf_.Get<float>();
        AscendC::LocalTensor<float> residualFloat = residualFloatBuf_.Get<float>();
        AscendC::LocalTensor<float> scalar = scalarBuf_.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceTmpBuf_.Get<float>();

        RmsNormBiasOps<T>::ToFloat(xFloat, xLocal, kTileElements);
        RmsNormBiasOps<T>::ToFloat(residualFloat, residualLocal, kTileElements);
        AscendC::Add<float>(xFloat, xFloat, residualFloat, kTileElements);
        AscendC::Mul<float>(residualFloat, xFloat, xFloat, kTileElements);
        AscendC::ReduceSum<float, true>(scalar, residualFloat, reduceTmp,
                                        static_cast<int32_t>(kTileElements));

        const float sum = scalar.GetValue(0);
        xQueue_.FreeTensor(xLocal);
        residualQueue_.FreeTensor(residualLocal);
        return sum;
    }

    __aicore__ inline float AccumulateRowSquares(uint64_t row)
    {
        float sumSquares = 0.0f;
        for (uint64_t col = 0; col < tiling_->dim; col += kTileElements) {
            const uint32_t count = GetCount(col);
            CopyInPrimary(row, col, count);
            sumSquares += ComputeTileSum();
        }
        return sumSquares;
    }

    __aicore__ inline void ComputeTileOutput(uint32_t count, float inverseRms)
    {
        AscendC::LocalTensor<T> xLocal = xQueue_.DeQue<T>();
        AscendC::LocalTensor<T> residualLocal = residualQueue_.DeQue<T>();
        AscendC::LocalTensor<T> gammaLocal = gammaQueue_.DeQue<T>();
        AscendC::LocalTensor<T> biasLocal = biasQueue_.DeQue<T>();
        AscendC::LocalTensor<T> outputLocal = outputQueue_.AllocTensor<T>();

        AscendC::LocalTensor<float> xFloat = xFloatBuf_.Get<float>();
        AscendC::LocalTensor<float> residualFloat = residualFloatBuf_.Get<float>();
        AscendC::LocalTensor<float> gammaFloat = gammaFloatBuf_.Get<float>();
        AscendC::LocalTensor<float> biasFloat = biasFloatBuf_.Get<float>();
        AscendC::LocalTensor<float> outputFloat = outputFloatBuf_.Get<float>();

        RmsNormBiasOps<T>::ToFloat(xFloat, xLocal, kTileElements);
        RmsNormBiasOps<T>::ToFloat(residualFloat, residualLocal, kTileElements);
        RmsNormBiasOps<T>::ToFloat(gammaFloat, gammaLocal, kTileElements);
        RmsNormBiasOps<T>::ToFloat(biasFloat, biasLocal, kTileElements);

        AscendC::Add<float>(xFloat, xFloat, residualFloat, kTileElements);
        AscendC::Muls<float>(xFloat, xFloat, inverseRms, kTileElements);
        AscendC::Mul<float>(xFloat, xFloat, gammaFloat, kTileElements);
        RmsNormBiasOps<T>::Finalize(outputLocal, xFloat, biasFloat, outputFloat, kTileElements);
        outputQueue_.EnQue(outputLocal);

        xQueue_.FreeTensor(xLocal);
        residualQueue_.FreeTensor(residualLocal);
        gammaQueue_.FreeTensor(gammaLocal);
        biasQueue_.FreeTensor(biasLocal);

        AscendC::LocalTensor<T> output = outputQueue_.DeQue<T>();
        const AscendC::DataCopyParams copyParams{
            1, static_cast<uint16_t>(count * sizeof(T)), 0, 0};
        AscendC::DataCopyPad(outputGm_, output, copyParams);
        outputQueue_.FreeTensor(output);
    }

    __aicore__ inline void NormalizeRow(uint64_t row, float inverseRms)
    {
        for (uint64_t col = 0; col < tiling_->dim; col += kTileElements) {
            const uint32_t count = GetCount(col);
            CopyInFull(row, col, count);
            ComputeTileOutput(count, inverseRms);
        }
    }

private:
    AscendC::TPipe* pipe_;
    const __gm__ RmsNormBiasTilingData* tiling_;

    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;

    AscendC::TQue<AscendC::TPosition::VECIN, 1> xQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> residualQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gammaQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue_;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outputQueue_;

    AscendC::TBuf<AscendC::TPosition::VECCALC> xFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> residualFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outputFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceTmpBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf_;
};

template <typename T>
__aicore__ inline void LaunchKernel(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias,
                                    GM_ADDR output, GM_ADDR tiling)
{
    AscendC::TPipe pipe;
    KernelRmsNormBias<T> op(&pipe);
    op.Init(x, residual, gamma, bias, output,
            reinterpret_cast<__gm__ RmsNormBiasTilingData*>(tiling));
    op.Process();
}

}  // namespace

extern "C" __global__ __vector__ void add_rms_norm_bias_fp32(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR output, GM_ADDR tiling)
{
    LaunchKernel<float>(x, residual, gamma, bias, output, tiling);
}

extern "C" __global__ __vector__ void add_rms_norm_bias_fp16(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR output, GM_ADDR tiling)
{
    LaunchKernel<half>(x, residual, gamma, bias, output, tiling);
}

extern "C" __global__ __vector__ void add_rms_norm_bias_bf16(
    GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias, GM_ADDR output, GM_ADDR tiling)
{
    LaunchKernel<bfloat16_t>(x, residual, gamma, bias, output, tiling);
}

extern "C" void run_kernel(GM_ADDR x, const TensorGroupInfo& info_x, GM_ADDR residual,
                            const TensorGroupInfo& info_residual, GM_ADDR gamma,
                            const TensorGroupInfo& info_gamma, GM_ADDR bias,
                            const TensorGroupInfo& info_bias, GM_ADDR output,
                            const TensorGroupInfo& info_output, int64_t availableCoreNum,
                            aclrtStream stream, float epsilon)
{
    (void)info_residual;
    (void)info_gamma;
    (void)info_bias;
    (void)info_output;

    if (info_x.numTensors <= 0 || info_x.tensors == nullptr || availableCoreNum <= 0) {
        return;
    }

    const TensorInfo& xInfo = info_x.tensors[0];
    if (xInfo.shape == nullptr || xInfo.numDims <= 0) {
        return;
    }

    uint64_t outer = 1;
    for (int64_t i = 0; i + 1 < xInfo.numDims; ++i) {
        outer *= static_cast<uint64_t>(xInfo.shape[i]);
    }
    const uint64_t dim = static_cast<uint64_t>(xInfo.shape[xInfo.numDims - 1]);
    if (outer == 0 || dim == 0) {
        return;
    }

    RmsNormBiasTilingData tiling{};
    tiling.outer = outer;
    tiling.dim = dim;
    tiling.inverseDim = 1.0f / static_cast<float>(dim);
    tiling.epsilon = epsilon;
    tiling.blockNum = outer < static_cast<uint64_t>(availableCoreNum)
        ? static_cast<uint32_t>(outer)
        : static_cast<uint32_t>(availableCoreNum);

    uint8_t* tilingDevice = nullptr;
    if (aclrtMalloc(reinterpret_cast<void**>(&tilingDevice), sizeof(tiling),
                    ACL_MEM_MALLOC_HUGE_FIRST) != ACL_SUCCESS) {
        return;
    }
    if (aclrtMemcpy(tilingDevice, sizeof(tiling), &tiling, sizeof(tiling),
                    ACL_MEMCPY_HOST_TO_DEVICE) != ACL_SUCCESS) {
        aclrtFree(tilingDevice);
        return;
    }

    switch (xInfo.dtype) {
        case 0:
            add_rms_norm_bias_fp32<<<tiling.blockNum, nullptr, stream>>>(
                x, residual, gamma, bias, output, tilingDevice);
            break;
        case 1:
            add_rms_norm_bias_fp16<<<tiling.blockNum, nullptr, stream>>>(
                x, residual, gamma, bias, output, tilingDevice);
            break;
        case 2:
            add_rms_norm_bias_bf16<<<tiling.blockNum, nullptr, stream>>>(
                x, residual, gamma, bias, output, tilingDevice);
            break;
        default:
            aclrtFree(tilingDevice);
            return;
    }

    aclrtSynchronizeStream(stream);
    aclrtFree(tilingDevice);
}
