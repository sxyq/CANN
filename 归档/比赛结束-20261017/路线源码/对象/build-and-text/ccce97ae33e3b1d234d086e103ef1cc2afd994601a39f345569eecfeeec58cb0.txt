#include <cmath>
#include <cstdint>
#include "kernel_operator.h"

/*
 * AddRmsNormBias
 *
 * R013-V001
 * Route: MTE2 / Vector input double buffering
 *
 * Parent:
 *   D baseline
 *
 * ONLY optimization:
 *
 *   For rows wider than one 8192-element tile:
 *
 *       Tile N:
 *           Vector compute
 *
 *       Tile N+1:
 *           x/residual GM -> UB prefetch
 *
 *   using two ping-pong input buffer pairs.
 *
 * Activation:
 *
 *       rowWidth > 8192
 *
 * Everything else remains D baseline:
 *
 *   - tile = 8192
 *   - two-pass algorithm
 *   - FP32 intermediate arithmetic
 *   - ReduceSum
 *   - residual scratch reuse
 *   - RMS scalar synchronization
 *   - per-tile gamma/bias load
 *   - Div + Sqrt
 *   - per-tile output store
 *   - row/core scheduling
 *
 * IMPORTANT:
 *
 * FP32:
 *   Pair A = xBuf_ / residualBuf_
 *   Pair B = xFp32Buf_ / residualFp32Buf_
 *
 *   No additional large buffers.
 *
 * FP16/BF16:
 *   Pair A = xBuf_ / residualBuf_
 *   Pair B = xPingBuf_ / residualPingBuf_
 *
 *   xFp32Buf_/residualFp32Buf_ remain conversion/work buffers.
 *
 * Estimated main UB:
 *
 *   FP32:
 *       5 * 8192 * 4 ~= 160 KiB
 *
 *   FP16/BF16:
 *       4 native input tiles ~= 64 KiB
 *       3 FP32 tiles         ~= 96 KiB
 *       total               ~= 160 KiB
 */

template <typename T>
class AddRmsNormBiasKernel {
public:
    static constexpr int32_t kTileElems = 8192;
    static constexpr int32_t kScalarElems = 16;

    __aicore__ inline void Init(
        GM_ADDR x,
        GM_ADDR residual,
        GM_ADDR gamma,
        GM_ADDR bias,
        GM_ADDR output)
    {
        xGm_.SetGlobalBuffer(
            (__gm__ T *)x);

        residualGm_.SetGlobalBuffer(
            (__gm__ T *)residual);

        gammaGm_.SetGlobalBuffer(
            (__gm__ T *)gamma);

        biasGm_.SetGlobalBuffer(
            (__gm__ T *)bias);

        outputGm_.SetGlobalBuffer(
            (__gm__ T *)output);

        /*
         * Pair A.
         */
        tpipe.InitBuffer(
            xBuf_,
            kTileElems * sizeof(T));

        tpipe.InitBuffer(
            residualBuf_,
            kTileElems * sizeof(T));

        /*
         * FP32 work buffers.
         *
         * For T=float these ALSO form ping-pong pair B.
         */
        tpipe.InitBuffer(
            xFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            residualFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            yFp32Buf_,
            kTileElems * sizeof(float));

        /*
         * Only low precision requires a second native input pair.
         */
        if constexpr (
            !AscendC::IsSameType<
                T,
                float>::value) {

            tpipe.InitBuffer(
                xPingBuf_,
                kTileElems * sizeof(T));

            tpipe.InitBuffer(
                residualPingBuf_,
                kTileElems * sizeof(T));
        }

        tpipe.InitBuffer(
            sumBuf_,
            kScalarElems * sizeof(float));
    }

    __aicore__ inline void Process(
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowGroup,
        uint64_t groupCount,
        uint32_t blockCount,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t blockIdx =
            static_cast<uint64_t>(
                AscendC::GetBlockIdx());

        const uint64_t baseGroups =
            groupCount /
            static_cast<uint64_t>(
                blockCount);

        const uint64_t extraGroups =
            groupCount %
            static_cast<uint64_t>(
                blockCount);

        const uint64_t beginGroup =
            blockIdx * baseGroups +
            (blockIdx < extraGroups
                 ? blockIdx
                 : extraGroups);

        const uint64_t localGroups =
            baseGroups +
            (blockIdx < extraGroups
                 ? 1ULL
                 : 0ULL);

        const uint64_t endGroup =
            beginGroup + localGroups;

        const uint64_t beginRow =
            beginGroup * rowGroup;

        uint64_t endRow =
            endGroup * rowGroup;

        if (endRow > rowCount) {
            endRow = rowCount;
        }

        /*
         * R013 only has meaning if one row contains >= 2 tiles.
         */
        const bool useDoubleBuffer =
            rowWidth >
            static_cast<uint64_t>(
                kTileElems);

        for (uint64_t row = beginRow;
             row < endRow;
             ++row) {

            if (useDoubleBuffer) {

                ProcessDoubleBufferedRow(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);

            } else {

                /*
                 * Exact D-baseline path.
                 */
                ProcessRow(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);
            }
        }
    }

private:

    /*
     * =============================================================
     * Basic helpers
     * =============================================================
     */

    __aicore__ inline int32_t TileLength(
        uint64_t remain) const
    {
        return static_cast<int32_t>(
            remain >
                    static_cast<uint64_t>(
                        kTileElems)
                ? kTileElems
                : remain);
    }

    __aicore__ inline void Load(
        const AscendC::LocalTensor<T>& dst,
        const AscendC::GlobalTensor<T>& src,
        uint64_t offset,
        int32_t count)
    {
        const AscendC::DataCopyExtParams copyParams(
            1,
            static_cast<uint32_t>(
                count *
                static_cast<int32_t>(
                    sizeof(T))),
            0,
            0,
            0);

        const AscendC::DataCopyPadExtParams<T>
            padParams;

        AscendC::DataCopyPad(
            dst,
            src[offset],
            copyParams,
            padParams);
    }

    __aicore__ inline void Store(
        const AscendC::GlobalTensor<T>& dst,
        uint64_t offset,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        const AscendC::DataCopyExtParams copyParams(
            1,
            static_cast<uint32_t>(
                count *
                static_cast<int32_t>(
                    sizeof(T))),
            0,
            0,
            0);

        AscendC::DataCopyPad(
            dst[offset],
            src,
            copyParams);
    }

    __aicore__ inline void ToFloat(
        const AscendC::LocalTensor<float>& dst,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::Adds(
                dst,
                src,
                0.0f,
                count);

        } else {

            AscendC::Cast(
                dst,
                src,
                AscendC::RoundMode::CAST_NONE,
                count);
        }
    }

    /*
     * =============================================================
     * D-baseline synchronization helpers
     *
     * R013 double-buffer path does NOT use MTE2/V helpers below.
     * It uses allocated ping-pong event IDs instead.
     * =============================================================
     */

    __aicore__ inline void SyncMTE2ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE2_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE2_V>(0);
    }

    __aicore__ inline void SyncVToMTE2()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE2>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE2>(0);
    }

    __aicore__ inline void SyncVToMTE3()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE3>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE3>(0);
    }

    __aicore__ inline void SyncMTE3ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE3_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE3_V>(0);
    }

    __aicore__ inline void SyncVToS()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_S>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_S>(0);
    }

    __aicore__ inline void SyncSToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::S_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::S_V>(0);
    }

    /*
     * =============================================================
     * Ping-pong input pair helpers
     * =============================================================
     */

    __aicore__ inline AscendC::LocalTensor<T>
    GetXPair(bool second)
    {
        if (!second) {
            return xBuf_.Get<T>();
        }

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            /*
             * For FP32, xFp32Buf_ itself is Pair B.
             */
            return xFp32Buf_.Get<T>();

        } else {

            return xPingBuf_.Get<T>();
        }
    }

    __aicore__ inline AscendC::LocalTensor<T>
    GetResidualPair(bool second)
    {
        if (!second) {
            return residualBuf_.Get<T>();
        }

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            return residualFp32Buf_.Get<T>();

        } else {

            return residualPingBuf_.Get<T>();
        }
    }

    __aicore__ inline void LoadInputPair(
        bool second,
        uint64_t offset,
        int32_t valid)
    {
        AscendC::LocalTensor<T> xInput =
            GetXPair(second);

        AscendC::LocalTensor<T> residualInput =
            GetResidualPair(second);

        Load(
            xInput,
            xGm_,
            offset,
            valid);

        Load(
            residualInput,
            residualGm_,
            offset,
            valid);
    }

    /*
     * =============================================================
     * R013 FIRST PASS
     *
     * MTE2(N+1) overlaps Vector(N).
     * =============================================================
     */

    __aicore__ inline void
    FirstPassDoubleBuffered(
        uint64_t rowOffset,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon,
        float& rmsValue)
    {
        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * Two MTE2 -> V events:
         * one for each input pair.
         */
        const AscendC::TEventID ready0 =
            tpipe.AllocEventID<
                AscendC::HardEvent::MTE2_V>();

        const AscendC::TEventID ready1 =
            tpipe.AllocEventID<
                AscendC::HardEvent::MTE2_V>();

        /*
         * Two V -> MTE2 release events.
         */
        const AscendC::TEventID release0 =
            tpipe.AllocEventID<
                AscendC::HardEvent::V_MTE2>();

        const AscendC::TEventID release1 =
            tpipe.AllocEventID<
                AscendC::HardEvent::V_MTE2>();

        bool releasePending0 = false;
        bool releasePending1 = false;

        const uint64_t tileCount =
            (rowWidth +
             static_cast<uint64_t>(
                 kTileElems) -
             1ULL) /
            static_cast<uint64_t>(
                kTileElems);

        /*
         * Prime Pair A with tile 0.
         */
        {
            const int32_t valid0 =
                TileLength(
                    rowWidth);

            LoadInputPair(
                false,
                rowOffset,
                valid0);

            AscendC::SetFlag<
                AscendC::HardEvent::MTE2_V>(
                    ready0);
        }

        for (uint64_t tile = 0;
             tile < tileCount;
             ++tile) {

            const bool second =
                (tile & 1ULL) != 0ULL;

            const AscendC::TEventID readyCur =
                second
                    ? ready1
                    : ready0;

            const AscendC::TEventID releaseCur =
                second
                    ? release1
                    : release0;

            /*
             * Current tile must be ready before Vector consumes it.
             */
            AscendC::WaitFlag<
                AscendC::HardEvent::MTE2_V>(
                    readyCur);

            /*
             * =====================================================
             * Prefetch Tile N+1 BEFORE Vector computes Tile N.
             *
             * This is the actual R013 overlap.
             * =====================================================
             */
            if (tile + 1ULL < tileCount) {

                const bool nextSecond =
                    !second;

                const AscendC::TEventID
                    readyNext =
                        nextSecond
                            ? ready1
                            : ready0;

                const AscendC::TEventID
                    releaseNext =
                        nextSecond
                            ? release1
                            : release0;

                bool& releasePendingNext =
                    nextSecond
                        ? releasePending1
                        : releasePending0;

                /*
                 * Starting from tile 2, the destination pair may
                 * still contain data from its previous Vector use.
                 */
                if (releasePendingNext) {

                    AscendC::WaitFlag<
                        AscendC::HardEvent::V_MTE2>(
                            releaseNext);

                    releasePendingNext = false;
                }

                const uint64_t nextCol =
                    (tile + 1ULL) *
                    static_cast<uint64_t>(
                        kTileElems);

                const int32_t nextValid =
                    TileLength(
                        rowWidth - nextCol);

                LoadInputPair(
                    nextSecond,
                    rowOffset + nextCol,
                    nextValid);

                /*
                 * Do NOT wait here.
                 *
                 * MTE2 is now allowed to run while Vector handles
                 * the current tile.
                 */
                AscendC::SetFlag<
                    AscendC::HardEvent::MTE2_V>(
                        readyNext);
            }

            const uint64_t col =
                tile *
                static_cast<uint64_t>(
                    kTileElems);

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            AscendC::LocalTensor<T> xInput =
                GetXPair(second);

            AscendC::LocalTensor<T> residualInput =
                GetResidualPair(second);

            /*
             * =====================================================
             * Current tile Vector work.
             * =====================================================
             */

            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                /*
                 * For FP32 the input pair itself is already FP32.
                 *
                 * y = x + residual
                 */
                AscendC::Add(
                    yFp32,
                    xInput,
                    residualInput,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                /*
                 * xInput is dead after Add.
                 * Reuse it as y^2.
                 */
                AscendC::Mul(
                    xInput,
                    yFp32,
                    yFp32,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                /*
                 * residualInput becomes ReduceSum workspace.
                 */
                AscendC::ReduceSum(
                    partial,
                    xInput,
                    residualInput,
                    valid);

            } else {

                AscendC::LocalTensor<float> xFp32 =
                    xFp32Buf_.Get<float>();

                AscendC::LocalTensor<float> residualFp32 =
                    residualFp32Buf_.Get<float>();

                ToFloat(
                    xFp32,
                    xInput,
                    valid);

                ToFloat(
                    residualFp32,
                    residualInput,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Add(
                    yFp32,
                    xFp32,
                    residualFp32,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Mul(
                    xFp32,
                    yFp32,
                    yFp32,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::ReduceSum(
                    partial,
                    xFp32,
                    residualFp32,
                    valid);
            }

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                rms,
                rms,
                partial,
                1);

            AscendC::PipeBarrier<PIPE_V>();

            /*
             * Signal that this pair can eventually be overwritten
             * by a future MTE2 load.
             */
            AscendC::SetFlag<
                AscendC::HardEvent::V_MTE2>(
                    releaseCur);

            if (second) {
                releasePending1 = true;
            } else {
                releasePending0 = true;
            }
        }

        /*
         * Finish scalar RMS arithmetic.
         */
        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToS();

        rmsValue =
            rms.GetValue(0);

        SyncSToV();

        /*
         * Drain outstanding pair-release events before releasing
         * event IDs.
         */
        if (releasePending0) {

            AscendC::WaitFlag<
                AscendC::HardEvent::V_MTE2>(
                    release0);

            releasePending0 = false;
        }

        if (releasePending1) {

            AscendC::WaitFlag<
                AscendC::HardEvent::V_MTE2>(
                    release1);

            releasePending1 = false;
        }

        tpipe.ReleaseEventID<
            AscendC::HardEvent::MTE2_V>(
                ready0);

        tpipe.ReleaseEventID<
            AscendC::HardEvent::MTE2_V>(
                ready1);

        tpipe.ReleaseEventID<
            AscendC::HardEvent::V_MTE2>(
                release0);

        tpipe.ReleaseEventID<
            AscendC::HardEvent::V_MTE2>(
                release1);
    }

    /*
     * =============================================================
     * R013 SECOND PASS
     *
     * x/residual Tile N+1 is prefetched while Vector handles
     * current Tile N.
     *
     * Gamma/Bias loading itself remains D-baseline behavior.
     * =============================================================
     */

    __aicore__ inline void
    SecondPassDoubleBuffered(
        uint64_t rowOffset,
        uint64_t rowWidth,
        float rmsValue)
    {
        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        const AscendC::TEventID ready0 =
            tpipe.AllocEventID<
                AscendC::HardEvent::MTE2_V>();

        const AscendC::TEventID ready1 =
            tpipe.AllocEventID<
                AscendC::HardEvent::MTE2_V>();

        const AscendC::TEventID release0 =
            tpipe.AllocEventID<
                AscendC::HardEvent::V_MTE2>();

        const AscendC::TEventID release1 =
            tpipe.AllocEventID<
                AscendC::HardEvent::V_MTE2>();

        bool releasePending0 = false;
        bool releasePending1 = false;

        const uint64_t tileCount =
            (rowWidth +
             static_cast<uint64_t>(
                 kTileElems) -
             1ULL) /
            static_cast<uint64_t>(
                kTileElems);

        /*
         * Prime tile 0.
         */
        {
            const int32_t valid0 =
                TileLength(
                    rowWidth);

            LoadInputPair(
                false,
                rowOffset,
                valid0);

            AscendC::SetFlag<
                AscendC::HardEvent::MTE2_V>(
                    ready0);
        }

        for (uint64_t tile = 0;
             tile < tileCount;
             ++tile) {

            const bool second =
                (tile & 1ULL) != 0ULL;

            const AscendC::TEventID readyCur =
                second
                    ? ready1
                    : ready0;

            const AscendC::TEventID releaseCur =
                second
                    ? release1
                    : release0;

            AscendC::WaitFlag<
                AscendC::HardEvent::MTE2_V>(
                    readyCur);

            /*
             * =====================================================
             * Prefetch next x/residual pair.
             * =====================================================
             */
            if (tile + 1ULL < tileCount) {

                const bool nextSecond =
                    !second;

                const AscendC::TEventID
                    readyNext =
                        nextSecond
                            ? ready1
                            : ready0;

                const AscendC::TEventID
                    releaseNext =
                        nextSecond
                            ? release1
                            : release0;

                bool& releasePendingNext =
                    nextSecond
                        ? releasePending1
                        : releasePending0;

                if (releasePendingNext) {

                    AscendC::WaitFlag<
                        AscendC::HardEvent::V_MTE2>(
                            releaseNext);

                    releasePendingNext = false;
                }

                const uint64_t nextCol =
                    (tile + 1ULL) *
                    static_cast<uint64_t>(
                        kTileElems);

                const int32_t nextValid =
                    TileLength(
                        rowWidth - nextCol);

                LoadInputPair(
                    nextSecond,
                    rowOffset + nextCol,
                    nextValid);

                AscendC::SetFlag<
                    AscendC::HardEvent::MTE2_V>(
                        readyNext);
            }

            const uint64_t col =
                tile *
                static_cast<uint64_t>(
                    kTileElems);

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            AscendC::LocalTensor<T> xInput =
                GetXPair(second);

            AscendC::LocalTensor<T> residualInput =
                GetResidualPair(second);

            /*
             * =====================================================
             * Current x/residual computation.
             * =====================================================
             */
            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                AscendC::Add(
                    yFp32,
                    xInput,
                    residualInput,
                    valid);

            } else {

                AscendC::LocalTensor<float> xFp32 =
                    xFp32Buf_.Get<float>();

                AscendC::LocalTensor<float> residualFp32 =
                    residualFp32Buf_.Get<float>();

                ToFloat(
                    xFp32,
                    xInput,
                    valid);

                ToFloat(
                    residualFp32,
                    residualInput,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Add(
                    yFp32,
                    xFp32,
                    residualFp32,
                    valid);
            }

            AscendC::PipeBarrier<PIPE_V>();

            /*
             * Normalization denominator workspace.
             */
            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                /*
                 * residualInput can now be reused.
                 */
                AscendC::Duplicate(
                    residualInput,
                    rmsValue,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Div(
                    yFp32,
                    yFp32,
                    residualInput,
                    valid);

            } else {

                AscendC::LocalTensor<float> residualFp32 =
                    residualFp32Buf_.Get<float>();

                AscendC::Duplicate(
                    residualFp32,
                    rmsValue,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Div(
                    yFp32,
                    yFp32,
                    residualFp32,
                    valid);
            }

            AscendC::PipeBarrier<PIPE_V>();

            /*
             * =====================================================
             * Current input pair can now be overwritten.
             *
             * Consume this event immediately because gamma/bias
             * will reuse the same physical pair.
             * =====================================================
             */
            AscendC::SetFlag<
                AscendC::HardEvent::V_MTE2>(
                    releaseCur);

            AscendC::WaitFlag<
                AscendC::HardEvent::V_MTE2>(
                    releaseCur);

            /*
             * =====================================================
             * Gamma/Bias behavior remains D baseline.
             *
             * Load into current pair.
             * =====================================================
             */
            Load(
                xInput,
                gammaGm_,
                col,
                valid);

            Load(
                residualInput,
                biasGm_,
                col,
                valid);

            /*
             * Reuse readyCur event for parameter readiness.
             */
            AscendC::SetFlag<
                AscendC::HardEvent::MTE2_V>(
                    readyCur);

            AscendC::WaitFlag<
                AscendC::HardEvent::MTE2_V>(
                    readyCur);

            /*
             * =====================================================
             * Parameter application.
             * =====================================================
             */
            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                AscendC::Mul(
                    yFp32,
                    yFp32,
                    xInput,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Add(
                    yFp32,
                    yFp32,
                    residualInput,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    yFp32,
                    valid);

                SyncMTE3ToV();

            } else {

                AscendC::LocalTensor<float> xFp32 =
                    xFp32Buf_.Get<float>();

                AscendC::LocalTensor<float> residualFp32 =
                    residualFp32Buf_.Get<float>();

                ToFloat(
                    xFp32,
                    xInput,
                    valid);

                ToFloat(
                    residualFp32,
                    residualInput,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Mul(
                    yFp32,
                    yFp32,
                    xFp32,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                AscendC::Add(
                    yFp32,
                    yFp32,
                    residualFp32,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                /*
                 * Native x pair becomes output staging.
                 */
                AscendC::Cast(
                    xInput,
                    yFp32,
                    AscendC::RoundMode::CAST_RINT,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    xInput,
                    valid);

                SyncMTE3ToV();
            }

            /*
             * Pair is now fully released for a future x/residual
             * prefetch.
             */
            AscendC::SetFlag<
                AscendC::HardEvent::V_MTE2>(
                    releaseCur);

            if (second) {
                releasePending1 = true;
            } else {
                releasePending0 = true;
            }
        }

        /*
         * Drain final release flags.
         */
        if (releasePending0) {

            AscendC::WaitFlag<
                AscendC::HardEvent::V_MTE2>(
                    release0);

            releasePending0 = false;
        }

        if (releasePending1) {

            AscendC::WaitFlag<
                AscendC::HardEvent::V_MTE2>(
                    release1);

            releasePending1 = false;
        }

        tpipe.ReleaseEventID<
            AscendC::HardEvent::MTE2_V>(
                ready0);

        tpipe.ReleaseEventID<
            AscendC::HardEvent::MTE2_V>(
                ready1);

        tpipe.ReleaseEventID<
            AscendC::HardEvent::V_MTE2>(
                release0);

        tpipe.ReleaseEventID<
            AscendC::HardEvent::V_MTE2>(
                release1);
    }

    /*
     * =============================================================
     * Complete R013 row.
     * =============================================================
     */

    __aicore__ inline void
    ProcessDoubleBufferedRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row * rowWidth;

        float rmsValue = 0.0f;

        FirstPassDoubleBuffered(
            rowOffset,
            rowWidth,
            invRowWidth,
            epsilon,
            rmsValue);

        SecondPassDoubleBuffered(
            rowOffset,
            rowWidth,
            rmsValue);
    }

    /*
     * =============================================================
     * D BASELINE
     *
     * Used for rowWidth <= 8192.
     * =============================================================
     */

    __aicore__ inline void ProcessRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row * rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * PASS 1
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                xFp32,
                yFp32,
                yFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum(
                partial,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                rms,
                rms,
                partial,
                1);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE2();
        }

        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToS();

        const float rmsValue =
            rms.GetValue(0);

        SyncSToV();

        /*
         * PASS 2
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Duplicate(
                residualFp32,
                rmsValue,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Div(
                yFp32,
                yFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE2();

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                yFp32,
                yFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    yFp32,
                    valid);

                SyncMTE3ToV();

            } else {

                AscendC::Cast(
                    xLocal,
                    yFp32,
                    AscendC::RoundMode::CAST_RINT,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    xLocal,
                    valid);

                SyncMTE3ToV();
            }

            SyncVToMTE2();
        }
    }

private:

    AscendC::TPipe tpipe;

    /*
     * Ping-pong Pair A.
     */
    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualBuf_;

    /*
     * FP32 work buffers.
     *
     * For FP32 these become Ping-pong Pair B.
     */
    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        yFp32Buf_;

    /*
     * Low precision only:
     * native Ping-pong Pair B.
     */
    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xPingBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualPingBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        sumBuf_;

    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;
};


/*
 * =============================================================
 * Kernel
 * =============================================================
 */

template <typename T>
__global__ __vector__ void add_rms_norm_bias_custom(
    GM_ADDR x,
    GM_ADDR residual,
    GM_ADDR gamma,
    GM_ADDR bias,
    GM_ADDR output,
    uint64_t rowCount,
    uint64_t rowWidth,
    uint64_t rowGroup,
    uint64_t groupCount,
    uint32_t blockCount,
    float invRowWidth,
    float epsilon)
{
    AddRmsNormBiasKernel<T> op;

    op.Init(
        x,
        residual,
        gamma,
        bias,
        output);

    op.Process(
        rowCount,
        rowWidth,
        rowGroup,
        groupCount,
        blockCount,
        invRowWidth,
        epsilon);
}


#if !defined(ASCENDC_CPU_DEBUG)

/*
 * =============================================================
 * Host direct invocation
 * =============================================================
 */

extern "C" void run_kernel(
    GM_ADDR x,
    const TensorGroupInfo& info_x,

    GM_ADDR residual,
    const TensorGroupInfo& info_residual,

    GM_ADDR gamma,
    const TensorGroupInfo& info_gamma,

    GM_ADDR bias,
    const TensorGroupInfo& info_bias,

    GM_ADDR output,
    const TensorGroupInfo& info_output,

    int64_t availableCoreNum,
    aclrtStream stream,
    float epsilon)
{
    if (x == nullptr ||
        residual == nullptr ||
        gamma == nullptr ||
        bias == nullptr ||
        output == nullptr ||

        info_x.tensors == nullptr ||
        info_residual.tensors == nullptr ||
        info_gamma.tensors == nullptr ||
        info_bias.tensors == nullptr ||
        info_output.tensors == nullptr ||

        info_x.numTensors < 1 ||
        info_residual.numTensors < 1 ||
        info_gamma.numTensors < 1 ||
        info_bias.numTensors < 1 ||
        info_output.numTensors < 1) {

        return;
    }

    const TensorInfo& xInfo =
        info_x.tensors[0];

    const TensorInfo& residualInfo =
        info_residual.tensors[0];

    const TensorInfo& gammaInfo =
        info_gamma.tensors[0];

    const TensorInfo& biasInfo =
        info_bias.tensors[0];

    const TensorInfo& outputInfo =
        info_output.tensors[0];

    if (xInfo.shape == nullptr ||
        residualInfo.shape == nullptr ||
        gammaInfo.shape == nullptr ||
        biasInfo.shape == nullptr ||
        outputInfo.shape == nullptr ||

        xInfo.numDims < 1 ||

        residualInfo.numDims !=
            xInfo.numDims ||

        outputInfo.numDims !=
            xInfo.numDims ||

        gammaInfo.numDims != 1 ||
        biasInfo.numDims != 1 ||

        xInfo.dtype !=
            residualInfo.dtype ||

        xInfo.dtype !=
            gammaInfo.dtype ||

        xInfo.dtype !=
            biasInfo.dtype ||

        xInfo.dtype !=
            outputInfo.dtype) {

        return;
    }

    if (xInfo.dtype != 0 &&
        xInfo.dtype != 1 &&
        xInfo.dtype != 2) {

        return;
    }

    const int64_t widthSigned =
        xInfo.shape[
            xInfo.numDims - 1];

    if (widthSigned <= 0 ||
        gammaInfo.shape[0] !=
            widthSigned ||
        biasInfo.shape[0] !=
            widthSigned) {

        return;
    }

    const uint64_t maxU64 =
        ~static_cast<uint64_t>(0);

    uint64_t rowCount = 1;

    for (int64_t dim = 0;
         dim < xInfo.numDims - 1;
         ++dim) {

        const int64_t extentSigned =
            xInfo.shape[dim];

        if (extentSigned <= 0) {
            return;
        }

        if (residualInfo.shape[dim] !=
                extentSigned ||
            outputInfo.shape[dim] !=
                extentSigned) {

            return;
        }

        const uint64_t extent =
            static_cast<uint64_t>(
                extentSigned);

        if (rowCount >
            maxU64 / extent) {

            return;
        }

        rowCount *= extent;
    }

    if (residualInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        outputInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        rowCount == 0) {

        return;
    }

    const uint64_t rowWidth =
        static_cast<uint64_t>(
            widthSigned);

    const uint64_t elementBytes =
        xInfo.dtype == 0
            ? 4ULL
            : 2ULL;

    if (rowWidth >
            maxU64 / elementBytes ||
        rowCount >
            maxU64 / rowWidth) {

        return;
    }

    const uint64_t rowBytes =
        rowWidth * elementBytes;

    uint64_t gcdA =
        rowBytes;

    uint64_t gcdB =
        32ULL;

    while (gcdB != 0ULL) {

        const uint64_t remainder =
            gcdA % gcdB;

        gcdA = gcdB;
        gcdB = remainder;
    }

    const uint64_t rowGroup =
        32ULL / gcdA;

    const uint64_t groupCount =
        rowCount / rowGroup +
        (rowCount % rowGroup == 0ULL
             ? 0ULL
             : 1ULL);

    uint64_t requestedBlocks =
        availableCoreNum > 0
            ? static_cast<uint64_t>(
                  availableCoreNum)
            : 1ULL;

    if (requestedBlocks >
        groupCount) {

        requestedBlocks =
            groupCount;
    }

    const uint64_t maxBlockCount =
        static_cast<uint64_t>(
            0xffffffffU);

    if (requestedBlocks >
        maxBlockCount) {

        requestedBlocks =
            maxBlockCount;
    }

    if (requestedBlocks == 0ULL) {
        requestedBlocks = 1ULL;
    }

    const uint32_t blockCount =
        static_cast<uint32_t>(
            requestedBlocks);

    const float invRowWidth =
        1.0f /
        static_cast<float>(
            widthSigned);

    if (xInfo.dtype == 0) {

        add_rms_norm_bias_custom<float>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else if (xInfo.dtype == 1) {

        add_rms_norm_bias_custom<half>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else {

        add_rms_norm_bias_custom<bfloat16_t>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);
    }
}
