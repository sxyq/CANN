#include <cmath>
#include <cstdint>
#include "kernel_operator.h"

/*
 * AddRmsNormBias - MIX-R014-R002-R019-V001
 *
 * Parent: MIX-R014-R002-V001
 * Status: candidate / not yet online tested
 *
 * ONLY new mechanism over the current Champion:
 *   - replace scalar rmsValue + vector Duplicate + vector Div
 *     with scalar invRms = 1/rms + vector Muls.
 *
 * Existing Champion mechanisms retained unchanged:
 *   - R014-V002: gamma/bias residency + low-precision FP32 parameter cache
 *   - R002: retain y=x+residual in UB for cached rows
 *
 * The R019 delta is applied to both the cached-row fast path and the
 * fallback D-style path. This remains one global mechanism change.
 *
 * Core properties:
 *   - kTileElems = 8192
 *   - FP32 intermediates
 *   - ReduceSum unchanged
 *   - UB layout unchanged
 *   - row/core scheduling unchanged
 */

template <typename T>
class AddRmsNormBiasKernel {
public:
    static constexpr int32_t kTileElems = 8192;
    static constexpr int32_t kParamCacheOffset = 4096;
    static constexpr uint64_t kParamCacheMaxWidth = 4096ULL;
    static constexpr int32_t kScalarElems = 16;

    __aicore__ inline void Init(
        GM_ADDR x,
        GM_ADDR residual,
        GM_ADDR gamma,
        GM_ADDR bias,
        GM_ADDR output)
    {
        xGm_.SetGlobalBuffer((__gm__ T *)x);
        residualGm_.SetGlobalBuffer((__gm__ T *)residual);
        gammaGm_.SetGlobalBuffer((__gm__ T *)gamma);
        biasGm_.SetGlobalBuffer((__gm__ T *)bias);
        outputGm_.SetGlobalBuffer((__gm__ T *)output);

        tpipe.InitBuffer(
            xBuf_,
            kTileElems * sizeof(T));

        tpipe.InitBuffer(
            residualBuf_,
            kTileElems * sizeof(T));

        tpipe.InitBuffer(
            xFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            residualFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            yFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            sumBuf_,
            kScalarElems * sizeof(float));

        /*
         * R014-V002:
         * FP16/BF16 parameters converted to FP32 once per AI Core.
         */
        if constexpr (
            !AscendC::IsSameType<T, float>::value) {

            tpipe.InitBuffer(
                gammaFp32Buf_,
                kParamCacheOffset * sizeof(float));

            tpipe.InitBuffer(
                biasFp32Buf_,
                kParamCacheOffset * sizeof(float));
        }
    }

    __aicore__ inline void Process(
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowGroup,
        uint64_t groupCount,
        uint32_t blockCount,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t blockIdx =
            static_cast<uint64_t>(
                AscendC::GetBlockIdx());

        const uint64_t baseGroups =
            groupCount /
            static_cast<uint64_t>(
                blockCount);

        const uint64_t extraGroups =
            groupCount %
            static_cast<uint64_t>(
                blockCount);

        const uint64_t beginGroup =
            blockIdx * baseGroups +
            (blockIdx < extraGroups
                 ? blockIdx
                 : extraGroups);

        const uint64_t localGroups =
            baseGroups +
            (blockIdx < extraGroups
                 ? 1ULL
                 : 0ULL);

        const uint64_t endGroup =
            beginGroup +
            localGroups;

        const uint64_t beginRow =
            beginGroup *
            rowGroup;

        uint64_t endRow =
            endGroup *
            rowGroup;

        if (endRow > rowCount) {
            endRow = rowCount;
        }

        const uint64_t localRows =
            endRow > beginRow
                ? endRow - beginRow
                : 0ULL;

        /*
         * Preserve R014-V002 activation condition exactly.
         */
        const bool useParamCache =
            rowWidth <=
                kParamCacheMaxWidth &&
            localRows > 1ULL;

        if (useParamCache) {

            PrepareParameterCache(
                rowWidth);

            for (uint64_t row = beginRow;
                 row < endRow;
                 ++row) {

                ProcessCachedParamsRetainedYRow(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);
            }

        } else {

            for (uint64_t row = beginRow;
                 row < endRow;
                 ++row) {

                ProcessRow(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);
            }
        }
    }

private:

    __aicore__ inline int32_t TileLength(
        uint64_t remain) const
    {
        return static_cast<int32_t>(
            remain >
                    static_cast<uint64_t>(
                        kTileElems)
                ? kTileElems
                : remain);
    }

    __aicore__ inline void Load(
        const AscendC::LocalTensor<T>& dst,
        const AscendC::GlobalTensor<T>& src,
        uint64_t offset,
        int32_t count)
    {
        const AscendC::DataCopyExtParams copyParams(
            1,
            static_cast<uint32_t>(
                count *
                static_cast<int32_t>(
                    sizeof(T))),
            0,
            0,
            0);

        const AscendC::DataCopyPadExtParams<T>
            padParams;

        AscendC::DataCopyPad(
            dst,
            src[offset],
            copyParams,
            padParams);
    }

    __aicore__ inline void Store(
        const AscendC::GlobalTensor<T>& dst,
        uint64_t offset,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        const AscendC::DataCopyExtParams copyParams(
            1,
            static_cast<uint32_t>(
                count *
                static_cast<int32_t>(
                    sizeof(T))),
            0,
            0,
            0);

        AscendC::DataCopyPad(
            dst[offset],
            src,
            copyParams);
    }

    __aicore__ inline void ToFloat(
        const AscendC::LocalTensor<float>& dst,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::Adds(
                dst,
                src,
                0.0f,
                count);

        } else {

            AscendC::Cast(
                dst,
                src,
                AscendC::RoundMode::CAST_NONE,
                count);
        }
    }

    __aicore__ inline void SyncMTE2ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE2_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE2_V>(0);
    }

    __aicore__ inline void SyncVToMTE2()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE2>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE2>(0);
    }

    __aicore__ inline void SyncVToMTE3()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE3>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE3>(0);
    }

    __aicore__ inline void SyncMTE3ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE3_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE3_V>(0);
    }

    __aicore__ inline void SyncVToS()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_S>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_S>(0);
    }

    __aicore__ inline void SyncSToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::S_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::S_V>(0);
    }

    /*
     * =============================================================
     * R014-V002
     *
     * gamma/bias are loaded once per AI Core and reused across rows.
     * =============================================================
     */
    __aicore__ inline void PrepareParameterCache(
        uint64_t rowWidth)
    {
        const int32_t valid =
            static_cast<int32_t>(
                rowWidth);

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        /*
         * second half:
         *
         * [0, 4096)    : row input
         * [4096, 8192) : cached parameters
         */
        AscendC::LocalTensor<T> gammaCached =
            xLocal[kParamCacheOffset];

        AscendC::LocalTensor<T> biasCached =
            residualLocal[kParamCacheOffset];

        Load(
            gammaCached,
            gammaGm_,
            0,
            valid);

        Load(
            biasCached,
            biasGm_,
            0,
            valid);

        SyncMTE2ToV();

        /*
         * FP16 / BF16:
         * convert once and retain FP32 parameters.
         */
        if constexpr (
            !AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::LocalTensor<float>
                gammaFp32 =
                    gammaFp32Buf_.Get<float>();

            AscendC::LocalTensor<float>
                biasFp32 =
                    biasFp32Buf_.Get<float>();

            ToFloat(
                gammaFp32,
                gammaCached,
                valid);

            ToFloat(
                biasFp32,
                biasCached,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE2();
        }
    }

    /*
     * =============================================================
     * Fast path:
     *
     * R014:
     *   cached gamma/bias
     *
     * R002:
     *   retained y=x+residual
     *
     * R019:
     *   scalar invRms + Muls
     * =============================================================
     */
    __aicore__ inline void
    ProcessCachedParamsRetainedYRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        /*
         * Fast path activation guarantees rowWidth <= 4096.
         */
        const int32_t valid =
            static_cast<int32_t>(
                rowWidth);

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<T> gammaCached =
            xLocal[kParamCacheOffset];

        AscendC::LocalTensor<T> biasCached =
            residualLocal[kParamCacheOffset];

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        /*
         * =========================================================
         * PASS 1
         *
         * R002 retains yFp32.
         * =========================================================
         */

        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        Load(
            xLocal,
            xGm_,
            rowOffset,
            valid);

        Load(
            residualLocal,
            residualGm_,
            rowOffset,
            valid);

        SyncMTE2ToV();

        ToFloat(
            xFp32,
            xLocal,
            valid);

        ToFloat(
            residualFp32,
            residualLocal,
            valid);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            xFp32,
            residualFp32,
            valid);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * y is retained.
         * xFp32 becomes square workspace.
         */
        AscendC::Mul(
            xFp32,
            yFp32,
            yFp32,
            valid);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * residualFp32 is dead after Add,
         * reuse it as ReduceSum workspace.
         */
        AscendC::ReduceSum(
            partial,
            xFp32,
            residualFp32,
            valid);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            rms,
            rms,
            partial,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToMTE2();

        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * =========================================================
         * R019
         *
         * Previous:
         *
         *   rmsValue = rms.GetValue(0)
         *   Duplicate(rms vector)
         *   Div(y, rms vector)
         *
         * New:
         *
         *   invRms = 1.0 / rms
         *   Muls(y, invRms)
         * =========================================================
         */

        SyncVToS();

        const float invRms =
            1.0f /
            rms.GetValue(0);

        SyncSToV();

        /*
         * R002:
         * no second x/residual GM load.
         */
        AscendC::Muls(
            yFp32,
            yFp32,
            invRms,
            valid);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * =========================================================
         * R014 parameter reuse
         * =========================================================
         */

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::Mul(
                yFp32,
                yFp32,
                gammaCached,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                biasCached,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

        } else {

            AscendC::LocalTensor<float>
                gammaFp32 =
                    gammaFp32Buf_.Get<float>();

            AscendC::LocalTensor<float>
                biasFp32 =
                    biasFp32Buf_.Get<float>();

            AscendC::Mul(
                yFp32,
                yFp32,
                gammaFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                biasFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();
        }

        /*
         * =========================================================
         * Output
         * =========================================================
         */

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset,
                yFp32,
                valid);

            SyncMTE3ToV();

        } else {

            /*
             * first half of xLocal is safe for output staging;
             * gamma cache starts at offset 4096.
             */
            AscendC::Cast(
                xLocal,
                yFp32,
                AscendC::RoundMode::CAST_RINT,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE3();

            Store(
                outputGm_,
                rowOffset,
                xLocal,
                valid);

            SyncMTE3ToV();
        }

        SyncVToMTE2();
    }

    /*
     * =============================================================
     * Fallback path
     *
     * D-style two-pass algorithm
     * +
     * R019 normalization.
     * =============================================================
     */
    __aicore__ inline void ProcessRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row *
            rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        /*
         * =========================================================
         * PASS 1
         * =========================================================
         */

        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                xFp32,
                yFp32,
                yFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum(
                partial,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                rms,
                rms,
                partial,
                1);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE2();
        }

        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * R019 scalar reciprocal.
         */
        SyncVToS();

        const float invRms =
            1.0f /
            rms.GetValue(0);

        SyncSToV();

        /*
         * =========================================================
         * PASS 2
         * =========================================================
         */

        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            /*
             * R019:
             *
             * no Duplicate(rms)
             * no vector Div
             */
            AscendC::Muls(
                yFp32,
                yFp32,
                invRms,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            /*
             * gamma / bias behavior unchanged.
             */
            SyncVToMTE2();

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                yFp32,
                yFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    yFp32,
                    valid);

                SyncMTE3ToV();

            } else {

                AscendC::Cast(
                    xLocal,
                    yFp32,
                    AscendC::RoundMode::CAST_RINT,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    xLocal,
                    valid);

                SyncMTE3ToV();
            }

            SyncVToMTE2();
        }
    }

private:

    AscendC::TPipe tpipe;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        yFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        sumBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        gammaFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        biasFp32Buf_;

    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;
};


/*
 * =============================================================
 * Device kernel
 * =============================================================
 */

template <typename T>
__global__ __vector__ void add_rms_norm_bias_custom(
    GM_ADDR x,
    GM_ADDR residual,
    GM_ADDR gamma,
    GM_ADDR bias,
    GM_ADDR output,
    uint64_t rowCount,
    uint64_t rowWidth,
    uint64_t rowGroup,
    uint64_t groupCount,
    uint32_t blockCount,
    float invRowWidth,
    float epsilon)
{
    AddRmsNormBiasKernel<T> op;

    op.Init(
        x,
        residual,
        gamma,
        bias,
        output);

    op.Process(
        rowCount,
        rowWidth,
        rowGroup,
        groupCount,
        blockCount,
        invRowWidth,
        epsilon);
}


#if !defined(ASCENDC_CPU_DEBUG)

/*
 * =============================================================
 * Direct-call host entry
 * =============================================================
 */

extern "C" void run_kernel(
    GM_ADDR x,
    const TensorGroupInfo& info_x,

    GM_ADDR residual,
    const TensorGroupInfo& info_residual,

    GM_ADDR gamma,
    const TensorGroupInfo& info_gamma,

    GM_ADDR bias,
    const TensorGroupInfo& info_bias,

    GM_ADDR output,
    const TensorGroupInfo& info_output,

    int64_t availableCoreNum,
    aclrtStream stream,
    float epsilon)
{
    if (x == nullptr ||
        residual == nullptr ||
        gamma == nullptr ||
        bias == nullptr ||
        output == nullptr ||

        info_x.tensors == nullptr ||
        info_residual.tensors == nullptr ||
        info_gamma.tensors == nullptr ||
        info_bias.tensors == nullptr ||
        info_output.tensors == nullptr ||

        info_x.numTensors < 1 ||
        info_residual.numTensors < 1 ||
        info_gamma.numTensors < 1 ||
        info_bias.numTensors < 1 ||
        info_output.numTensors < 1) {

        return;
    }

    const TensorInfo& xInfo =
        info_x.tensors[0];

    const TensorInfo& residualInfo =
        info_residual.tensors[0];

    const TensorInfo& gammaInfo =
        info_gamma.tensors[0];

    const TensorInfo& biasInfo =
        info_bias.tensors[0];

    const TensorInfo& outputInfo =
        info_output.tensors[0];

    if (xInfo.shape == nullptr ||
        residualInfo.shape == nullptr ||
        gammaInfo.shape == nullptr ||
        biasInfo.shape == nullptr ||
        outputInfo.shape == nullptr ||

        xInfo.numDims < 1 ||

        residualInfo.numDims !=
            xInfo.numDims ||

        outputInfo.numDims !=
            xInfo.numDims ||

        gammaInfo.numDims != 1 ||
        biasInfo.numDims != 1 ||

        xInfo.dtype !=
            residualInfo.dtype ||

        xInfo.dtype !=
            gammaInfo.dtype ||

        xInfo.dtype !=
            biasInfo.dtype ||

        xInfo.dtype !=
            outputInfo.dtype) {

        return;
    }

    if (xInfo.dtype != 0 &&
        xInfo.dtype != 1 &&
        xInfo.dtype != 2) {

        return;
    }

    const int64_t widthSigned =
        xInfo.shape[
            xInfo.numDims - 1];

    if (widthSigned <= 0 ||
        gammaInfo.shape[0] !=
            widthSigned ||
        biasInfo.shape[0] !=
            widthSigned) {

        return;
    }

    const uint64_t maxU64 =
        ~static_cast<uint64_t>(0);

    uint64_t rowCount = 1;

    for (int64_t dim = 0;
         dim < xInfo.numDims - 1;
         ++dim) {

        const int64_t extentSigned =
            xInfo.shape[dim];

        if (extentSigned <= 0) {
            return;
        }

        if (residualInfo.shape[dim] !=
                extentSigned ||
            outputInfo.shape[dim] !=
                extentSigned) {

            return;
        }

        const uint64_t extent =
            static_cast<uint64_t>(
                extentSigned);

        if (rowCount >
            maxU64 / extent) {

            return;
        }

        rowCount *= extent;
    }

    if (residualInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        outputInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        rowCount == 0) {

        return;
    }

    const uint64_t rowWidth =
        static_cast<uint64_t>(
            widthSigned);

    const uint64_t elementBytes =
        xInfo.dtype == 0
            ? 4ULL
            : 2ULL;

    if (rowWidth >
            maxU64 / elementBytes ||
        rowCount >
            maxU64 / rowWidth) {

        return;
    }

    const uint64_t rowBytes =
        rowWidth *
        elementBytes;

    uint64_t gcdA =
        rowBytes;

    uint64_t gcdB =
        32ULL;

    while (gcdB != 0ULL) {

        const uint64_t remainder =
            gcdA %
            gcdB;

        gcdA = gcdB;
        gcdB = remainder;
    }

    const uint64_t rowGroup =
        32ULL /
        gcdA;

    const uint64_t groupCount =
        rowCount /
            rowGroup +
        (rowCount %
                    rowGroup ==
                0ULL
             ? 0ULL
             : 1ULL);

    uint64_t requestedBlocks =
        availableCoreNum > 0
            ? static_cast<uint64_t>(
                  availableCoreNum)
            : 1ULL;

    if (requestedBlocks >
        groupCount) {

        requestedBlocks =
            groupCount;
    }

    const uint64_t maxBlockCount =
        static_cast<uint64_t>(
            0xffffffffU);

    if (requestedBlocks >
        maxBlockCount) {

        requestedBlocks =
            maxBlockCount;
    }

    if (requestedBlocks == 0ULL) {
        requestedBlocks = 1ULL;
    }

    const uint32_t blockCount =
        static_cast<uint32_t>(
            requestedBlocks);

    const float invRowWidth =
        1.0f /
        static_cast<float>(
            widthSigned);

    if (xInfo.dtype == 0) {

        add_rms_norm_bias_custom<float>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else if (xInfo.dtype == 1) {

        add_rms_norm_bias_custom<half>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else {

        add_rms_norm_bias_custom<bfloat16_t>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);
    }
}
