#include <stdint.h>
#include "kernel_operator.h"

namespace {

constexpr uint32_t TILE_LENGTH = 2048;
constexpr uint32_t WORK_LENGTH = 1024;
constexpr uint32_t BLOCK_BYTES = 32;

template <typename T>
struct CastToFloat {
    __aicore__ static inline void Run(const AscendC::LocalTensor<float>& dst,
                                      const AscendC::LocalTensor<T>& src,
                                      uint32_t count)
    {
        AscendC::Cast<float, T>(dst, src, AscendC::RoundMode::CAST_NONE, count);
    }
};

template <>
struct CastToFloat<float> {
    __aicore__ static inline void Run(const AscendC::LocalTensor<float>& dst,
                                      const AscendC::LocalTensor<float>& src,
                                      uint32_t count)
    {
        AscendC::DataCopy(dst, src, count);
    }
};

template <typename T>
struct CastFromFloat {
    __aicore__ static inline void Run(const AscendC::LocalTensor<T>& dst,
                                      const AscendC::LocalTensor<float>& src,
                                      uint32_t count)
    {
        AscendC::Cast<T, float>(dst, src, AscendC::RoundMode::CAST_RINT, count);
    }
};

template <>
struct CastFromFloat<float> {
    __aicore__ static inline void Run(const AscendC::LocalTensor<float>& dst,
                                      const AscendC::LocalTensor<float>& src,
                                      uint32_t count)
    {
        AscendC::DataCopy(dst, src, count);
    }
};

template <typename T>
class AddRmsNormBiasKernel {
public:
    __aicore__ inline AddRmsNormBiasKernel(AscendC::TPipe* pipe) : pipe_(pipe) {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma, GM_ADDR bias,
                                GM_ADDR output, uint64_t dim, uint64_t outer,
                                uint64_t rowsPerBlock, float invDim, float epsilon)
    {
        dim_ = dim;
        rowStart_ = static_cast<uint64_t>(AscendC::GetBlockIdx()) * rowsPerBlock;
        rowCount_ = rowStart_ < outer ? outer - rowStart_ : 0;
        if (rowCount_ > rowsPerBlock) {
            rowCount_ = rowsPerBlock;
        }
        invDim_ = invDim;
        epsilon_ = epsilon;

        xGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(x));
        residualGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(residual));
        gammaGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(gamma));
        biasGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(bias));
        outputGm_.SetGlobalBuffer(reinterpret_cast<__gm__ T*>(output));

        pipe_->InitBuffer(xQueue_, 1, TILE_LENGTH * sizeof(T));
        pipe_->InitBuffer(residualQueue_, 1, TILE_LENGTH * sizeof(T));
        pipe_->InitBuffer(gammaQueue_, 1, TILE_LENGTH * sizeof(T));
        pipe_->InitBuffer(biasQueue_, 1, TILE_LENGTH * sizeof(T));
        pipe_->InitBuffer(outputQueue_, 1, TILE_LENGTH * sizeof(T));

        pipe_->InitBuffer(xFloatBuf_, TILE_LENGTH * sizeof(float));
        pipe_->InitBuffer(residualFloatBuf_, TILE_LENGTH * sizeof(float));
        pipe_->InitBuffer(gammaFloatBuf_, TILE_LENGTH * sizeof(float));
        pipe_->InitBuffer(biasFloatBuf_, TILE_LENGTH * sizeof(float));
        pipe_->InitBuffer(squareFloatBuf_, TILE_LENGTH * sizeof(float));
        pipe_->InitBuffer(reduceTmpBuf_, WORK_LENGTH * sizeof(float));
        pipe_->InitBuffer(sumBuf_, BLOCK_BYTES);
    }

    __aicore__ inline void Process()
    {
        for (uint64_t row = 0; row < rowCount_; ++row) {
            const uint64_t rowIndex = rowStart_ + row;
            const uint64_t rowBase = rowIndex * dim_;
            const float rms = ComputeRms(rowBase);
            WriteNormalizedRow(rowBase, rms);
        }
    }

private:
    __aicore__ inline uint32_t AlignCount(uint32_t count) const
    {
        const uint32_t elementsPerBlock = BLOCK_BYTES / sizeof(T);
        return (count + elementsPerBlock - 1) / elementsPerBlock * elementsPerBlock;
    }

    __aicore__ inline AscendC::DataCopyExtParams MakeCopyParams(uint32_t count) const
    {
        return AscendC::DataCopyExtParams(1, count * sizeof(T), 0, 0, 0);
    }

    __aicore__ inline AscendC::DataCopyPadExtParams<T> MakePadParams(uint32_t count,
                                                                      uint32_t alignedCount) const
    {
        AscendC::DataCopyPadExtParams<T> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = static_cast<uint8_t>(alignedCount - count);
        padParams.paddingValue = static_cast<T>(0);
        return padParams;
    }

    __aicore__ inline void CopyInputPair(uint64_t rowBase, uint32_t offset, uint32_t count,
                                         uint32_t alignedCount)
    {
        AscendC::LocalTensor<T> xLocal = xQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> residualLocal = residualQueue_.AllocTensor<T>();
        const AscendC::DataCopyExtParams copyParams = MakeCopyParams(count);
        const AscendC::DataCopyPadExtParams<T> padParams = MakePadParams(count, alignedCount);
        AscendC::DataCopyPad(xLocal, xGm_[rowBase + offset], copyParams, padParams);
        AscendC::DataCopyPad(residualLocal, residualGm_[rowBase + offset], copyParams, padParams);
        xQueue_.EnQue(xLocal);
        residualQueue_.EnQue(residualLocal);
    }

    __aicore__ inline void CopyAllInputs(uint64_t rowBase, uint32_t offset, uint32_t count,
                                         uint32_t alignedCount)
    {
        AscendC::LocalTensor<T> xLocal = xQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> residualLocal = residualQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> gammaLocal = gammaQueue_.AllocTensor<T>();
        AscendC::LocalTensor<T> biasLocal = biasQueue_.AllocTensor<T>();
        const AscendC::DataCopyExtParams copyParams = MakeCopyParams(count);
        const AscendC::DataCopyPadExtParams<T> padParams = MakePadParams(count, alignedCount);
        AscendC::DataCopyPad(xLocal, xGm_[rowBase + offset], copyParams, padParams);
        AscendC::DataCopyPad(residualLocal, residualGm_[rowBase + offset], copyParams, padParams);
        AscendC::DataCopyPad(gammaLocal, gammaGm_[offset], copyParams, padParams);
        AscendC::DataCopyPad(biasLocal, biasGm_[offset], copyParams, padParams);
        xQueue_.EnQue(xLocal);
        residualQueue_.EnQue(residualLocal);
        gammaQueue_.EnQue(gammaLocal);
        biasQueue_.EnQue(biasLocal);
    }

    __aicore__ inline float ComputeRms(uint64_t rowBase)
    {
        float sum = 0.0f;
        uint32_t offset = 0;
        while (offset < dim_) {
            const uint32_t remaining = static_cast<uint32_t>(dim_ - offset);
            const uint32_t count = remaining < TILE_LENGTH ? remaining : TILE_LENGTH;
            const uint32_t alignedCount = AlignCount(count);
            CopyInputPair(rowBase, offset, count, alignedCount);

            AscendC::LocalTensor<T> xLocal = xQueue_.DeQue<T>();
            AscendC::LocalTensor<T> residualLocal = residualQueue_.DeQue<T>();
            AscendC::LocalTensor<float> xFloat = xFloatBuf_.Get<float>();
            AscendC::LocalTensor<float> residualFloat = residualFloatBuf_.Get<float>();
            AscendC::LocalTensor<float> squareFloat = squareFloatBuf_.Get<float>();
            AscendC::LocalTensor<float> reduceTmp = reduceTmpBuf_.Get<float>();
            AscendC::LocalTensor<float> sumLocal = sumBuf_.Get<float>();

            CastToFloat<T>::Run(xFloat, xLocal, alignedCount);
            CastToFloat<T>::Run(residualFloat, residualLocal, alignedCount);
            AscendC::Add(xFloat, xFloat, residualFloat, count);
            AscendC::Duplicate(squareFloat, 0.0f, alignedCount);
            AscendC::Mul(squareFloat, xFloat, xFloat, count);
            AscendC::ReduceSum<float>(sumLocal, squareFloat, reduceTmp, alignedCount);
            AscendC::PipeBarrier<PIPE_V>();
            sum += sumLocal.GetValue(0);

            xQueue_.FreeTensor(xLocal);
            residualQueue_.FreeTensor(residualLocal);
            offset += count;
        }

        AscendC::LocalTensor<float> scalar = squareFloatBuf_.Get<float>();
        float mean = sum * invDim_ + epsilon_;
        AscendC::Duplicate(scalar, mean, BLOCK_BYTES / sizeof(float));
        AscendC::Sqrt(scalar, scalar, BLOCK_BYTES / sizeof(float));
        AscendC::PipeBarrier<PIPE_V>();
        return scalar.GetValue(0);
    }

    __aicore__ inline void WriteNormalizedRow(uint64_t rowBase, float rms)
    {
        uint32_t offset = 0;
        while (offset < dim_) {
            const uint32_t remaining = static_cast<uint32_t>(dim_ - offset);
            const uint32_t count = remaining < TILE_LENGTH ? remaining : TILE_LENGTH;
            const uint32_t alignedCount = AlignCount(count);
            CopyAllInputs(rowBase, offset, count, alignedCount);

            AscendC::LocalTensor<T> xLocal = xQueue_.DeQue<T>();
            AscendC::LocalTensor<T> residualLocal = residualQueue_.DeQue<T>();
            AscendC::LocalTensor<T> gammaLocal = gammaQueue_.DeQue<T>();
            AscendC::LocalTensor<T> biasLocal = biasQueue_.DeQue<T>();
            AscendC::LocalTensor<T> outputLocal = outputQueue_.AllocTensor<T>();
            AscendC::LocalTensor<float> xFloat = xFloatBuf_.Get<float>();
            AscendC::LocalTensor<float> residualFloat = residualFloatBuf_.Get<float>();
            AscendC::LocalTensor<float> gammaFloat = gammaFloatBuf_.Get<float>();
            AscendC::LocalTensor<float> biasFloat = biasFloatBuf_.Get<float>();

            CastToFloat<T>::Run(xFloat, xLocal, alignedCount);
            CastToFloat<T>::Run(residualFloat, residualLocal, alignedCount);
            CastToFloat<T>::Run(gammaFloat, gammaLocal, alignedCount);
            CastToFloat<T>::Run(biasFloat, biasLocal, alignedCount);
            AscendC::Add(xFloat, xFloat, residualFloat, count);
            AscendC::Muls(xFloat, xFloat, 1.0f / rms, count);
            AscendC::Mul(xFloat, xFloat, gammaFloat, count);
            AscendC::Add(xFloat, xFloat, biasFloat, count);
            CastFromFloat<T>::Run(outputLocal, xFloat, count);
            outputQueue_.EnQue(outputLocal);

            AscendC::LocalTensor<T> outputReady = outputQueue_.DeQue<T>();
            const AscendC::DataCopyExtParams copyParams = MakeCopyParams(count);
            AscendC::DataCopyPad(outputGm_[rowBase + offset], outputReady, copyParams);

            xQueue_.FreeTensor(xLocal);
            residualQueue_.FreeTensor(residualLocal);
            gammaQueue_.FreeTensor(gammaLocal);
            biasQueue_.FreeTensor(biasLocal);
            outputQueue_.FreeTensor(outputReady);
            offset += count;
        }
    }

    AscendC::TPipe* pipe_;
    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;

    AscendC::TQue<AscendC::TPosition::VECIN, 1> xQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> residualQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gammaQueue_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue_;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outputQueue_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> residualFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> squareFloatBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceTmpBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf_;

    uint64_t dim_ = 0;
    uint64_t rowStart_ = 0;
    uint64_t rowCount_ = 0;
    float invDim_ = 0.0f;
    float epsilon_ = 0.0f;
};

template <typename T>
__global__ __vector__ void add_rms_norm_bias_kernel(GM_ADDR x, GM_ADDR residual, GM_ADDR gamma,
                                                     GM_ADDR bias, GM_ADDR output, uint64_t dim,
                                                     uint64_t outer, uint64_t rowsPerBlock,
                                                     float invDim, float epsilon)
{
    AscendC::TPipe pipe;
    AddRmsNormBiasKernel<T> kernel(&pipe);
    kernel.Init(x, residual, gamma, bias, output, dim, outer, rowsPerBlock, invDim, epsilon);
    kernel.Process();
}

inline uint64_t Product(const TensorInfo& tensor)
{
    uint64_t result = 1;
    for (int64_t i = 0; i < tensor.numDims; ++i) {
        result *= static_cast<uint64_t>(tensor.shape[i]);
    }
    return result;
}

inline uint64_t LastDim(const TensorInfo& tensor)
{
    return static_cast<uint64_t>(tensor.shape[tensor.numDims - 1]);
}

inline uint64_t Gcd(uint64_t a, uint64_t b)
{
    while (b != 0) {
        const uint64_t next = a % b;
        a = b;
        b = next;
    }
    return a;
}

inline uint64_t RowsPerCacheAlignedGroup(uint64_t dim, uint32_t elementBytes)
{
    const uint64_t rowBytes = dim * elementBytes;
    const uint64_t remainder = rowBytes % BLOCK_BYTES;
    if (remainder == 0) {
        return 1;
    }
    return BLOCK_BYTES / Gcd(remainder, BLOCK_BYTES);
}

template <typename T>
inline void LaunchAddRmsNormBias(GM_ADDR x, const TensorGroupInfo& info_x,
                                 GM_ADDR residual, const TensorGroupInfo& info_residual,
                                 GM_ADDR gamma, const TensorGroupInfo& info_gamma,
                                 GM_ADDR bias, const TensorGroupInfo& info_bias,
                                 GM_ADDR output, const TensorGroupInfo& info_output,
                                 int64_t availableCoreNum, aclrtStream stream, float epsilon)
{
    (void)info_residual;
    (void)info_gamma;
    (void)info_bias;
    (void)info_output;

    const TensorInfo& xTensor = info_x.tensors[0];
    const uint64_t totalElements = Product(xTensor);
    const uint64_t dim = LastDim(xTensor);
    const uint64_t outer = totalElements / dim;
    const float invDim = 1.0f / static_cast<float>(dim);
    const uint64_t groupRows = RowsPerCacheAlignedGroup(dim, sizeof(T));
    const uint64_t groupCount = (outer + groupRows - 1) / groupRows;
    const uint64_t coreLimit = availableCoreNum > 0 ? static_cast<uint64_t>(availableCoreNum) : 1;
    const uint64_t blockNum = groupCount < coreLimit ? groupCount : coreLimit;
    const uint64_t groupsPerBlock = (groupCount + blockNum - 1) / blockNum;
    const uint64_t rowsPerBlock = groupsPerBlock * groupRows;

    add_rms_norm_bias_kernel<T><<<static_cast<uint32_t>(blockNum), nullptr, stream>>>(
        x, residual, gamma, bias, output, dim, outer, rowsPerBlock, invDim, epsilon);
}

} // namespace

extern "C" void run_kernel(GM_ADDR x, const TensorGroupInfo& info_x, GM_ADDR residual,
                            const TensorGroupInfo& info_residual, GM_ADDR gamma,
                            const TensorGroupInfo& info_gamma, GM_ADDR bias,
                            const TensorGroupInfo& info_bias, GM_ADDR output,
                            const TensorGroupInfo& info_output, int64_t availableCoreNum,
                            aclrtStream stream, float epsilon)
{
    const int32_t dtype = info_x.tensors[0].dtype;
    if (dtype == 0) {
        LaunchAddRmsNormBias<float>(x, info_x, residual, info_residual, gamma, info_gamma,
                                    bias, info_bias, output, info_output, availableCoreNum,
                                    stream, epsilon);
    } else if (dtype == 1) {
        LaunchAddRmsNormBias<half>(x, info_x, residual, info_residual, gamma, info_gamma,
                                   bias, info_bias, output, info_output, availableCoreNum,
                                   stream, epsilon);
    } else if (dtype == 2) {
        LaunchAddRmsNormBias<bfloat16_t>(x, info_x, residual, info_residual, gamma, info_gamma,
                                         bias, info_bias, output, info_output, availableCoreNum,
                                         stream, epsilon);
    }
}
