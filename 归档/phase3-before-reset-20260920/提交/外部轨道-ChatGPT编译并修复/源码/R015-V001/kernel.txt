#include <cmath>
#include <cstdint>
#include "kernel_operator.h"

/*
 * AddRmsNormBias
 *
 * R015-V001
 *
 * Parent:
 *   D baseline
 *
 * Only optimization in this version:
 *
 *   When two consecutive rows owned by the same core satisfy
 *
 *       2 * rowWidth <= kTileElems
 *
 *   load x/residual for the two rows with ONE contiguous DMA
 *   in each pass.
 *
 * Everything else remains intentionally unchanged:
 *
 *   - kTileElems = 8192
 *   - two-pass algorithm
 *   - FP32 intermediates
 *   - ReduceSum
 *   - residualFp32 reused as ReduceSum workspace
 *   - per-row RMS scalar synchronization
 *   - per-row gamma/bias load
 *   - per-row output store
 *   - Div + Sqrt
 *
 * This is an isolated R015 multi-row-input-DMA experiment.
 */

template <typename T>
class AddRmsNormBiasKernel {
public:
    static constexpr int32_t kTileElems = 8192;
    static constexpr int32_t kScalarElems = 16;

    __aicore__ inline void Init(
        GM_ADDR x,
        GM_ADDR residual,
        GM_ADDR gamma,
        GM_ADDR bias,
        GM_ADDR output)
    {
        xGm_.SetGlobalBuffer((__gm__ T *)x);
        residualGm_.SetGlobalBuffer((__gm__ T *)residual);
        gammaGm_.SetGlobalBuffer((__gm__ T *)gamma);
        biasGm_.SetGlobalBuffer((__gm__ T *)bias);
        outputGm_.SetGlobalBuffer((__gm__ T *)output);

        tpipe.InitBuffer(
            xBuf_,
            kTileElems * sizeof(T));

        tpipe.InitBuffer(
            residualBuf_,
            kTileElems * sizeof(T));

        tpipe.InitBuffer(
            xFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            residualFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            yFp32Buf_,
            kTileElems * sizeof(float));

        tpipe.InitBuffer(
            sumBuf_,
            kScalarElems * sizeof(float));
    }

    __aicore__ inline void Process(
        uint64_t rowCount,
        uint64_t rowWidth,
        uint64_t rowGroup,
        uint64_t groupCount,
        uint32_t blockCount,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t blockIdx =
            static_cast<uint64_t>(
                AscendC::GetBlockIdx());

        const uint64_t baseGroups =
            groupCount /
            static_cast<uint64_t>(blockCount);

        const uint64_t extraGroups =
            groupCount %
            static_cast<uint64_t>(blockCount);

        const uint64_t beginGroup =
            blockIdx * baseGroups +
            (blockIdx < extraGroups
                 ? blockIdx
                 : extraGroups);

        const uint64_t localGroups =
            baseGroups +
            (blockIdx < extraGroups
                 ? 1
                 : 0);

        const uint64_t endGroup =
            beginGroup + localGroups;

        /*
         * The groups assigned to one block are contiguous.
         *
         * Therefore their rows are also contiguous and can be
         * safely paired without changing row ownership.
         */
        const uint64_t beginRow =
            beginGroup * rowGroup;

        uint64_t endRow =
            endGroup * rowGroup;

        if (endRow > rowCount) {
            endRow = rowCount;
        }

        /*
         * R015-V001:
         *
         * Only activate the 2-row path when both rows fit inside
         * the existing 8192-element buffers.
         *
         * No additional UB allocation is introduced.
         */
        const bool useTwoRowDma =
            rowWidth <=
            static_cast<uint64_t>(
                kTileElems / 2);

        uint64_t row = beginRow;

        if (useTwoRowDma) {
            for (;
                 row + 1 < endRow;
                 row += 2) {

                ProcessTwoRows(
                    row,
                    rowWidth,
                    invRowWidth,
                    epsilon);
            }
        }

        /*
         * Odd remaining row, or all rows for width > 4096,
         * use the unchanged D-baseline path.
         */
        for (;
             row < endRow;
             ++row) {

            ProcessRow(
                row,
                rowWidth,
                invRowWidth,
                epsilon);
        }
    }

private:
    __aicore__ inline int32_t TileLength(
        uint64_t remain) const
    {
        return static_cast<int32_t>(
            remain >
                    static_cast<uint64_t>(
                        kTileElems)
                ? kTileElems
                : remain);
    }

    __aicore__ inline void Load(
        const AscendC::LocalTensor<T>& dst,
        const AscendC::GlobalTensor<T>& src,
        uint64_t offset,
        int32_t count)
    {
        const AscendC::DataCopyExtParams copyParams(
            1,
            static_cast<uint32_t>(
                count *
                static_cast<int32_t>(
                    sizeof(T))),
            0,
            0,
            0);

        const AscendC::DataCopyPadExtParams<T>
            padParams;

        AscendC::DataCopyPad(
            dst,
            src[offset],
            copyParams,
            padParams);
    }

    __aicore__ inline void Store(
        const AscendC::GlobalTensor<T>& dst,
        uint64_t offset,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        const AscendC::DataCopyExtParams copyParams(
            1,
            static_cast<uint32_t>(
                count *
                static_cast<int32_t>(
                    sizeof(T))),
            0,
            0,
            0);

        AscendC::DataCopyPad(
            dst[offset],
            src,
            copyParams);
    }

    __aicore__ inline void ToFloat(
        const AscendC::LocalTensor<float>& dst,
        const AscendC::LocalTensor<T>& src,
        int32_t count)
    {
        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            AscendC::Adds(
                dst,
                src,
                0.0f,
                count);

        } else {

            AscendC::Cast(
                dst,
                src,
                AscendC::RoundMode::CAST_NONE,
                count);
        }
    }

    __aicore__ inline void SyncMTE2ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE2_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE2_V>(0);
    }

    __aicore__ inline void SyncVToMTE2()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE2>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE2>(0);
    }

    __aicore__ inline void SyncVToMTE3()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_MTE3>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_MTE3>(0);
    }

    __aicore__ inline void SyncMTE3ToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::MTE3_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::MTE3_V>(0);
    }

    __aicore__ inline void SyncVToS()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::V_S>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::V_S>(0);
    }

    __aicore__ inline void SyncSToV()
    {
        AscendC::SetFlag<
            AscendC::HardEvent::S_V>(0);

        AscendC::WaitFlag<
            AscendC::HardEvent::S_V>(0);
    }

    /*
     * =============================================================
     * R015-V001
     *
     * TWO-ROW CONTIGUOUS INPUT DMA
     * =============================================================
     *
     * This function intentionally keeps:
     *
     *   - independent per-row reduction
     *   - independent per-row RMS scalar synchronization
     *   - independent gamma/bias loads
     *   - independent output stores
     *
     * The only major change is:
     *
     *     row0 x + row1 x
     *         one contiguous GM -> UB transfer
     *
     *     row0 residual + row1 residual
     *         one contiguous GM -> UB transfer
     *
     * This occurs in both passes.
     */
    __aicore__ inline void ProcessTwoRows(
        uint64_t firstRow,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t firstOffset =
            firstRow * rowWidth;

        const int32_t width =
            static_cast<int32_t>(
                rowWidth);

        const int32_t totalElems =
            static_cast<int32_t>(
                rowWidth * 2ULL);

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        /*
         * Keep the two scalar slots 32 bytes apart.
         *
         * sumSlots[0] : row 0
         * sumSlots[8] : row 1
         */
        AscendC::LocalTensor<float> rms0 =
            sumSlots;

        AscendC::LocalTensor<float> rms1 =
            sumSlots[8];

        /*
         * =========================================================
         * PASS 1
         *
         * OLD:
         *
         *   Load row0 x
         *   Load row0 residual
         *   ...
         *   Load row1 x
         *   Load row1 residual
         *
         * NEW:
         *
         *   Load row0+row1 x        once
         *   Load row0+row1 residual once
         * =========================================================
         */
        Load(
            xLocal,
            xGm_,
            firstOffset,
            totalElems);

        Load(
            residualLocal,
            residualGm_,
            firstOffset,
            totalElems);

        SyncMTE2ToV();

        float rmsValue0 = 0.0f;
        float rmsValue1 = 0.0f;

        /*
         * -------------------------
         * Row 0 reduction
         * -------------------------
         */
        ToFloat(
            xFp32,
            xLocal,
            width);

        ToFloat(
            residualFp32,
            residualLocal,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            xFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(
            xFp32,
            yFp32,
            yFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * residualFp32 is dead after Add,
         * identical to D baseline.
         */
        AscendC::ReduceSum(
            rms0,
            xFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Muls(
            rms0,
            rms0,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms0,
            rms0,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms0,
            rms0,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToS();

        rmsValue0 =
            rms0.GetValue(0);

        SyncSToV();

        /*
         * Keep the original synchronization discipline.
         */
        SyncVToMTE2();

        /*
         * -------------------------
         * Row 1 reduction
         * -------------------------
         */
        const AscendC::LocalTensor<T>
            xRow1 =
                xLocal[rowWidth];

        const AscendC::LocalTensor<T>
            residualRow1 =
                residualLocal[rowWidth];

        ToFloat(
            xFp32,
            xRow1,
            width);

        ToFloat(
            residualFp32,
            residualRow1,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            xFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(
            xFp32,
            yFp32,
            yFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum(
            rms1,
            xFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Muls(
            rms1,
            rms1,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms1,
            rms1,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms1,
            rms1,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToS();

        rmsValue1 =
            rms1.GetValue(0);

        SyncSToV();

        SyncVToMTE2();

        /*
         * =========================================================
         * PASS 2
         *
         * Again merge only the two x/residual GM input transfers.
         * =========================================================
         */
        Load(
            xLocal,
            xGm_,
            firstOffset,
            totalElems);

        Load(
            residualLocal,
            residualGm_,
            firstOffset,
            totalElems);

        SyncMTE2ToV();

        /*
         * =========================================================
         * Output row 0
         * =========================================================
         */
        ToFloat(
            xFp32,
            xLocal,
            width);

        ToFloat(
            residualFp32,
            residualLocal,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            xFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Duplicate(
            residualFp32,
            rmsValue0,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Div(
            yFp32,
            yFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToMTE2();

        /*
         * Gamma/bias intentionally remain per-row.
         *
         * R014 gamma/bias caching is NOT part of R015-V001.
         */
        Load(
            xLocal,
            gammaGm_,
            0,
            width);

        Load(
            residualLocal,
            biasGm_,
            0,
            width);

        SyncMTE2ToV();

        ToFloat(
            xFp32,
            xLocal,
            width);

        ToFloat(
            residualFp32,
            residualLocal,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(
            yFp32,
            yFp32,
            xFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            yFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            SyncVToMTE3();

            Store(
                outputGm_,
                firstOffset,
                yFp32,
                width);

            SyncMTE3ToV();

        } else {

            AscendC::Cast(
                xLocal,
                yFp32,
                AscendC::RoundMode::CAST_RINT,
                width);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE3();

            Store(
                outputGm_,
                firstOffset,
                xLocal,
                width);

            SyncMTE3ToV();
        }

        SyncVToMTE2();

        /*
         * =========================================================
         * Output row 1
         *
         * Important:
         *
         * gamma/bias/output for row0 only overwrote [0, width).
         * The original row1 inputs are still stored in
         *
         *   xLocal[width ... 2*width)
         *   residualLocal[width ... 2*width)
         * =========================================================
         */
        const AscendC::LocalTensor<T>
            xRow1Pass2 =
                xLocal[rowWidth];

        const AscendC::LocalTensor<T>
            residualRow1Pass2 =
                residualLocal[rowWidth];

        ToFloat(
            xFp32,
            xRow1Pass2,
            width);

        ToFloat(
            residualFp32,
            residualRow1Pass2,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            xFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Duplicate(
            residualFp32,
            rmsValue1,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Div(
            yFp32,
            yFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToMTE2();

        Load(
            xLocal,
            gammaGm_,
            0,
            width);

        Load(
            residualLocal,
            biasGm_,
            0,
            width);

        SyncMTE2ToV();

        ToFloat(
            xFp32,
            xLocal,
            width);

        ToFloat(
            residualFp32,
            residualLocal,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(
            yFp32,
            yFp32,
            xFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(
            yFp32,
            yFp32,
            residualFp32,
            width);

        AscendC::PipeBarrier<PIPE_V>();

        if constexpr (
            AscendC::IsSameType<
                T,
                float>::value) {

            SyncVToMTE3();

            Store(
                outputGm_,
                firstOffset + rowWidth,
                yFp32,
                width);

            SyncMTE3ToV();

        } else {

            AscendC::Cast(
                xLocal,
                yFp32,
                AscendC::RoundMode::CAST_RINT,
                width);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE3();

            Store(
                outputGm_,
                firstOffset + rowWidth,
                xLocal,
                width);

            SyncMTE3ToV();
        }

        SyncVToMTE2();
    }

    /*
     * =============================================================
     * D BASELINE SINGLE-ROW PATH
     *
     * Intentionally unchanged.
     * =============================================================
     */
    __aicore__ inline void ProcessRow(
        uint64_t row,
        uint64_t rowWidth,
        float invRowWidth,
        float epsilon)
    {
        const uint64_t rowOffset =
            row * rowWidth;

        AscendC::LocalTensor<T> xLocal =
            xBuf_.Get<T>();

        AscendC::LocalTensor<T> residualLocal =
            residualBuf_.Get<T>();

        AscendC::LocalTensor<float> xFp32 =
            xFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> residualFp32 =
            residualFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> yFp32 =
            yFp32Buf_.Get<float>();

        AscendC::LocalTensor<float> sumSlots =
            sumBuf_.Get<float>();

        AscendC::LocalTensor<float> partial =
            sumSlots;

        AscendC::LocalTensor<float> rms =
            sumSlots[8];

        AscendC::Duplicate(
            rms,
            0.0f,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        /*
         * PASS 1
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                xFp32,
                yFp32,
                yFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum(
                partial,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                rms,
                rms,
                partial,
                1);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE2();
        }

        AscendC::Muls(
            rms,
            rms,
            invRowWidth,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Adds(
            rms,
            rms,
            epsilon,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Sqrt(
            rms,
            rms,
            1);

        AscendC::PipeBarrier<PIPE_V>();

        SyncVToS();

        const float rmsValue =
            rms.GetValue(0);

        SyncSToV();

        /*
         * PASS 2
         */
        for (uint64_t col = 0;
             col < rowWidth;
             col += kTileElems) {

            const int32_t valid =
                TileLength(
                    rowWidth - col);

            Load(
                xLocal,
                xGm_,
                rowOffset + col,
                valid);

            Load(
                residualLocal,
                residualGm_,
                rowOffset + col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                xFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Duplicate(
                residualFp32,
                rmsValue,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Div(
                yFp32,
                yFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            SyncVToMTE2();

            Load(
                xLocal,
                gammaGm_,
                col,
                valid);

            Load(
                residualLocal,
                biasGm_,
                col,
                valid);

            SyncMTE2ToV();

            ToFloat(
                xFp32,
                xLocal,
                valid);

            ToFloat(
                residualFp32,
                residualLocal,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(
                yFp32,
                yFp32,
                xFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(
                yFp32,
                yFp32,
                residualFp32,
                valid);

            AscendC::PipeBarrier<PIPE_V>();

            if constexpr (
                AscendC::IsSameType<
                    T,
                    float>::value) {

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    yFp32,
                    valid);

                SyncMTE3ToV();

            } else {

                AscendC::Cast(
                    xLocal,
                    yFp32,
                    AscendC::RoundMode::CAST_RINT,
                    valid);

                AscendC::PipeBarrier<PIPE_V>();

                SyncVToMTE3();

                Store(
                    outputGm_,
                    rowOffset + col,
                    xLocal,
                    valid);

                SyncMTE3ToV();
            }

            SyncVToMTE2();
        }
    }

private:
    AscendC::TPipe tpipe;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualBuf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        xFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        residualFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        yFp32Buf_;

    AscendC::TBuf<
        AscendC::TPosition::VECCALC>
        sumBuf_;

    AscendC::GlobalTensor<T> xGm_;
    AscendC::GlobalTensor<T> residualGm_;
    AscendC::GlobalTensor<T> gammaGm_;
    AscendC::GlobalTensor<T> biasGm_;
    AscendC::GlobalTensor<T> outputGm_;
};


template <typename T>
__global__ __vector__ void add_rms_norm_bias_custom(
    GM_ADDR x,
    GM_ADDR residual,
    GM_ADDR gamma,
    GM_ADDR bias,
    GM_ADDR output,
    uint64_t rowCount,
    uint64_t rowWidth,
    uint64_t rowGroup,
    uint64_t groupCount,
    uint32_t blockCount,
    float invRowWidth,
    float epsilon)
{
    AddRmsNormBiasKernel<T> op;

    op.Init(
        x,
        residual,
        gamma,
        bias,
        output);

    op.Process(
        rowCount,
        rowWidth,
        rowGroup,
        groupCount,
        blockCount,
        invRowWidth,
        epsilon);
}


#if !defined(ASCENDC_CPU_DEBUG)

extern "C" void run_kernel(
    GM_ADDR x,
    const TensorGroupInfo& info_x,

    GM_ADDR residual,
    const TensorGroupInfo& info_residual,

    GM_ADDR gamma,
    const TensorGroupInfo& info_gamma,

    GM_ADDR bias,
    const TensorGroupInfo& info_bias,

    GM_ADDR output,
    const TensorGroupInfo& info_output,

    int64_t availableCoreNum,
    aclrtStream stream,
    float epsilon)
{
    if (x == nullptr ||
        residual == nullptr ||
        gamma == nullptr ||
        bias == nullptr ||
        output == nullptr ||

        info_x.tensors == nullptr ||
        info_residual.tensors == nullptr ||
        info_gamma.tensors == nullptr ||
        info_bias.tensors == nullptr ||
        info_output.tensors == nullptr ||

        info_x.numTensors < 1 ||
        info_residual.numTensors < 1 ||
        info_gamma.numTensors < 1 ||
        info_bias.numTensors < 1 ||
        info_output.numTensors < 1) {

        return;
    }

    const TensorInfo& xInfo =
        info_x.tensors[0];

    const TensorInfo& residualInfo =
        info_residual.tensors[0];

    const TensorInfo& gammaInfo =
        info_gamma.tensors[0];

    const TensorInfo& biasInfo =
        info_bias.tensors[0];

    const TensorInfo& outputInfo =
        info_output.tensors[0];

    if (xInfo.shape == nullptr ||
        residualInfo.shape == nullptr ||
        gammaInfo.shape == nullptr ||
        biasInfo.shape == nullptr ||
        outputInfo.shape == nullptr ||

        xInfo.numDims < 1 ||

        residualInfo.numDims !=
            xInfo.numDims ||

        outputInfo.numDims !=
            xInfo.numDims ||

        gammaInfo.numDims != 1 ||
        biasInfo.numDims != 1 ||

        xInfo.dtype !=
            residualInfo.dtype ||

        xInfo.dtype !=
            gammaInfo.dtype ||

        xInfo.dtype !=
            biasInfo.dtype ||

        xInfo.dtype !=
            outputInfo.dtype) {

        return;
    }

    if (xInfo.dtype != 0 &&
        xInfo.dtype != 1 &&
        xInfo.dtype != 2) {

        return;
    }

    const int64_t widthSigned =
        xInfo.shape[
            xInfo.numDims - 1];

    if (widthSigned <= 0 ||
        gammaInfo.shape[0] !=
            widthSigned ||
        biasInfo.shape[0] !=
            widthSigned) {

        return;
    }

    const uint64_t maxU64 =
        ~static_cast<uint64_t>(0);

    uint64_t rowCount = 1;

    for (int64_t dim = 0;
         dim < xInfo.numDims - 1;
         ++dim) {

        const int64_t extentSigned =
            xInfo.shape[dim];

        if (extentSigned <= 0) {
            return;
        }

        if (residualInfo.shape[dim] !=
                extentSigned ||
            outputInfo.shape[dim] !=
                extentSigned) {

            return;
        }

        const uint64_t extent =
            static_cast<uint64_t>(
                extentSigned);

        if (rowCount >
            maxU64 / extent) {

            return;
        }

        rowCount *= extent;
    }

    if (residualInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        outputInfo.shape[
            xInfo.numDims - 1] !=
            widthSigned ||

        rowCount == 0) {

        return;
    }

    const uint64_t rowWidth =
        static_cast<uint64_t>(
            widthSigned);

    const uint64_t elementBytes =
        xInfo.dtype == 0
            ? 4ULL
            : 2ULL;

    if (rowWidth >
            maxU64 / elementBytes ||
        rowCount >
            maxU64 / rowWidth) {

        return;
    }

    const uint64_t rowBytes =
        rowWidth * elementBytes;

    uint64_t gcdA = rowBytes;
    uint64_t gcdB = 32;

    while (gcdB != 0) {

        const uint64_t remainder =
            gcdA % gcdB;

        gcdA = gcdB;
        gcdB = remainder;
    }

    const uint64_t rowGroup =
        32 / gcdA;

    const uint64_t groupCount =
        rowCount / rowGroup +
        (rowCount % rowGroup == 0
             ? 0
             : 1);

    uint64_t requestedBlocks =
        availableCoreNum > 0
            ? static_cast<uint64_t>(
                  availableCoreNum)
            : 1ULL;

    if (requestedBlocks >
        groupCount) {

        requestedBlocks =
            groupCount;
    }

    const uint64_t maxBlockCount =
        static_cast<uint64_t>(
            0xffffffffU);

    if (requestedBlocks >
        maxBlockCount) {

        requestedBlocks =
            maxBlockCount;
    }

    if (requestedBlocks == 0) {
        requestedBlocks = 1;
    }

    const uint32_t blockCount =
        static_cast<uint32_t>(
            requestedBlocks);

    const float invRowWidth =
        1.0f /
        static_cast<float>(
            widthSigned);

    if (xInfo.dtype == 0) {

        add_rms_norm_bias_custom<float>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else if (xInfo.dtype == 1) {

        add_rms_norm_bias_custom<half>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);

    } else {

        add_rms_norm_bias_custom<bfloat16_t>
            <<<blockCount, nullptr, stream>>>(
                x,
                residual,
                gamma,
                bias,
                output,
                rowCount,
                rowWidth,
                rowGroup,
                groupCount,
                blockCount,
                invRowWidth,
                epsilon);
    }
}

#endif
